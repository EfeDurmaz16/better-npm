// crates/better-core/src/sandbox/permissions.rs

use serde::{Deserialize, Serialize};
use std::path::PathBuf;

/// Permission set for a sandboxed script execution.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SandboxPermissions {
    /// Filesystem access rules
    pub fs: FsPermissions,
    /// Network access rules
    pub net: NetPermissions,
    /// Environment variable access
    pub env: EnvPermissions,
    /// Child process spawning
    pub process: ProcessPermissions,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct FsPermissions {
    /// Read access paths (absolute). Empty = no read.
    pub read: Vec<PathBuf>,
    /// Write access paths (absolute). Empty = no write.
    pub write: Vec<PathBuf>,
    /// Whether to allow tmp dir access
    pub allow_tmp: bool,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct NetPermissions {
    /// Allow outbound network. Default: false.
    pub allow: bool,
    /// Allowed hosts (if allow=true). Empty = all hosts.
    pub allowed_hosts: Vec<String>,
    /// Allowed ports. Empty = all ports.
    pub allowed_ports: Vec<u16>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct EnvPermissions {
    /// Allow reading env vars. Default: false.
    pub allow_read: bool,
    /// Specific env vars allowed. Empty = all (if allow_read=true).
    pub allowed_vars: Vec<String>,
    /// Env vars that are NEVER exposed (e.g., secrets).
    pub blocked_vars: Vec<String>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ProcessPermissions {
    /// Allow spawning child processes. Default: false.
    pub allow: bool,
    /// Allowed executables. Empty = all (if allow=true).
    pub allowed_executables: Vec<String>,
}

impl SandboxPermissions {
    /// Default: locked down. No network, no env, fs only within package dir.
    pub fn default_for_package(package_dir: &Path) -> Self {
        Self {
            fs: FsPermissions {
                read: vec![package_dir.to_path_buf()],
                write: vec![package_dir.to_path_buf()],
                allow_tmp: true,
            },
            net: NetPermissions {
                allow: false,
                allowed_hosts: vec![],
                allowed_ports: vec![],
            },
            env: EnvPermissions {
                allow_read: false,
                allowed_vars: vec![
                    "PATH".to_string(),
                    "HOME".to_string(),
                    "NODE_ENV".to_string(),
                ],
                blocked_vars: vec![
                    "NPM_TOKEN".to_string(),
                    "GITHUB_TOKEN".to_string(),
                    "AWS_SECRET_ACCESS_KEY".to_string(),
                    "DATABASE_URL".to_string(),
                ],
            },
            process: ProcessPermissions {
                allow: true,
                allowed_executables: vec![
                    "node".to_string(),
                    "sh".to_string(),
                    "bash".to_string(),
                ],
            },
        }
    }

    /// Fully permissive (for --no-sandbox fallback).
    pub fn unrestricted() -> Self {
        Self {
            fs: FsPermissions {
                read: vec![PathBuf::from("/")],
                write: vec![PathBuf::from("/")],
                allow_tmp: true,
            },
            net: NetPermissions {
                allow: true,
                allowed_hosts: vec![],
                allowed_ports: vec![],
            },
            env: EnvPermissions {
                allow_read: true,
                allowed_vars: vec![],
                blocked_vars: vec![],
            },
            process: ProcessPermissions {
                allow: true,
                allowed_executables: vec![],
            },
        }
    }
}

/// .better-scripts.json v2 format -- per-package permission overrides.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ScriptPolicyV2 {
    pub version: u32,
    /// Default policy: "sandbox" (default) or "allow"
    pub default_policy: String,
    /// Per-package permission overrides
    pub packages: std::collections::HashMap<String, PackageScriptPolicy>,
    /// Trusted scopes (all packages in scope get "allow")
    pub trusted_scopes: Vec<String>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PackageScriptPolicy {
    /// "sandbox", "allow", or "block"
    pub policy: String,
    /// Custom permissions (only for policy="sandbox")
    pub permissions: Option<SandboxPermissions>,
    /// Reason for the policy
    pub reason: Option<String>,
}

// crates/better-core/src/sandbox/macos.rs

use super::permissions::SandboxPermissions;
use std::path::Path;
use std::process::Command;

/// Generate a macOS sandbox-exec profile from SandboxPermissions.
///
/// sandbox-exec uses Apple's Seatbelt SBPL (Scheme-based Policy Language).
/// Ref: https://reverse.put.as/wp-content/uploads/2011/09/Apple-Sandbox-Guide-v1.0.pdf
pub fn generate_seatbelt_profile(perms: &SandboxPermissions) -> String {
    let mut profile = String::new();
    profile.push_str("(version 1)\n");
    profile.push_str("(deny default)\n");

    // Always allow process execution within sandbox
    profile.push_str("(allow process-exec)\n");
    profile.push_str("(allow process-fork)\n");
    profile.push_str("(allow sysctl-read)\n");
    profile.push_str("(allow mach-lookup)\n");

    // Filesystem: read
    for read_path in &perms.fs.read {
        profile.push_str(&format!(
            "(allow file-read* (subpath \"{}\"))\n",
            read_path.display()
        ));
    }

    // Filesystem: write
    for write_path in &perms.fs.write {
        profile.push_str(&format!(
            "(allow file-write* (subpath \"{}\"))\n",
            write_path.display()
        ));
    }

    // Tmp access
    if perms.fs.allow_tmp {
        profile.push_str("(allow file-read* file-write* (subpath \"/tmp\"))\n");
        profile.push_str(
            "(allow file-read* file-write* (subpath \"/private/tmp\"))\n",
        );
    }

    // Always allow reading system libs and node
    profile.push_str("(allow file-read* (subpath \"/usr/lib\"))\n");
    profile.push_str("(allow file-read* (subpath \"/usr/local\"))\n");
    profile.push_str("(allow file-read* (subpath \"/opt/homebrew\"))\n");
    profile.push_str("(allow file-read* (subpath \"/System\"))\n");
    profile.push_str(
        "(allow file-read* (subpath \"/Library/Frameworks\"))\n",
    );

    // Network
    if perms.net.allow {
        profile.push_str("(allow network*)\n");
    } else {
        // Deny all network by default (inherited from deny default)
        // Allow local IPC (needed for node)
        profile.push_str("(allow network-inbound (local ip))\n");
        profile.push_str("(allow network-outbound (local ip))\n");
    }

    profile
}

/// Execute a script inside a macOS sandbox.
pub fn execute_sandboxed_macos(
    command: &str,
    args: &[&str],
    working_dir: &Path,
    perms: &SandboxPermissions,
    env_vars: &[(String, String)],
) -> Result<SandboxResult, SandboxError> {
    let profile = generate_seatbelt_profile(perms);

    // Write profile to temp file
    let profile_path = std::env::temp_dir().join(format!(
        "better-sandbox-{}.sb",
        std::process::id()
    ));
    std::fs::write(&profile_path, &profile)?;

    // Build filtered env
    let filtered_env: Vec<(String, String)> = if perms.env.allow_read {
        env_vars
            .iter()
            .filter(|(k, _)| !perms.env.blocked_vars.contains(k))
            .filter(|(k, _)| {
                perms.env.allowed_vars.is_empty()
                    || perms.env.allowed_vars.contains(k)
            })
            .cloned()
            .collect()
    } else {
        // Only pass explicitly allowed vars
        env_vars
            .iter()
            .filter(|(k, _)| perms.env.allowed_vars.contains(k))
            .cloned()
            .collect()
    };

    let output = Command::new("sandbox-exec")
        .arg("-f")
        .arg(&profile_path)
        .arg(command)
        .args(args)
        .current_dir(working_dir)
        .env_clear()
        .envs(filtered_env)
        .output()?;

    // Clean up profile
    let _ = std::fs::remove_file(&profile_path);

    Ok(SandboxResult {
        exit_code: output.status.code().unwrap_or(-1),
        stdout: String::from_utf8_lossy(&output.stdout).to_string(),
        stderr: String::from_utf8_lossy(&output.stderr).to_string(),
        sandbox_violations: parse_sandbox_violations(&output.stderr),
    })
}

#[derive(Debug, Clone)]
pub struct SandboxResult {
    pub exit_code: i32,
    pub stdout: String,
    pub stderr: String,
    pub sandbox_violations: Vec<String>,
}

fn parse_sandbox_violations(stderr: &str) -> Vec<String> {
    stderr
        .lines()
        .filter(|l| {
            l.contains("deny")
                || l.contains("Sandbox")
                || l.contains("sandbox")
        })
        .map(|l| l.to_string())
        .collect()
}

#[derive(Debug)]
pub enum SandboxError {
    Io(std::io::Error),
    ProfileGeneration(String),
    SandboxExecNotFound,
}

impl From<std::io::Error> for SandboxError {
    fn from(e: std::io::Error) -> Self {
        Self::Io(e)
    }
}

// crates/better-core/src/sandbox/linux.rs

use super::permissions::SandboxPermissions;
use std::path::Path;
use std::process::Command;

/// Linux sandboxing using Landlock LSM (kernel 5.13+) with
/// seccomp-bpf fallback, and unshare namespace as last resort.
///
/// Priority:
/// 1. Landlock LSM (best: fine-grained fs access control)
/// 2. seccomp-bpf (good: syscall filtering)
/// 3. unshare namespace (fallback: network/mount isolation)

/// Check available sandboxing mechanisms on this Linux system.
pub fn detect_sandbox_capability() -> LinuxSandboxCapability {
    let landlock = check_landlock();
    let seccomp = check_seccomp();
    let unshare = check_unshare();

    LinuxSandboxCapability {
        landlock,
        seccomp,
        unshare,
    }
}

#[derive(Debug, Clone)]
pub struct LinuxSandboxCapability {
    pub landlock: Option<u32>, // ABI version, None if unavailable
    pub seccomp: bool,
    pub unshare: bool,
}

/// Execute a script inside a Linux sandbox.
pub fn execute_sandboxed_linux(
    command: &str,
    args: &[&str],
    working_dir: &Path,
    perms: &SandboxPermissions,
    env_vars: &[(String, String)],
) -> Result<SandboxResult, SandboxError> {
    let cap = detect_sandbox_capability();

    if let Some(abi_version) = cap.landlock {
        return execute_with_landlock(
            command,
            args,
            working_dir,
            perms,
            env_vars,
            abi_version,
        );
    }

    if cap.seccomp {
        return execute_with_seccomp(
            command, args, working_dir, perms, env_vars,
        );
    }

    if cap.unshare {
        return execute_with_unshare(
            command, args, working_dir, perms, env_vars,
        );
    }

    Err(SandboxError::NoSandboxAvailable)
}

/// Landlock sandbox: restrict filesystem access at kernel level.
///
/// Uses the landlock_create_ruleset, landlock_add_rule, and
/// landlock_restrict_self syscalls via the `landlock` crate.
fn execute_with_landlock(
    command: &str,
    args: &[&str],
    working_dir: &Path,
    perms: &SandboxPermissions,
    env_vars: &[(String, String)],
    _abi_version: u32,
) -> Result<SandboxResult, SandboxError> {
    // Build filtered env
    let filtered_env = filter_env(env_vars, perms);

    // Use unshare for network isolation + Landlock for fs isolation
    let mut cmd = Command::new("unshare");
    if !perms.net.allow {
        cmd.arg("--net");
    }
    cmd.arg("--")
        .arg(command)
        .args(args)
        .current_dir(working_dir)
        .env_clear()
        .envs(filtered_env);

    let output = cmd.output()?;

    Ok(SandboxResult {
        exit_code: output.status.code().unwrap_or(-1),
        stdout: String::from_utf8_lossy(&output.stdout).to_string(),
        stderr: String::from_utf8_lossy(&output.stderr).to_string(),
        sandbox_violations: vec![],
    })
}

/// seccomp-bpf: filter syscalls.
fn execute_with_seccomp(
    command: &str,
    args: &[&str],
    working_dir: &Path,
    perms: &SandboxPermissions,
    env_vars: &[(String, String)],
) -> Result<SandboxResult, SandboxError> {
    // Uses seccomp-bpf via the `seccompiler` crate or a helper binary.
    // Blocks: connect() (no network), open() outside allowed paths, etc.
    execute_with_unshare(command, args, working_dir, perms, env_vars)
}

/// unshare namespace: basic isolation.
fn execute_with_unshare(
    command: &str,
    args: &[&str],
    working_dir: &Path,
    perms: &SandboxPermissions,
    env_vars: &[(String, String)],
) -> Result<SandboxResult, SandboxError> {
    let filtered_env = filter_env(env_vars, perms);

    let mut cmd_args = vec!["--mount"];
    if !perms.net.allow {
        cmd_args.push("--net");
    }
    cmd_args.push("--");

    let output = Command::new("unshare")
        .args(&cmd_args)
        .arg(command)
        .args(args)
        .current_dir(working_dir)
        .env_clear()
        .envs(filtered_env)
        .output()?;

    Ok(SandboxResult {
        exit_code: output.status.code().unwrap_or(-1),
        stdout: String::from_utf8_lossy(&output.stdout).to_string(),
        stderr: String::from_utf8_lossy(&output.stderr).to_string(),
        sandbox_violations: vec![],
    })
}

fn filter_env(
    env_vars: &[(String, String)],
    perms: &SandboxPermissions,
) -> Vec<(String, String)> {
    if perms.env.allow_read {
        env_vars
            .iter()
            .filter(|(k, _)| !perms.env.blocked_vars.contains(k))
            .cloned()
            .collect()
    } else {
        env_vars
            .iter()
            .filter(|(k, _)| perms.env.allowed_vars.contains(k))
            .cloned()
            .collect()
    }
}

fn check_landlock() -> Option<u32> {
    std::fs::read_to_string("/sys/kernel/security/landlock/status")
        .ok()
        .and_then(|s| s.trim().parse().ok())
}

fn check_seccomp() -> bool {
    std::fs::read_to_string("/proc/self/status")
        .ok()
        .map_or(false, |s| s.contains("Seccomp:"))
}

fn check_unshare() -> bool {
    Command::new("unshare")
        .arg("--help")
        .output()
        .map_or(false, |o| o.status.success())
}

use super::macos::SandboxResult;

#[derive(Debug)]
pub enum SandboxError {
    Io(std::io::Error),
    NoSandboxAvailable,
}

impl From<std::io::Error> for SandboxError {
    fn from(e: std::io::Error) -> Self {
        Self::Io(e)
    }
}

use std::path::Path;

/// Platform-agnostic sandbox executor.
pub fn execute_sandboxed(
    command: &str,
    args: &[&str],
    working_dir: &Path,
    perms: &SandboxPermissions,
) -> Result<SandboxResult, SandboxError> {
    let env_vars: Vec<(String, String)> = std::env::vars().collect();

    #[cfg(target_os = "macos")]
    {
        super::macos::execute_sandboxed_macos(
            command,
            args,
            working_dir,
            perms,
            &env_vars,
        )
    }

    #[cfg(target_os = "linux")]
    {
        super::linux::execute_sandboxed_linux(
            command,
            args,
            working_dir,
            perms,
            &env_vars,
        )
    }

    #[cfg(not(any(target_os = "macos", target_os = "linux")))]
    {
        // Windows / other: no sandbox, warn user
        eprintln!(
            "WARNING: Sandboxing not available on this platform. \
             Running script without restrictions."
        );
        let output = std::process::Command::new(command)
            .args(args)
            .current_dir(working_dir)
            .output()?;
        Ok(SandboxResult {
            exit_code: output.status.code().unwrap_or(-1),
            stdout: String::from_utf8_lossy(&output.stdout).to_string(),
            stderr: String::from_utf8_lossy(&output.stderr).to_string(),
            sandbox_violations: vec![],
        })
    }
}

use serde::Serialize;
use chrono::Utc;

/// CycloneDX 1.6 BOM generator.
///
/// Spec: https://cyclonedx.org/docs/1.6/
/// Key additions in 1.6: vulnerability section, formulation, machine learning.
///
/// We generate:
/// - metadata (tool, component, manufacture, timestamp)
/// - components (all deps with purl, hashes, licenses)
/// - dependencies (graph relationships)
/// - vulnerabilities (from audit, with VEX status)

#[derive(Debug, Serialize)]
#[serde(rename_all = "camelCase")]
pub struct CycloneDxBom {
    pub bom_format: String, // "CycloneDX"
    pub spec_version: String, // "1.6"
    pub serial_number: String, // urn:uuid:...
    pub version: u32,
    pub metadata: BomMetadata,
    pub components: Vec<BomComponent>,
    pub dependencies: Vec<BomDependency>,
    #[serde(skip_serializing_if = "Vec::is_empty")]
    pub vulnerabilities: Vec<BomVulnerability>,
}

#[derive(Debug, Serialize)]
#[serde(rename_all = "camelCase")]
pub struct BomMetadata {
    pub timestamp: String,
    pub tools: Vec<BomTool>,
    pub component: BomComponent, // The root project
    #[serde(skip_serializing_if = "Option::is_none")]
    pub manufacture: Option<BomOrganization>,
}

#[derive(Debug, Serialize)]

    pub fn generate(
        project: &ProjectInfo,
        lock_graph: &LockGraph,
        dep_classifier: &DepClassifier,
    ) -> Self {
        let packages: Vec<SpdxPackage> = lock_graph
            .packages()
            .map(|(key, info)| {
                let spdxid = format!(
                    "SPDXRef-Package-{}-{}",
                    sanitize_spdx_id(&info.name),
                    sanitize_spdx_id(&info.version)
                );
                SpdxPackage {
                    spdxid,
                    name: info.name.clone(),
                    version_info: info.version.clone(),
                    download_location: info
                        .resolved_url
                        .clone()
                        .unwrap_or_else(|| "NOASSERTION".to_string()),
                    files_analyzed: false,
                    license_concluded: info.license.clone(),
                    license_declared: info.license.clone(),
                    copyright_text: "NOASSERTION".to_string(),
                    external_refs: Some(vec![SpdxExternalRef {
                        reference_category: "PACKAGE-MANAGER".to_string(),
                        reference_type: "purl".to_string(),
                        reference_locator: format!(
                            "pkg:npm/{}@{}",
                            info.name, info.version
                        ),
                    }]),
                }
            })
            .collect();

        // Build relationships with context-aware types
        let mut relationships = vec![SpdxRelationship {
            spdx_element_id: "SPDXRef-DOCUMENT".to_string(),
            relationship_type: "DESCRIBES".to_string(),
            related_spdx_element: "SPDXRef-RootPackage".to_string(),
        }];

        for (key, info) in lock_graph.packages() {
            let parent_id = format!(
                "SPDXRef-Package-{}-{}",
                sanitize_spdx_id(&info.name),
                sanitize_spdx_id(&info.version)
            );

            for (dep_name, dep_version) in &info.dependencies {
                let child_id = format!(
                    "SPDXRef-Package-{}-{}",
                    sanitize_spdx_id(dep_name),
                    sanitize_spdx_id(dep_version)
                );

                let dep_key = format!("{}@{}", dep_name, dep_version);
                let ctx = dep_classifier.get(&dep_key);
                let rel_type = match ctx {
                    DepContext::Dev => "DEV_DEPENDENCY_OF",
                    DepContext::Optional => "OPTIONAL_DEPENDENCY_OF",
                    DepContext::Build => "BUILD_DEPENDENCY_OF",
                    _ => "DEPENDS_ON",
                };

                relationships.push(SpdxRelationship {
                    spdx_element_id: parent_id.clone(),
                    relationship_type: rel_type.to_string(),
                    related_spdx_element: child_id,
                });
            }
        }

        Self {
            spdx_version: "SPDX-2.3".to_string(),
            data_license: "CC0-1.0".to_string(),
            spdxid: "SPDXRef-DOCUMENT".to_string(),
            name: format!("{}-sbom", project.name),
            document_namespace: format!(
                "https://better.sh/spdx/{}/{}",
                project.name,
                uuid::Uuid::new_v4()
            ),
            creation_info: SpdxCreationInfo {
                created: Utc::now().to_rfc3339(),
                creators: vec![format!(
                    "Tool: better-{}",
                    env!("CARGO_PKG_VERSION")
                )],
                license_list_version: "3.22".to_string(),
            },
            packages,
            relationships,
        }
    }
}

fn sanitize_spdx_id(s: &str) -> String {
    s.chars()
        .map(|c| if c.is_alphanumeric() || c == '-' || c == '.' { c } else { '-' })
        .collect()
}

/// OpenVEX document format.
/// Spec: https://github.com/openvex/spec
#[derive(Debug, Serialize)]
pub struct VexDocument {
    #[serde(rename = "@context")]
    pub context: String,
    #[serde(rename = "@id")]
    pub id: String,
    pub author: String,
    pub role: String,
    pub timestamp: String,
    pub version: u32,
    pub statements: Vec<VexStatement>,
}

#[derive(Debug, Serialize)]
pub struct VexStatement {
    pub vulnerability: VexVulnerability,
    pub products: Vec<VexProduct>,
    pub status: String,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub justification: Option<String>,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub impact_statement: Option<String>,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub action_statement: Option<String>,
}

#[derive(Debug, Serialize)]
pub struct VexVulnerability {
    pub name: String, // CVE or GHSA ID
}

#[derive(Debug, Serialize)]
pub struct VexProduct {
    #[serde(rename = "@id")]
    pub id: String, // purl
}

impl VexDocument {
    /// Generate VEX from audit results + allow-list waivers.
    ///
    /// - Waived vulns -> "not_affected" with justification from waiver reason
    /// - Dev-only vulns with context scoring -> "not_affected" with justification
    /// - Active vulns -> "affected" with action_statement (upgrade path)
    pub fn from_audit(
        vulns: &[ScoredVuln],
        waivers: &AuditAllowList,
        _project_purl: &str,
    ) -> Self {
        let statements: Vec<VexStatement> = vulns
            .iter()
            .map(|v| {
                let waiver_result = waivers.is_waived(
                    &v.id,
                    &format!("{}@{}", v.package_name, v.package_version),
                    chrono::Utc::now().date_naive(),
                );

                match waiver_result {
                    WaiverResult::Waived(w) => VexStatement {
                        vulnerability: VexVulnerability {
                            name: v.id.clone(),
                        },
                        products: vec![VexProduct {
                            id: format!(
                                "pkg:npm/{}@{}",
                                v.package_name, v.package_version
                            ),
                        }],
                        status: "not_affected".to_string(),
                        justification: Some(
                            "vulnerable_code_not_in_execute_path"
                                .to_string(),
                        ),
                        impact_statement: Some(w.reason.clone()),
                        action_statement: None,
                    },
                    _ => VexStatement {
                        vulnerability: VexVulnerability {
                            name: v.id.clone(),
                        },
                        products: vec![VexProduct {
                            id: format!(
                                "pkg:npm/{}@{}",
                                v.package_name, v.package_version
                            ),
                        }],
                        status: "affected".to_string(),
                        justification: None,
                        impact_statement: None,
                        action_statement: v
                            .fix_available
                            .as_ref()
                            .map(|f| format!("Upgrade to {}", f)),
                    },
                }
            })
            .collect();

        Self {
            context: "https://openvex.dev/ns/v0.2.0".to_string(),
            id: format!(
                "https://better.sh/vex/{}",
                uuid::Uuid::new_v4()
            ),
            author: "better".to_string(),
            role: "tool".to_string(),
            timestamp: chrono::Utc::now().to_rfc3339(),
            version: 1,
            statements,
        }
    }
}

use serde::{Deserialize, Serialize};

/// Policy Engine v2 rule definitions.
/// Each rule has: id, enabled flag, threshold, severity.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PolicyRuleV2 {
    pub id: String,
    pub enabled: bool,
    pub severity: RuleSeverity,
    pub config: RuleConfig,
}

#[derive(Debug, Clone, Copy, Serialize, Deserialize)]
#[serde(rename_all = "lowercase")]
pub enum RuleSeverity {
    Error,   // Exit 1
    Warning, // Warn but continue
    Info,    // Informational only
}

#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(tag = "type")]
pub enum RuleConfig {
    /// Maximum total install size in bytes.
    /// Prevents bloated dependency trees.
    #[serde(rename = "max-install-size")]
    MaxInstallSize {
        max_bytes: u64,
        include_dev: bool,
    },

    /// Minimum number of maintainers for a package.
    /// Bus factor protection.
    #[serde(rename = "min-maintainers")]
    MinMaintainers {
        min: u32,
        exclude_scopes: Vec<String>,
    },

    /// Minimum age since first publish (in days).
    /// Anti-typosquat: new packages are suspicious.
    #[serde(rename = "min-publish-age")]
    MinPublishAge {
        min_days: u32,
        exclude_scopes: Vec<String>,
    },

    /// Require packages to have 2FA enabled for publishing.
    #[serde(rename = "require-2fa")]
    Require2fa {
        exclude_scopes: Vec<String>,
    },

    /// Require packages to have source repository link.
    #[serde(rename = "require-source")]
    RequireSource {
        require_url: bool,
        require_directory: bool,
    },

    /// Maximum number of direct dependencies.
    #[serde(rename = "max-direct-deps")]
    MaxDirectDeps {
        max: u32,
    },

    /// Maximum dependency tree depth.
    #[serde(rename = "max-depth")]
    MaxDepth {
        max: u32,
    },

    /// Require packages to have a LICENSE file.
    #[serde(rename = "require-license")]
    RequireLicense {},

    /// Block packages with known vulnerabilities above threshold.
    #[serde(rename = "no-known-vulns")]
    NoKnownVulns {
        min_severity: String,
    },
}

/// Evaluate a single rule against package metadata.
pub fn evaluate_rule(
    rule: &PolicyRuleV2,
    context: &PolicyContext,
) -> RuleResult {
    if !rule.enabled {
        return RuleResult::Skipped;
    }

    match &rule.config {
        RuleConfig::MaxInstallSize {
            max_bytes,
            include_dev,
        } => {
            let total = if *include_dev {
                context.total_size_bytes
            } else {
                context.prod_size_bytes
            };
            if total > *max_bytes {
                RuleResult::Violation {
                    rule_id: rule.id.clone(),
                    severity: rule.severity,
                    message: format!(
                        "Install size {} exceeds max {}",
                        format_bytes(total),
                        format_bytes(*max_bytes)
                    ),
                    remediation: Some(
                        "Remove unused dependencies or use lighter alternatives"
                            .to_string(),
                    ),
                }
            } else {
                RuleResult::Pass
            }
        }

        RuleConfig::MinMaintainers {
            min,
            exclude_scopes,
        } => {
            let violations: Vec<String> = context
                .packages
                .iter()
                .filter(|p| {
                    !exclude_scopes.iter().any(|s| p.name.starts_with(s))
                })
                .filter(|p| p.maintainer_count < *min)
                .map(|p| {
                    format!(
                        "{}: {} maintainer(s) (min: {})",
                        p.name, p.maintainer_count, min
                    )
                })
                .collect();

            if violations.is_empty() {
                RuleResult::Pass
            } else {
                RuleResult::Violation {
                    rule_id: rule.id.clone(),
                    severity: rule.severity,
                    message: format!(
                        "{} package(s) below min maintainers",
                        violations.len()
                    ),
                    remediation: Some(
                        "Consider alternatives with more maintainers"
                            .to_string(),
                    ),
                }
            }
        }

        RuleConfig::MinPublishAge {
            min_days,
            exclude_scopes,
        } => {
            let violations: Vec<String> = context
                .packages
                .iter()
                .filter(|p| {
                    !exclude_scopes.iter().any(|s| p.name.starts_with(s))
                })
                .filter(|p| p.days_since_first_publish < *min_days)
                .map(|p| {
                    format!(
                        "{}: published {} days ago (min: {})",
                        p.name, p.days_since_first_publish, min_days
                    )
                })
                .collect();

            if violations.is_empty() {
                RuleResult::Pass
            } else {
                RuleResult::Violation {
                    rule_id: rule.id.clone(),
                    severity: rule.severity,
                    message: format!(
                        "{} package(s) below min publish age",
                        violations.len()
                    ),
                    remediation: Some(
                        "Newly published packages may be typosquats. \
                         Review carefully."
                            .to_string(),
                    ),
                }
            }
        }

        RuleConfig::MaxDirectDeps { max } => {
            let count = context.direct_dep_count;
            if count > *max {
                RuleResult::Violation {
                    rule_id: rule.id.clone(),
                    severity: rule.severity,
                    message: format!(
                        "{} direct deps exceeds max {}",
                        count, max
                    ),
                    remediation: Some(
                        "Consolidate dependencies or split into \
                         workspace packages"
                            .to_string(),
                    ),
                }
            } else {
                RuleResult::Pass
            }
        }

        // Other rules follow same pattern
        _ => RuleResult::Skipped,
    }
}

    pub max_score: f64,    // 100

    pub warnings: Vec<RuleResult>,
    pub info: Vec<RuleResult>,

    pub fn new(rules: Vec<PolicyRuleV2>) -> Self {
        Self { rules }
    }

    pub fn evaluate(&self, context: &PolicyContext) -> PolicyReport {
        let mut errors = Vec::new();
        let mut warnings = Vec::new();
        let mut info = Vec::new();
        let mut passed = 0;
        let total = self.rules.iter().filter(|r| r.enabled).count();

        for rule in &self.rules {
            let result = evaluate_rule(rule, context);
            match &result {
                RuleResult::Pass => passed += 1,
                RuleResult::Skipped => {}
                RuleResult::Violation { severity, .. } => match severity {
                    RuleSeverity::Error => errors.push(result),
                    RuleSeverity::Warning => warnings.push(result),
                    RuleSeverity::Info => info.push(result),
                },
            }
        }

        let score = if total > 0 {
            (passed as f64 / total as f64) * 100.0
        } else {
            100.0
        };

        PolicyReport {
            score,
            max_score: 100.0,
            errors,
            warnings,
            info,
            passed,
            total,
        }
    }

use serde::{Deserialize, Serialize};

/// npm provenance attestation structure.
/// npm uses Sigstore (Fulcio + Rekor) since npm@9.5.0.
///
/// Attestation bundle contains:
/// - DSSE envelope with in-toto SLSA provenance predicate
/// - Fulcio certificate (X.509, short-lived, tied to OIDC identity)
/// - Rekor transparency log entry
///
/// We verify:
/// 1. Signature validity (DSSE envelope)
/// 2. Fulcio certificate chain (root -> intermediate -> leaf)
/// 3. Rekor inclusion proof (entry exists in transparency log)
/// 4. OIDC identity matches expected publisher

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ProvenanceAttestation {
    pub bundle: DsseBundle,
    pub source_repository: Option<String>,
    pub source_commit: Option<String>,
    pub build_trigger: Option<String>,
    pub builder_id: Option<String>,
    pub issuer: Option<String>,
    pub subject_alternative_name: Option<String>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct DsseBundle {
    pub payload_type: String,
    pub payload: String, // base64-encoded SLSA provenance
    pub signatures: Vec<DsseSignature>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct DsseSignature {
    pub keyid: Option<String>,
    pub sig: String, // base64-encoded signature
}

/// Verify provenance for a package.
///
/// Steps:
/// 1. Fetch attestation from npm registry (/_attestations endpoint)
/// 2. Parse DSSE bundle
/// 3. Verify signature against Fulcio cert
/// 4. Verify Fulcio cert chain against Sigstore root
/// 5. Check Rekor transparency log for inclusion
/// 6. Extract and return provenance metadata
pub async fn verify_provenance(
    package_name: &str,
    package_version: &str,
    registry_url: &str,
) -> Result<ProvenanceResult, ProvenanceError> {
    // Step 1: Fetch attestation
    let attestation_url = format!(
        "{}/-/npm/v1/attestations/{}@{}",
        registry_url, package_name, package_version
    );
    let attestation_response =
        fetch_attestation(&attestation_url).await?;

    if attestation_response.attestations.is_empty() {
        return Ok(ProvenanceResult::NoAttestation);
    }

    // Step 2: Parse DSSE bundle
    let bundle = &attestation_response.attestations[0];
    let _payload = parse_dsse_payload(&bundle.bundle)?;

    // Step 3: Verify signature
    let sig_valid = verify_dsse_signature(&bundle.bundle)?;
    if !sig_valid {
        return Ok(ProvenanceResult::InvalidSignature);
    }

    // Step 4: Verify Fulcio certificate chain
    let cert_valid = verify_fulcio_chain(&bundle.bundle)?;
    if !cert_valid {
        return Ok(ProvenanceResult::InvalidCertificate);
    }

    // Step 5: Check Rekor log
    let rekor_valid =
        verify_rekor_entry(&bundle.bundle).await?;
    if !rekor_valid {
        return Ok(ProvenanceResult::NotInTransparencyLog);
    }

    // Step 6: Extract provenance metadata
    let attestation = extract_provenance(&bundle)?;

    Ok(ProvenanceResult::Verified(attestation))
}

#[derive(Debug)]
pub enum ProvenanceResult {
    Verified(ProvenanceAttestation),
    NoAttestation,
    InvalidSignature,
    InvalidCertificate,
    NotInTransparencyLog,
}

#[derive(Debug)]
pub enum ProvenanceError {
    NetworkError(String),
    ParseError(String),
    CryptoError(String),
}

// crates/better-core/src/firewall/typosquat.rs

/// Typosquat detection using Levenshtein distance and common
/// typosquatting patterns against a list of popular packages.
///
/// Patterns detected:
/// 1. Character swap: "lod ash" -> "lodash"
/// 2. Missing char: "lodas" -> "lodash"
/// 3. Extra char: "lodassh" -> "lodash"
/// 4. Homoglyph: "l0dash" -> "lodash"
/// 5. Scope confusion: "@lodash/core" vs "lodash"
/// 6. Hyphen/underscore swap: "lodash_fp" -> "lodash-fp"

/// Top 1000 most popular npm packages (loaded from embedded data).
const POPULAR_PACKAGES: &str = include_str!("../../data/popular-packages.txt");

pub struct TyposquatResult {
    pub package_name: String,
    pub is_suspicious: bool,
    pub similar_to: Option<String>,
    pub distance: usize,
    pub pattern: Option<TyposquatPattern>,
    pub confidence: f64, // 0.0 - 1.0
}

#[derive(Debug, Clone)]
pub enum TyposquatPattern {
    CharSwap,
    MissingChar,
    ExtraChar,
    Homoglyph,
    ScopeConfusion,
    HyphenUnderscore,
}

/// Check if a package name looks like a typosquat.
pub fn check_typosquat(name: &str) -> TyposquatResult {
    let popular: Vec<&str> = POPULAR_PACKAGES.lines().collect();

    // Skip if the package itself is popular
    if popular.contains(&name) {
        return TyposquatResult {
            package_name: name.to_string(),
            is_suspicious: false,
            similar_to: None,
            distance: 0,
            pattern: None,
            confidence: 0.0,
        };
    }

    let mut best_match: Option<(&str, usize)> = None;

    for &popular_name in &popular {
        let dist = levenshtein(name, popular_name);

        // Only flag if very close (distance 1-2) and names are similar length
        if dist > 0
            && dist <= 2
            && (name.len() as i32 - popular_name.len() as i32)
                .unsigned_abs()
                <= 2
        {
            if best_match.is_none() || dist < best_match.unwrap().1 {
                best_match = Some((popular_name, dist));
            }
        }
    }

    // Also check patterns
    let pattern = detect_pattern(name, best_match.map(|(n, _)| n));

    if let Some((similar, dist)) = best_match {
        let confidence = 1.0
            - (dist as f64 / name.len().max(similar.len()) as f64);
        TyposquatResult {
            package_name: name.to_string(),
            is_suspicious: confidence > 0.7,
            similar_to: Some(similar.to_string()),
            distance: dist,
            pattern,
            confidence,
        }
    } else {
        TyposquatResult {
            package_name: name.to_string(),
            is_suspicious: false,
            similar_to: None,
            distance: 0,
            pattern: None,
            confidence: 0.0,
        }
    }
}

/// Standard Levenshtein distance.
fn levenshtein(a: &str, b: &str) -> usize {
    let a_len = a.len();
    let b_len = b.len();
    let mut dp = vec![vec![0usize; b_len + 1]; a_len + 1];

    for i in 0..=a_len {
        dp[i][0] = i;
    }
    for j in 0..=b_len {
        dp[0][j] = j;
    }

    let a_bytes = a.as_bytes();
    let b_bytes = b.as_bytes();

    for i in 1..=a_len {
        for j in 1..=b_len {
            let cost = if a_bytes[i - 1] == b_bytes[j - 1] {
                0
            } else {
                1
            };
            dp[i][j] = (dp[i - 1][j] + 1)
                .min(dp[i][j - 1] + 1)
                .min(dp[i - 1][j - 1] + cost);
        }
    }

    dp[a_len][b_len]
}

fn detect_pattern(
    name: &str,
    similar: Option<&str>,
) -> Option<TyposquatPattern> {
    let similar = similar?;

    // Homoglyph detection: 0/o, 1/l, rn/m
    if (name.contains('0') && similar.contains('o'))
        || (name.contains('1') && similar.contains('l'))
    {
        return Some(TyposquatPattern::Homoglyph);
    }

    // Hyphen/underscore swap
    if name.replace('-', "_") == similar.replace('-', "_") {
        return Some(TyposquatPattern::HyphenUnderscore);
    }

    if name.len() < similar.len() {
        Some(TyposquatPattern::MissingChar)
    } else if name.len() > similar.len() {
        Some(TyposquatPattern::ExtraChar)
    } else {
        Some(TyposquatPattern::CharSwap)
    }
}

/// - Image assets (png, jpg, svg, gif, ico)
/// - Font files (woff, woff2, ttf, otf)

const ALLOWED_BINARY_EXTENSIONS: &[&str] = &[
    "node", "wasm", "png", "jpg", "jpeg", "gif", "svg", "ico",
    "woff", "woff2", "ttf", "otf", "eot",
];

pub struct BinaryBlobResult {
    pub package_name: String,
    pub suspicious_files: Vec<SuspiciousBinary>,
    pub has_suspicious: bool,
}

pub struct SuspiciousBinary {
    pub path: String,
    pub size_bytes: u64,
    pub file_type: String, // detected via magic bytes
    pub reason: String,
}

/// Scan a package directory for suspicious binary files.
pub fn scan_for_binaries(
    package_dir: &Path,
) -> BinaryBlobResult {
    let mut suspicious = Vec::new();

    for entry in walkdir::WalkDir::new(package_dir)
        .into_iter()
        .flatten()
    {
        let path = entry.path();
        if !path.is_file() {
            continue;
        }

        let ext = path
            .extension()
            .and_then(|e| e.to_str())
            .unwrap_or("");

        // Read first bytes for magic number detection
        if let Ok(bytes) = std::fs::read(path) {
            if bytes.len() >= 4 && is_binary_content(&bytes) {
                if !ALLOWED_BINARY_EXTENSIONS.contains(&ext) {
                    suspicious.push(SuspiciousBinary {
                        path: path.display().to_string(),
                        size_bytes: bytes.len() as u64,
                        file_type: detect_binary_type(&bytes),
                        reason: format!(
                            "Unexpected binary file with .{} extension",
                            ext
                        ),
                    });
                }
            }
        }
    }

    BinaryBlobResult {
        package_name: package_dir
            .file_name()
            .and_then(|n| n.to_str())
            .unwrap_or("unknown")
            .to_string(),
        has_suspicious: !suspicious.is_empty(),
        suspicious_files: suspicious,
    }
}

fn is_binary_content(bytes: &[u8]) -> bool {
    // Check for null bytes in first 8KB (strong binary indicator)
    let check_len = bytes.len().min(8192);
    bytes[..check_len].contains(&0)
}

fn detect_binary_type(bytes: &[u8]) -> String {
    if bytes.len() < 4 {
        return "unknown".to_string();
    }
    // ELF
    if bytes[0..4] == [0x7f, 0x45, 0x4c, 0x46] {
        return "ELF executable".to_string();
    }
    // Mach-O (various magic numbers)
    if bytes[0..4] == [0xfe, 0xed, 0xfa, 0xce]
        || bytes[0..4] == [0xfe, 0xed, 0xfa, 0xcf]
        || bytes[0..4] == [0xce, 0xfa, 0xed, 0xfe]
        || bytes[0..4] == [0xcf, 0xfa, 0xed, 0xfe]
    {
        return "Mach-O executable".to_string();
    }
    // PE (Windows)
    if bytes[0..2] == [0x4d, 0x5a] {
        return "PE executable".to_string();
    }
    // Jar/Zip
    if bytes[0..4] == [0x50, 0x4b, 0x03, 0x04] {
        return "ZIP/JAR archive".to_string();
    }
    "binary".to_string()
}

// crates/better-core/src/firewall/obfuscation.rs

/// Detect obfuscated JavaScript patterns.
///
/// Heuristics:
/// 1. Extremely long single lines (>5000 chars with no newlines)
/// 2. High ratio of hex escapes (\x41\x42\x43)
/// 3. Dynamic code execution with concatenated strings
/// 4. Encoded payloads (base64, hex-encoded strings > 500 chars)
/// 5. Unusual variable names (all single-char or hex-like: _0x4a3b)

use regex::Regex;
use std::path::Path;

pub struct ObfuscationResult {
    pub file_path: String,
    pub is_obfuscated: bool,
    pub confidence: f64,
    pub indicators: Vec<ObfuscationIndicator>,
}

#[derive(Debug, Clone)]
pub struct ObfuscationIndicator {
    pub pattern: String,
    pub description: String,
    pub line: Option<usize>,
    pub severity: f64, // 0.0 - 1.0
}

pub fn scan_for_obfuscation(
    file_path: &Path,
) -> Option<ObfuscationResult> {
    let content = std::fs::read_to_string(file_path).ok()?;
    let mut indicators = Vec::new();

    // 1. Long lines
    for (i, line) in content.lines().enumerate() {
        if line.len() > 5000 {
            indicators.push(ObfuscationIndicator {
                pattern: "long-line".to_string(),
                description: format!(
                    "Line {} has {} chars (threshold: 5000)",
                    i + 1,
                    line.len()
                ),
                line: Some(i + 1),
                severity: 0.7,
            });
        }
    }

    // 2. Hex escape ratio
    let hex_escape_re =
        Regex::new(r"\\x[0-9a-fA-F]{2}").unwrap();
    let hex_count = hex_escape_re.find_iter(&content).count();
    let hex_ratio =
        hex_count as f64 / content.len().max(1) as f64 * 100.0;
    if hex_ratio > 5.0 {
        indicators.push(ObfuscationIndicator {
            pattern: "hex-escapes".to_string(),
            description: format!(
                "{:.1}% hex escape density (threshold: 5%)",
                hex_ratio
            ),
            line: None,
            severity: 0.8,
        });
    }

    // 3. Dynamic code execution with string concatenation
    // Pattern: Function("string" + "string") or similar code-from-string patterns
    let code_exec_re = Regex::new(
        r#"Function\s*\(\s*(?:['"_a-zA-Z0-9\+\s\[\]]+\+)"#,
    )
    .unwrap();
    if code_exec_re.is_match(&content) {
        indicators.push(ObfuscationIndicator {
            pattern: "code-exec-concat".to_string(),
            description:
                "Dynamic code execution with string concatenation detected"
                    .to_string(),
            line: None,
            severity: 0.9,
        });
    }

    // 4. Long encoded strings
    let base64_re =
        Regex::new(r#"['"][A-Za-z0-9+/=]{500,}['"]"#).unwrap();
    if base64_re.is_match(&content) {
        indicators.push(ObfuscationIndicator {
            pattern: "encoded-payload".to_string(),
            description:
                "Long base64-like encoded string detected".to_string(),
            line: None,
            severity: 0.8,
        });
    }

    // 5. Obfuscator variable patterns
    let obf_var_re =
        Regex::new(r"_0x[0-9a-fA-F]{4,}").unwrap();
    let obf_var_count = obf_var_re.find_iter(&content).count();
    if obf_var_count > 10 {
        indicators.push(ObfuscationIndicator {
            pattern: "obfuscator-vars".to_string(),
            description: format!(
                "{} obfuscator-style variables (_0x...)",
                obf_var_count
            ),
            line: None,
            severity: 0.9,
        });
    }

    let total_severity: f64 =
        indicators.iter().map(|i| i.severity).sum();
    let confidence =
        (total_severity / indicators.len().max(1) as f64).min(1.0);

    Some(ObfuscationResult {
        file_path: file_path.display().to_string(),
        is_obfuscated: confidence > 0.6 && !indicators.is_empty(),
        confidence,
        indicators,
    })
}

// crates/better-core/src/registry/config.rs

use serde::{Deserialize, Serialize};
use std::collections::HashMap;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct RegistryConfig {
    pub registries: Vec<Registry>,
    pub routing: RoutingConfig,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Registry {
    pub name: String,
    pub url: String,
    pub priority: u32,
    pub scopes: Vec<String>,
    pub auth: Option<RegistryAuth>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(tag = "type")]
pub enum RegistryAuth {
    #[serde(rename = "token")]
    Token {
        token_env: String,
        #[serde(skip_serializing_if = "Option::is_none")]
        token: Option<String>,
    },
    #[serde(rename = "basic")]
    Basic {
        username_env: String,
        password_env: String,
    },
    #[serde(rename = "bearer")]
    Bearer { token_env: String },
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct RoutingConfig {
    pub default: String,
    pub fallback: bool,
}

impl RegistryConfig {
    /// Resolve which registry to use for a given package name.
    ///
    /// Algorithm:
    /// 1. If package has a scope, find registry with matching scope
    /// 2. If multiple match, use lowest priority number (highest priority)
    /// 3. If no scope match, use default registry
    /// 4. If fallback=true and primary fails, try next priority
    pub fn resolve_registry(
        &self,
        package_name: &str,
    ) -> &Registry {
        let scope = extract_scope(package_name);

        if let Some(scope) = scope {
            let mut candidates: Vec<&Registry> = self
                .registries
                .iter()
                .filter(|r| {
                    r.scopes.contains(&scope.to_string())
                })
                .collect();
            candidates.sort_by_key(|r| r.priority);

            if let Some(registry) = candidates.first() {
                return registry;
            }
        }

        // Default
        self.registries
            .iter()
            .find(|r| r.name == self.routing.default)
            .unwrap_or(&self.registries[0])
    }

    /// Get fallback registries in priority order.
    pub fn fallback_chain(
        &self,
        primary: &str,
    ) -> Vec<&Registry> {
        if !self.routing.fallback {
            return vec![];
        }
        let mut others: Vec<&Registry> = self
            .registries
            .iter()
            .filter(|r| r.name != primary)
            .collect();
        others.sort_by_key(|r| r.priority);
        others
    }
}

fn extract_scope(name: &str) -> Option<&str> {
    if name.starts_with('@') {
        name.split('/').next()
    } else {
        None
    }
}

/// Token rotation for registry auth.
///
/// Workflow:
/// 1. `better registry rotate <name>` prompts for new token
/// 2. Updates token in env file or keychain
/// 3. Verifies new token works with a whoami equivalent
/// 4. Logs rotation in receipt

pub struct TokenRotation {
    pub registry_name: String,
    pub old_token_hash: String, // SHA-256 of old token (for audit)
    pub new_token_hash: String,
    pub rotated_at: String,
    pub verified: bool,
}

/// Verify a registry token by making an authenticated request.
pub async fn verify_token(
    registry_url: &str,
    auth: &RegistryAuth,
) -> Result<TokenVerification, RegistryError> {
    let url = format!("{}/-/whoami", registry_url);
    let token = resolve_token(auth)?;

    let client = reqwest::Client::new();
    let response = client
        .get(&url)
        .header("Authorization", format!("Bearer {}", token))
        .send()
        .await
        .map_err(|e| {
            RegistryError::NetworkError(e.to_string())
        })?;

    if response.status().is_success() {
        let body: serde_json::Value =
            response.json().await.map_err(|e| {
                RegistryError::NetworkError(e.to_string())
            })?;
        Ok(TokenVerification {
            valid: true,
            username: body["username"]
                .as_str()
                .map(|s| s.to_string()),
        })
    } else {
        Ok(TokenVerification {
            valid: false,
            username: None,
        })
    }
}

pub struct TokenVerification {
    pub valid: bool,
    pub username: Option<String>,
}

fn resolve_token(
    auth: &RegistryAuth,
) -> Result<String, RegistryError> {
    match auth {
        RegistryAuth::Token { token_env, token } => {
            if let Some(t) = token {
                Ok(t.clone())
            } else {
                std::env::var(token_env).map_err(|_| {
                    RegistryError::MissingEnvVar(
                        token_env.clone(),
                    )
                })
            }
        }
        RegistryAuth::Bearer { token_env } => {
            std::env::var(token_env).map_err(|_| {
                RegistryError::MissingEnvVar(token_env.clone())
            })
        }
        _ => Err(RegistryError::UnsupportedAuthType),
    }
}

#[derive(Debug)]
pub enum RegistryError {
    MissingEnvVar(String),
    UnsupportedAuthType,
    NetworkError(String),
    AuthFailed,
}

## Security Considerations

### Script Sandboxing (Task 30)
- macOS `sandbox-exec` is deprecated but still functional; monitor for removal in future macOS versions
- Linux Landlock requires kernel 5.13+; fallback chain ensures coverage
- Windows has no equivalent; document as unsupported with clear warning

### Provenance (Task 33)
- Sigstore verification requires network access to Rekor; cache results locally
- Not all packages have provenance; `--require-provenance` should be opt-in
- Certificate transparency logs are append-only; verify inclusion proofs

### Firewall (Task 35)
- Typosquat detection has false positives; confidence threshold tuning is critical
- Binary detection must allow legitimate native addons (.node files)
- Obfuscation detection should not flag minified code (minified != obfuscated)

### Registry (Task 36)
- Tokens must NEVER be stored in config files; always use env var references
- Token rotation should be atomic: verify new token before invalidating old
- Multi-registry fallback must not leak auth tokens to wrong registries
