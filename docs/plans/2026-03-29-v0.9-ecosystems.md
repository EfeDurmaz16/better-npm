// crates/better-core/src/engines/mod.rs
use std::path::Path;

/// Core trait that every ecosystem engine implements.
pub trait PackageEngine: Send + Sync {
    /// Engine identifier (e.g., "npm", "python", "cargo", "go").
    fn name(&self) -> &'static str;

    /// Detect if this engine should handle the given project root.
    /// Returns true if the project contains this engine's manifest files.
    fn detect(&self, project_root: &Path) -> bool;

    /// Resolve dependencies from manifest into a locked dependency graph.
    fn resolve(&self, project_root: &Path) -> Result<LockGraph, EngineError>;

    /// Fetch all packages in the lock graph into the CAS.
    fn fetch(&self, graph: &LockGraph) -> Result<Vec<CasEntry>, EngineError>;

    /// Materialize packages from CAS into the project's dependency directory.
    fn materialize(&self, entries: &[CasEntry], project_root: &Path) -> Result<MaterializeResult, EngineError>;

    /// Audit the dependency graph for known vulnerabilities.
    fn audit(&self, graph: &LockGraph) -> Result<Vec<Vulnerability>, EngineError>;

    /// Return the lockfile section key for better.lock (e.g., "npm", "cargo").
    fn lockfile_section(&self) -> &'static str;

    /// Parse an existing ecosystem-specific lockfile for migration.
    fn parse_lockfile(&self, lockfile_path: &Path) -> Result<LockGraph, EngineError>;

    /// List outdated packages.
    fn outdated(&self, graph: &LockGraph) -> Result<Vec<OutdatedPackage>, EngineError>;
}

/// Resolved dependency graph — shared across all engines.
#[derive(Debug, Clone)]
pub struct LockGraph {
    pub packages: Vec<LockedPackage>,
    pub edges: Vec<(usize, usize)>,  // (parent_idx, child_idx) dependency edges
    pub ecosystem: String,
}

#[derive(Debug, Clone)]
pub struct LockedPackage {
    pub name: String,
    pub version: String,
    pub registry_url: String,
    pub integrity: String,          // SHA-256 or SHA-512
    pub cas_address: Option<String>, // CAS hash once fetched
    pub dependencies: Vec<String>,
    pub dev_only: bool,
    pub platform: Option<String>,   // OS/arch constraint
}

#[derive(Debug, Clone)]
pub struct CasEntry {
    pub cas_hash: String,
    pub package_name: String,
    pub version: String,
    pub size_bytes: u64,
    pub file_count: u32,
}

#[derive(Debug, Clone)]
pub struct MaterializeResult {
    pub files_linked: u64,
    pub files_copied: u64,
    pub total_bytes: u64,
    pub elapsed_ms: u64,
}

#[derive(Debug, Clone)]
pub struct Vulnerability {
    pub id: String,         // CVE or OSV ID
    pub package: String,
    pub version: String,
    pub severity: String,
    pub title: String,
    pub url: Option<String>,
    pub patched_versions: Vec<String>,
}

#[derive(Debug, Clone)]
pub struct OutdatedPackage {
    pub name: String,
    pub current: String,
    pub wanted: String,
    pub latest: String,
}

#[derive(Debug, thiserror::Error)]
pub enum EngineError {
    #[error("Manifest not found at {0}")]
    ManifestNotFound(String),
    #[error("Manifest parse error: {0}")]
    ManifestParseError(String),
    #[error("Resolution failed: {0}")]
    ResolutionFailed(String),
    #[error("Fetch failed for {package}@{version}: {reason}")]
    FetchFailed { package: String, version: String, reason: String },
    #[error("Registry error: {0}")]
    RegistryError(String),
    #[error("CAS error: {0}")]
    CasError(String),
    #[error("Materialization error: {0}")]
    MaterializeError(String),
    #[error("Lockfile parse error: {0}")]
    LockfileParseError(String),
    #[error("Network error: {0}")]
    NetworkError(String),
    #[error("IO error: {0}")]
    IoError(String),
}

/// Engine registry — auto-detects which engines apply to a project.
pub struct EngineRegistry {
    engines: Vec<Box<dyn PackageEngine>>,
}

impl EngineRegistry {
    pub fn new() -> Self {
        Self {
            engines: vec![
                Box::new(super::npm::NpmEngine),
                Box::new(super::python::PythonEngine),
                Box::new(super::cargo::CargoEngine),
                Box::new(super::go::GoEngine),
            ],
        }
    }

    /// Detect all applicable engines for a project root.
    pub fn detect(&self, project_root: &Path) -> Vec<&dyn PackageEngine> {
        self.engines.iter()
            .filter(|e| e.detect(project_root))
            .map(|e| e.as_ref())
            .collect()
    }

    /// Register a plugin engine dynamically.
    pub fn register(&mut self, engine: Box<dyn PackageEngine>) {
        self.engines.push(engine);
    }
}

use super::{PackageEngine, LockGraph, LockedPackage, CasEntry, MaterializeResult, Vulnerability, OutdatedPackage, EngineError};
use serde::Deserialize;

pub struct CargoEngine;

impl PackageEngine for CargoEngine {
    fn name(&self) -> &'static str { "cargo" }

    fn detect(&self, project_root: &Path) -> bool {
        project_root.join("Cargo.toml").exists()
    }

    fn resolve(&self, project_root: &Path) -> Result<LockGraph, EngineError> {
        let cargo_toml_path = project_root.join("Cargo.toml");
        let cargo_toml = parse_cargo_toml(&cargo_toml_path)?;

        // If Cargo.lock exists, parse it for exact versions
        let cargo_lock_path = project_root.join("Cargo.lock");
        if cargo_lock_path.exists() {
            return parse_cargo_lock(&cargo_lock_path);
        }

        // Otherwise, resolve from crates.io API
        resolve_from_registry(&cargo_toml)
    }

    fn fetch(&self, graph: &LockGraph) -> Result<Vec<CasEntry>, EngineError> {
        // For each package in graph:
        //   1. Check CAS at ~/.better/cas/crates/{name}-{version}.crate
        //   2. If miss: GET https://crates.io/api/v1/crates/{name}/{version}/download
        //   3. Verify SHA-256 checksum from Cargo.lock
        //   4. Store in CAS
        fetch_crates(graph)
    }

    fn materialize(&self, entries: &[CasEntry], project_root: &Path) -> Result<MaterializeResult, EngineError> {
        // Cargo uses a global registry cache, not node_modules-style
        // Materialize to ~/.cargo/registry/cache/ (symlinks from CAS)
        // Or if using better's own layout: project_root/.better-target/
        materialize_crates(entries, project_root)
    }

    fn audit(&self, graph: &LockGraph) -> Result<Vec<Vulnerability>, EngineError> {
        // Query https://osv.dev/v1/querybatch for crates.io ecosystem
        audit_crates(graph)
    }

    fn lockfile_section(&self) -> &'static str { "cargo" }

    fn parse_lockfile(&self, lockfile_path: &Path) -> Result<LockGraph, EngineError> {
        parse_cargo_lock(lockfile_path)
    }

    fn outdated(&self, graph: &LockGraph) -> Result<Vec<OutdatedPackage>, EngineError> {
        // For each package: GET https://crates.io/api/v1/crates/{name}
        // Compare current version vs latest
        check_outdated_crates(graph)
    }

struct CargoToml {

    dependencies: Option<toml::Table>,
    #[serde(rename = "dev-dependencies")]
    dev_dependencies: Option<toml::Table>,
    #[serde(rename = "build-dependencies")]

}

#[derive(Debug, Deserialize)]
struct CargoPackage {
    name: String,
    version: String,
    edition: Option<String>,
}

#[derive(Debug, Deserialize)]
struct CargoWorkspace {
    members: Option<Vec<String>>,
    exclude: Option<Vec<String>>,
}

/// Parse Cargo.toml into dependency list.
fn parse_cargo_toml(path: &Path) -> Result<CargoToml, EngineError> {
    let content = std::fs::read_to_string(path)
        .map_err(|e| EngineError::ManifestNotFound(e.to_string()))?;
    toml::from_str::<CargoToml>(&content)
        .map_err(|e| EngineError::ManifestParseError(format!("Cargo.toml: {}", e)))
}

/// Extract dependency specs from Cargo.toml tables.
/// Handles: `serde = "1.0"`, `serde = { version = "1.0", features = [...] }`, `serde.workspace = true`
fn extract_dependencies(table: &toml::Table) -> Vec<CargoDep> {
    let mut deps = Vec::new();
    for (name, value) in table {
        let dep = match value {
            toml::Value::String(version) => CargoDep {
                name: name.clone(),
                version_req: version.clone(),
                features: vec![],
                optional: false,
                git: None,
                path: None,
            },
            toml::Value::Table(t) => CargoDep {
                name: name.clone(),
                version_req: t.get("version")
                    .and_then(|v| v.as_str())
                    .unwrap_or("*")
                    .to_string(),
                features: t.get("features")
                    .and_then(|v| v.as_array())
                    .map(|a| a.iter().filter_map(|v| v.as_str().map(String::from)).collect())
                    .unwrap_or_default(),
                optional: t.get("optional")
                    .and_then(|v| v.as_bool())
                    .unwrap_or(false),
                git: t.get("git")
                    .and_then(|v| v.as_str())
                    .map(String::from),
                path: t.get("path")
                    .and_then(|v| v.as_str())
                    .map(String::from),
            },
            _ => continue,
        };
        deps.push(dep);
    }
    deps
}

    name: String,
    version_req: String,  // Cargo semver requirement (e.g., "^1.0", ">=0.10,<0.12")
    features: Vec<String>,
    optional: bool,
    git: Option<String>,
    path: Option<String>,
}

// --- Cargo.lock parsing ---

#[derive(Debug, Deserialize)]
struct CargoLockFile {
    version: Option<u32>,      // Cargo.lock format version (3 = latest)
    package: Vec<CargoLockPackage>,
}

#[derive(Debug, Deserialize)]
struct CargoLockPackage {
    name: String,
    version: String,
    source: Option<String>,     // "registry+https://github.com/rust-lang/crates.io-index"
    checksum: Option<String>,   // SHA-256
    dependencies: Option<Vec<String>>,  // "name version (source)"
}

/// Parse Cargo.lock into LockGraph.
fn parse_cargo_lock(path: &Path) -> Result<LockGraph, EngineError> {
    let content = std::fs::read_to_string(path)
        .map_err(|e| EngineError::LockfileParseError(e.to_string()))?;
    let lock: CargoLockFile = toml::from_str(&content)
        .map_err(|e| EngineError::LockfileParseError(format!("Cargo.lock: {}", e)))?;

    let mut packages = Vec::new();
    let mut edges = Vec::new();

    for (idx, pkg) in lock.package.iter().enumerate() {
        let is_registry = pkg.source.as_ref()
            .map(|s| s.starts_with("registry+"))
            .unwrap_or(false);

        packages.push(LockedPackage {
            name: pkg.name.clone(),
            version: pkg.version.clone(),
            registry_url: if is_registry {
                format!("https://crates.io/api/v1/crates/{}/{}/download", pkg.name, pkg.version)
            } else {
                String::new()
            },
            integrity: pkg.checksum.clone().unwrap_or_default(),
            cas_address: None,
            dependencies: pkg.dependencies.clone().unwrap_or_default(),
            dev_only: false,
            platform: None,
        });

        // Build edges from dependency strings
        if let Some(ref deps) = pkg.dependencies {
            for dep_str in deps {
                // Format: "name version" or "name version (source)"
                let dep_name = dep_str.split_whitespace().next().unwrap_or("");
                if let Some(dep_idx) = lock.package.iter().position(|p| p.name == dep_name) {
                    edges.push((idx, dep_idx));
                }
            }
        }
    }

    Ok(LockGraph {
        packages,
        edges,
        ecosystem: "cargo".into(),
    })
}

/// Fetch crates from crates.io into CAS.
fn fetch_crates(graph: &LockGraph) -> Result<Vec<CasEntry>, EngineError> {
    use rayon::prelude::*;

    let cas_root = dirs::home_dir()
        .expect("home dir")
        .join(".better")
        .join("cas")
        .join("crates");
    std::fs::create_dir_all(&cas_root)
        .map_err(|e| EngineError::CasError(e.to_string()))?;

    graph.packages.par_iter()
        .filter(|pkg| !pkg.registry_url.is_empty())
        .map(|pkg| {
            let cas_path = cas_root.join(format!("{}-{}.crate", pkg.name, pkg.version));

            // Check CAS hit
            if cas_path.exists() {
                let metadata = std::fs::metadata(&cas_path)
                    .map_err(|e| EngineError::CasError(e.to_string()))?;
                return Ok(CasEntry {
                    cas_hash: pkg.integrity.clone(),
                    package_name: pkg.name.clone(),
                    version: pkg.version.clone(),
                    size_bytes: metadata.len(),

        .collect()

use std::path::Path;
use super::{PackageEngine, LockGraph, LockedPackage, CasEntry, MaterializeResult, Vulnerability, OutdatedPackage, EngineError};

pub struct GoEngine;

impl PackageEngine for GoEngine {
    fn name(&self) -> &'static str { "go" }

    fn detect(&self, project_root: &Path) -> bool {
        project_root.join("go.mod").exists()
    }

    fn resolve(&self, project_root: &Path) -> Result<LockGraph, EngineError> {
        let go_mod_path = project_root.join("go.mod");
        let go_mod = parse_go_mod(&go_mod_path)?;

        let go_sum_path = project_root.join("go.sum");
        if go_sum_path.exists() {
            return parse_go_sum_with_mod(&go_mod, &go_sum_path);
        }

        resolve_from_proxy(&go_mod)
    }

    fn fetch(&self, graph: &LockGraph) -> Result<Vec<CasEntry>, EngineError> {
        fetch_modules(graph)
    }

    fn materialize(&self, entries: &[CasEntry], project_root: &Path) -> Result<MaterializeResult, EngineError> {
        // Go uses $GOPATH/pkg/mod/ as module cache
        // better links from CAS to $GOPATH/pkg/mod/cache/download/
        materialize_modules(entries, project_root)
    }

    fn audit(&self, graph: &LockGraph) -> Result<Vec<Vulnerability>, EngineError> {
        // Query https://vuln.go.dev/api/ OR osv.dev with ecosystem "Go"
        audit_modules(graph)
    }

    fn lockfile_section(&self) -> &'static str { "go" }

    fn parse_lockfile(&self, lockfile_path: &Path) -> Result<LockGraph, EngineError> {
        // go.sum is the "lockfile" in Go
        // Need go.mod alongside for module path
        let go_mod_path = lockfile_path.parent()
            .ok_or(EngineError::LockfileParseError("No parent dir".into()))?
            .join("go.mod");
        let go_mod = parse_go_mod(&go_mod_path)?;

/// Parsed go.mod file.
#[derive(Debug)]
struct GoMod {
    module: String,          // Module path (e.g., "github.com/user/repo")
    go_version: String,      // Go version (e.g., "1.22")
    require: Vec<GoRequire>,
    replace: Vec<GoReplace>,
    exclude: Vec<GoExclude>,
    retract: Vec<String>,

#[derive(Debug)]
struct GoRequire {
    module_path: String,     // e.g., "github.com/gin-gonic/gin"
    version: String,         // e.g., "v1.9.1"
    indirect: bool,          // // indirect comment
}

#[derive(Debug)]
struct GoReplace {
    old_path: String,
    old_version: Option<String>,
    new_path: String,
    new_version: Option<String>,
}

#[derive(Debug)]
struct GoExclude {
    module_path: String,
    version: String,
}

/// Parse go.mod file.
/// Format:
/// ```
/// module github.com/user/repo
///
/// go 1.22
///
/// require (
///     github.com/gin-gonic/gin v1.9.1
///     golang.org/x/net v0.21.0 // indirect
/// )
///
/// replace github.com/old => github.com/new v1.0.0
/// ```
fn parse_go_mod(path: &Path) -> Result<GoMod, EngineError> {
    let content = std::fs::read_to_string(path)
        .map_err(|e| EngineError::ManifestNotFound(e.to_string()))?;

    let mut module = String::new();
    let mut go_version = String::new();
    let mut require = Vec::new();
    let mut replace = Vec::new();
    let mut exclude = Vec::new();
    let retract = Vec::new();

    let mut in_require_block = false;
    let mut in_replace_block = false;
    let mut in_exclude_block = false;

    for line in content.lines() {
        let trimmed = line.trim();

        // Skip empty lines and comments
        if trimmed.is_empty() || trimmed.starts_with("//") {
            continue;
        }

        // Block end
        if trimmed == ")" {
            in_require_block = false;
            in_replace_block = false;
            in_exclude_block = false;
            continue;
        }

        // Block start
        if trimmed == "require (" {
            in_require_block = true;
            continue;
        }
        if trimmed == "replace (" {
            in_replace_block = true;
            continue;
        }
        if trimmed == "exclude (" {
            in_exclude_block = true;
            continue;
        }

        // Module declaration
        if trimmed.starts_with("module ") {
            module = trimmed[7..].trim().to_string();
            continue;
        }

        // Go version
        if trimmed.starts_with("go ") {
            go_version = trimmed[3..].trim().to_string();
            continue;
        }

        // Single-line require
        if trimmed.starts_with("require ") && !trimmed.contains('(') {
            if let Some(req) = parse_require_line(&trimmed[8..]) {
                require.push(req);
            }
            continue;
        }

        // Inside require block
        if in_require_block {
            if let Some(req) = parse_require_line(trimmed) {
                require.push(req);
            }
            continue;
        }

        // Inside replace block
        if in_replace_block {
            if let Some(rep) = parse_replace_line(trimmed) {
                replace.push(rep);
            }
            continue;
        }

        // Single-line replace
        if trimmed.starts_with("replace ") {
            if let Some(rep) = parse_replace_line(&trimmed[8..]) {
                replace.push(rep);
            }
            continue;
        }

        // Inside exclude block
        if in_exclude_block {
            if let Some(exc) = parse_exclude_line(trimmed) {
                exclude.push(exc);
            }
            continue;
        }
    }

    Ok(GoMod { module, go_version, require, replace, exclude, retract })
}

fn parse_require_line(line: &str) -> Option<GoRequire> {
    let trimmed = line.trim();
    let parts: Vec<&str> = trimmed.split_whitespace().collect();
    if parts.len() >= 2 {
        let indirect = trimmed.contains("// indirect");
        Some(GoRequire {
            module_path: parts[0].to_string(),
            version: parts[1].to_string(),
            indirect,
        })
    } else {
        None
    }
}

fn parse_replace_line(line: &str) -> Option<GoReplace> {
    let parts: Vec<&str> = line.split("=>").collect();
    if parts.len() == 2 {
        let old_parts: Vec<&str> = parts[0].trim().split_whitespace().collect();
        let new_parts: Vec<&str> = parts[1].trim().split_whitespace().collect();
        Some(GoReplace {
            old_path: old_parts.first()?.to_string(),
            old_version: old_parts.get(1).map(|s| s.to_string()),
            new_path: new_parts.first()?.to_string(),
            new_version: new_parts.get(1).map(|s| s.to_string()),
        })
    } else {
        None
    }
}

fn parse_exclude_line(line: &str) -> Option<GoExclude> {
    let parts: Vec<&str> = line.trim().split_whitespace().collect();
    if parts.len() >= 2 {
        Some(GoExclude {
            module_path: parts[0].to_string(),
            version: parts[1].to_string(),
        })
    } else {
        None
    }
}

/// Parse go.sum alongside go.mod to build LockGraph.
/// go.sum format: module_path version hash
/// e.g., github.com/gin-gonic/gin v1.9.1 h1:4+fr/el88TOO3ewCmQr8cx/ePFNNbF7CmMt8h1tBQ=
/// e.g., github.com/gin-gonic/gin v1.9.1/go.mod h1:abc123...
fn parse_go_sum_with_mod(go_mod: &GoMod, go_sum_path: &Path) -> Result<LockGraph, EngineError> {
    let content = std::fs::read_to_string(go_sum_path)
        .map_err(|e| EngineError::LockfileParseError(e.to_string()))?;

    let mut packages = Vec::new();
    let mut seen = std::collections::HashSet::new();

    for line in content.lines() {
        let parts: Vec<&str> = line.split_whitespace().collect();
        if parts.len() != 3 {
            continue;
        }

        let module_path = parts[0];
        let version_str = parts[1];
        let hash = parts[2];

        // Skip go.mod hashes (we want the source hash)
        if version_str.ends_with("/go.mod") {
            continue;
        }

        let key = format!("{}@{}", module_path, version_str);
        if seen.contains(&key) {
            continue;
        }
        seen.insert(key);

        let is_indirect = go_mod.require.iter()
            .find(|r| r.module_path == module_path)
            .map(|r| r.indirect)
            .unwrap_or(true);

        packages.push(LockedPackage {
            name: module_path.to_string(),
            version: version_str.to_string(),
            registry_url: format!(
                "https://proxy.golang.org/{}/@v/{}.zip",
                module_path, version_str
            ),
            integrity: hash.to_string(),
            cas_address: None,
            dependencies: vec![],  // Go doesn't list transitive deps in go.sum
            dev_only: false,
            platform: None,
        });
    }

    Ok(LockGraph {
        packages,
        edges: vec![],  // go.sum doesn't encode edges, would need `go mod graph`
        ecosystem: "go".into(),
    })
}

/// Topological sort of workspace members based on depends_on.
/// Returns indices in build order (dependencies first).
fn topological_sort(members: &[WorkspaceMember]) -> Result<Vec<usize>, EngineError> {
    let n = members.len();
    let mut in_degree = vec![0usize; n];
    let mut adj: Vec<Vec<usize>> = vec![vec![]; n];

    // Build adjacency list
    for (i, member) in members.iter().enumerate() {
        for dep_name in &member.depends_on {
            if let Some(dep_idx) = members.iter().position(|m| m.name == *dep_name) {
                adj[dep_idx].push(i);
                in_degree[i] += 1;
            }
        }
    }

    // Kahn's algorithm
    let mut queue: std::collections::VecDeque<usize> = std::collections::VecDeque::new();
    for i in 0..n {
        if in_degree[i] == 0 {
            queue.push_back(i);
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(node) = queue.pop_front() {
        order.push(node);
        for &neighbor in &adj[node] {
            in_degree[neighbor] -= 1;
            if in_degree[neighbor] == 0 {
                queue.push_back(neighbor);
            }
        }
    }

    if order.len() != n {
        return Err(EngineError::ResolutionFailed(
            "Circular dependency detected in workspace".into()
        ));
    }

    Ok(order)
}

/// Detect the ecosystem and framework to determine env file conventions.
pub fn detect_env_convention(project_root: &std::path::Path) -> EnvConvention {
    // Next.js / Vite / Node: .env, .env.local
    if project_root.join("package.json").exists() {
        if project_root.join("next.config.js").exists()
            || project_root.join("next.config.mjs").exists()
            || project_root.join("next.config.ts").exists() {
            return EnvConvention::NextJs;
        }
        if project_root.join("vite.config.js").exists()
            || project_root.join("vite.config.ts").exists() {
            return EnvConvention::Vite;
        }
        return EnvConvention::Node;
    }

    // Python: .env (python-dotenv), pyproject.toml [tool.poe.env]
    if project_root.join("pyproject.toml").exists()
        || project_root.join("requirements.txt").exists() {
        return EnvConvention::Python;
    }

    // Rust: .env (dotenvy crate)
    if project_root.join("Cargo.toml").exists() {
        return EnvConvention::Rust;
    }

    // Go: .env (godotenv)
    if project_root.join("go.mod").exists() {
        return EnvConvention::Go;
    }

    EnvConvention::Generic
}

    NextJs,     // .env.local (gitignored by default)
    Vite,       // .env.local
    Node,       // .env
    Python,     // .env
    Rust,       // .env
    Go,         // .env
    Generic,    // .env
}

impl EnvConvention {
    /// The output filename for generated env.
    pub fn output_filename(&self) -> &'static str {
        match self {
            Self::NextJs => ".env.local",
            Self::Vite => ".env.local",
            _ => ".env",
        }
    }

    /// Gitignore patterns to add.
    pub fn gitignore_patterns(&self) -> Vec<&'static str> {
        match self {
            Self::NextJs => vec![".env.local", ".env*.local"],
            Self::Vite => vec![".env.local", ".env*.local"],
            _ => vec![".env"],
        }
    }
}

/// Generate env file with framework-aware output path.
pub fn generate_env_cross_ecosystem(
    project_root: &std::path::Path,
    vault: &super::vault::Vault,
    agent_secret_key: &[u8; 32],
) -> Result<std::path::PathBuf, super::discovery::OspError> {
    let convention = detect_env_convention(project_root);
    let template_path = project_root.join(".env.osp");

    if !template_path.exists() {
        return Err(super::discovery::OspError::ParseError(
            "No .env.osp template found".into()
        ));
    }

    let content = std::fs::read_to_string(&template_path)
        .map_err(|e| super::discovery::OspError::VaultError(e.to_string()))?;
    let template = parse_env_template(&content);
    let env_content = generate_env(&template, vault, agent_secret_key)?;

    let output_path = project_root.join(convention.output_filename());
    std::fs::write(&output_path, &env_content)
        .map_err(|e| super::discovery::OspError::VaultError(e.to_string()))?;

    for pattern in convention.gitignore_patterns() {
        ensure_gitignore(project_root, pattern);
    }

    Ok(output_path)
}
