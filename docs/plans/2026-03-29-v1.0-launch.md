// crates/better-core/src/schema.rs

use serde::{Deserialize, Serialize};

/// Schema version follows better's minor version.
/// Breaking JSON output changes bump the schema major.
pub const SCHEMA_VERSION: &str = "1.0";

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct JsonEnvelope<T: Serialize> {
    pub schema_version: String,
    pub command: String,
    pub success: bool,
    pub data: T,
    #[serde(skip_serializing_if = "Vec::is_empty")]
    pub warnings: Vec<String>,
    #[serde(skip_serializing_if = "Vec::is_empty")]
    pub deprecations: Vec<DeprecationNotice>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct DeprecationNotice {
    pub feature: String,
    pub deprecated_in: String,  // "1.0.0"
    pub removed_in: String,     // "1.2.0"
    pub replacement: String,
    pub message: String,
}

impl<T: Serialize> JsonEnvelope<T> {
    pub fn ok(command: &str, data: T) -> Self {
        Self {
            schema_version: SCHEMA_VERSION.to_string(),
            command: command.to_string(),
            success: true,
            data,
            warnings: vec![],
            deprecations: vec![],
        }
    }

    pub fn with_deprecation(mut self, notice: DeprecationNotice) -> Self {
        self.deprecations.push(notice);
        self
    }
}
```

// crates/better-core/src/cli_compat.rs

use std::collections::HashMap;

/// Tracks CLI flags/commands that are deprecated.
/// Checked at startup before command dispatch.
pub struct CompatRegistry {
    deprecated: HashMap<String, DeprecatedFlag>,
}

pub struct DeprecatedFlag {
    pub flag: String,
    pub deprecated_in: semver::Version,
    pub removed_in: semver::Version,
    pub replacement: String,
}

impl CompatRegistry {
    pub fn new() -> Self {
        let mut registry = Self {
            deprecated: HashMap::new(),
        };
        // Register deprecations here as they occur
        // registry.deprecate("--legacy-peer-deps", "1.0.0", "1.2.0", "--peer-deps=legacy");
        registry
    }

    pub fn check(&self, args: &[String], current: &semver::Version) -> Vec<String> {
        let mut warnings = vec![];
        for arg in args {
            if let Some(dep) = self.deprecated.get(arg.as_str()) {
                if current >= &dep.removed_in {
                    warnings.push(format!(
                        "Error: '{}' was removed in {}. Use '{}' instead.",
                        dep.flag, dep.removed_in, dep.replacement
                    ));
                } else {
                    warnings.push(format!(
                        "Warning: '{}' is deprecated (since {}), will be removed in {}. Use '{}' instead.",
                        dep.flag, dep.deprecated_in, dep.removed_in, dep.replacement
                    ));
                }
            }
        }
        warnings
    }
}

class Better < Formula
  desc "Universal package manager — fastest, smartest, most secure"
  homepage "https://better.sh"
  version "1.0.0"

  on_macos do
    if Hardware::CPU.arm?
      url "https://github.com/nicepkg/better/releases/download/v#{version}/better-darwin-arm64.tar.gz"
      sha256 "PLACEHOLDER_ARM64_SHA256"
    else
      url "https://github.com/nicepkg/better/releases/download/v#{version}/better-darwin-x64.tar.gz"
      sha256 "PLACEHOLDER_X64_SHA256"
    end
  end

  on_linux do
    if Hardware::CPU.arm?
      url "https://github.com/nicepkg/better/releases/download/v#{version}/better-linux-arm64.tar.gz"
      sha256 "PLACEHOLDER_LINUX_ARM64_SHA256"
    else
      url "https://github.com/nicepkg/better/releases/download/v#{version}/better-linux-x64.tar.gz"
      sha256 "PLACEHOLDER_LINUX_X64_SHA256"
    end
  end

  def install
    bin.install "better"
    generate_completions_from_executable(bin/"better", "completions")
  end

  test do
    assert_match version.to_s, shell_output("#{bin}/better --version")
    assert_match "better", shell_output("#{bin}/better --help")
  end
end

// extensions/vscode/src/extension.ts
import * as vscode from 'vscode';
import { VersionHintProvider } from './providers/version-hints';
import { AuditDiagnosticProvider } from './providers/audit-diagnostics';
import { ContextPanelProvider } from './providers/context-panel';
import { DependencyTreeProvider } from './providers/dependency-tree';
import { OspServiceProvider } from './providers/osp-services';

export function activate(context: vscode.ExtensionContext) {
    const versionHints = new VersionHintProvider();
    const auditDiag = new AuditDiagnosticProvider();
    const contextPanel = new ContextPanelProvider(context);
    const depTree = new DependencyTreeProvider();
    const ospServices = new OspServiceProvider();

    // Inline version hints (CodeLens on package.json deps)
    context.subscriptions.push(
        vscode.languages.registerCodeLensProvider(
            { pattern: '**/package.json' },
            versionHints
        )
    );

    // Audit gutter warnings
    context.subscriptions.push(auditDiag.activate(context));

    // Sidebar tree views
    vscode.window.registerTreeDataProvider('betterDeps', depTree);
    vscode.window.registerTreeDataProvider('betterOsp', ospServices);

    // Commands
    context.subscriptions.push(
        vscode.commands.registerCommand('better.audit', () => auditDiag.runAudit()),
        vscode.commands.registerCommand('better.context', (pkg: string) =>
            contextPanel.showContext(pkg)
        ),
    );
}
```

// crates/better-core/src/upgrade.rs

use std::fs;
use std::path::{Path, PathBuf};

#[derive(Debug, Clone)]
pub struct UpgradeConfig {
    pub current_version: semver::Version,
    pub target_version: Option<semver::Version>,  // None = latest
    pub binary_path: PathBuf,
    pub check_only: bool,
}

#[derive(Debug, Clone, serde::Serialize)]
pub struct UpgradeResult {
    pub from_version: String,
    pub to_version: String,
    pub checksum_verified: bool,
    pub backup_path: String,
    pub success: bool,
}

pub async fn upgrade(config: UpgradeConfig) -> Result<UpgradeResult, UpgradeError> {
    // 1. Fetch latest release info from GitHub API
    let release = fetch_latest_release(config.target_version.as_ref()).await?;

    if release.version <= config.current_version {
        return Err(UpgradeError::AlreadyLatest(config.current_version.to_string()));
    }

    if config.check_only {
        return Ok(UpgradeResult {
            from_version: config.current_version.to_string(),
            to_version: release.version.to_string(),
            checksum_verified: false,
            backup_path: String::new(),
            success: true,
        });
    }

    // 2. Download new binary to temp
    let platform = detect_platform()?;
    let archive_bytes = download_release(&release, &platform).await?;

    // 3. Verify SHA256 checksum
    let expected_checksum = fetch_checksum(&release, &platform).await?;
    let actual_checksum = sha256_hash(&archive_bytes);
    if actual_checksum != expected_checksum {
        return Err(UpgradeError::ChecksumMismatch {
            expected: expected_checksum,
            actual: actual_checksum,
        });
    }

    // 4. Backup current binary
    let backup_path = config.binary_path.with_extension("bak");
    fs::copy(&config.binary_path, &backup_path)?;

    // 5. Extract and replace atomically
    let new_binary = extract_archive(&archive_bytes)?;
    match atomic_replace(&config.binary_path, &new_binary) {
        Ok(_) => Ok(UpgradeResult {
            from_version: config.current_version.to_string(),
            to_version: release.version.to_string(),
            checksum_verified: true,
            backup_path: backup_path.display().to_string(),
            success: true,
        }),
        Err(e) => {
            // 6. Rollback on failure
            fs::copy(&backup_path, &config.binary_path)?;
            Err(UpgradeError::ReplaceFailed(e.to_string()))
        }
    }
}

pub fn rollback(binary_path: &Path) -> Result<(), UpgradeError> {
    let backup_path = binary_path.with_extension("bak");
    if !backup_path.exists() {
        return Err(UpgradeError::ReplaceFailed("No backup found. Cannot rollback.".into()));
    }
    fs::copy(&backup_path, binary_path)?;
    fs::remove_file(&backup_path)?;
    Ok(())
}

#[derive(Debug, thiserror::Error)]
pub enum UpgradeError {
    #[error("Already on latest version ({0})")]
    AlreadyLatest(String),
    #[error("Checksum mismatch: expected {expected}, got {actual}")]
    ChecksumMismatch { expected: String, actual: String },
    #[error("Failed to replace binary: {0}")]
    ReplaceFailed(String),
    #[error("Network error: {0}")]
    Network(String),
    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),
}

fn detect_platform() -> Result<String, UpgradeError> {
    let os = if cfg!(target_os = "macos") { "darwin" }
        else if cfg!(target_os = "linux") { "linux" }
        else if cfg!(target_os = "windows") { "windows" }
        else { return Err(UpgradeError::Network("unsupported platform".into())); };
    let arch = if cfg!(target_arch = "aarch64") { "arm64" }
        else { "x64" };
    Ok(format!("{}-{}", os, arch))
}

// crates/better-core/src/telemetry.rs

use serde::Serialize;
use std::path::PathBuf;
use std::time::Duration;
use uuid::Uuid;

#[derive(Debug, Clone, Serialize)]
pub struct TelemetryEvent {
    pub event_id: String,
    pub session_id: String,
    pub command: String,
    pub duration_ms: u64,
    pub success: bool,
    pub ecosystems: Vec<String>,
    pub package_count: Option<usize>,
    pub cas_hit_rate: Option<f64>,
    pub os: String,
    pub arch: String,
    pub better_version: String,
    // Explicitly NOT included: project name, package names, paths, IPs
}

#[derive(Debug, Clone)]
pub struct TelemetryConfig {
    pub enabled: bool,
    pub endpoint: String,
    pub session_id: String,
    pub config_path: PathBuf,
}

impl TelemetryConfig {
    pub fn load() -> Self {
        let config_path = dirs::config_dir()
            .unwrap_or_default()
            .join("better")
            .join("telemetry.json");

        let enabled = if config_path.exists() {
            let content = std::fs::read_to_string(&config_path).unwrap_or_default();
            serde_json::from_str::<serde_json::Value>(&content)
                .ok()
                .and_then(|v| v["enabled"].as_bool())
                .unwrap_or(false)
        } else {
            false  // Opt-in: disabled by default
        };

        Self {
            enabled,
            endpoint: "https://telemetry.better.sh/v1/events".to_string(),
            session_id: Uuid::new_v4().to_string(),
            config_path,
        }
    }

    pub fn set_enabled(&self, enabled: bool) -> std::io::Result<()> {
        let json = serde_json::json!({ "enabled": enabled });
        if let Some(parent) = self.config_path.parent() {
            std::fs::create_dir_all(parent)?;
        }
        std::fs::write(&self.config_path, serde_json::to_string_pretty(&json)?)?;
        Ok(())
    }
}

/// Send telemetry event in background — never blocks CLI
pub fn send_event(config: &TelemetryConfig, event: TelemetryEvent) {
    if !config.enabled {
        return;
    }

    let endpoint = config.endpoint.clone();
    let event_json = serde_json::to_vec(&event).unwrap_or_default();

    // Fire-and-forget in background thread with 2s timeout
    std::thread::spawn(move || {
        let client = reqwest::blocking::Client::builder()
            .timeout(Duration::from_secs(2))
            .build();

        if let Ok(client) = client {
            let _ = client
                .post(&endpoint)
                .header("Content-Type", "application/json")
                .body(event_json)
                .send();
        }
    });
}
