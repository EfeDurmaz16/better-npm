```rust
// crates/better-core/src/audit/scoring.rs

/// Dependency context — how a package is reachable in the dep graph.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, serde::Serialize)]
#[serde(rename_all = "snake_case")]
pub enum DepContext {
    /// Listed in `dependencies` and reachable from app entry points
    Production,
    /// Listed in `devDependencies`
    Dev,
    /// Only used in build scripts / bundler plugins
    Build,
    /// Listed in `optionalDependencies`
    Optional,
    /// Not a direct dep — pulled in transitively
    Transitive,
}

impl DepContext {
    /// Weight multiplier for risk scoring.
    /// Production deps have full weight; dev deps are much lower.
    pub fn weight(&self) -> f64 {
        match self {
            Self::Production => 1.0,
            Self::Dev => 0.15,
            Self::Build => 0.2,
            Self::Optional => 0.3,
            Self::Transitive => 0.6,
        }
    }
}

/// CVSS-style severity mapped to numeric score.
#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, serde::Serialize)]
#[serde(rename_all = "snake_case")]
pub enum Severity {
    Unknown,
    Low,
    Medium,
    High,
    Critical,
}

impl Severity {
    pub fn base_score(&self) -> f64 {
        match self {
            Self::Critical => 10.0,
            Self::High => 7.0,
            Self::Medium => 4.0,
            Self::Low => 1.0,
            Self::Unknown => 2.0,
        }
    }

    pub fn from_str(s: &str) -> Self {
        match s.to_lowercase().as_str() {
            "critical" => Self::Critical,
            "high" => Self::High,
            "medium" => Self::Medium,
            "low" => Self::Low,
            _ => Self::Unknown,
        }
    }
}

/// A single vulnerability with context-aware scoring.
#[derive(Debug, Clone, serde::Serialize)]
pub struct ScoredVuln {
    pub id: String,
    pub aliases: Vec<String>,
    pub summary: String,
    pub severity: Severity,
    pub base_score: f64,
    pub context: DepContext,
    pub context_weight: f64,
    pub effective_score: f64,  // base_score * context_weight
    pub package_name: String,
    pub package_version: String,
    pub fix_available: Option<String>,
}

/// Calculate effective score: severity_base * context_weight
pub fn score_vuln(severity: Severity, context: DepContext) -> f64 {
    severity.base_score() * context.weight()
}
```

```rust
// crates/better-core/src/audit/classifier.rs

use super::scoring::DepContext;
use std::collections::{HashMap, HashSet};

/// Input: the resolved dependency graph from better.lock.
/// Output: a map of package_key -> DepContext.
///
/// Classification algorithm:
/// 1. Parse root package.json to get dependencies, devDependencies,
///    optionalDependencies, and build-related deps (bundler plugins, etc.)
/// 2. Walk the dep graph from root:
///    - Direct deps in `dependencies` -> Production
///    - Direct deps in `devDependencies` -> Dev
///    - Direct deps in `optionalDependencies` -> Optional
///    - Deps matching build tool patterns -> Build
///    - Everything else -> Transitive (inheriting highest-priority context from parents)
/// 3. A dep can be reachable through multiple paths; take the HIGHEST priority
///    context (Production > Transitive > Optional > Build > Dev).

pub struct DepClassifier {
    /// package_key ("name@version") -> DepContext
    pub classifications: HashMap<String, DepContext>,
}

/// Build tool package name patterns (prefixes) that indicate build-only deps.
const BUILD_PATTERNS: &[&str] = &[
    "webpack", "rollup", "esbuild", "vite", "parcel", "swc",
    "babel", "postcss", "autoprefixer", "tailwindcss",
    "tsup", "unbuild", "turbo",
];

impl DepClassifier {
    /// Classify all packages in the dependency graph.
    ///
    /// - `root_deps`: name -> version range from package.json `dependencies`
    /// - `root_dev_deps`: name -> version range from `devDependencies`
    /// - `root_optional_deps`: name -> version range from `optionalDependencies`
    /// - `dep_graph`: adjacency list — package_key -> Vec<child_package_key>
    /// - `resolved_versions`: package name -> resolved version
    pub fn classify(
        root_deps: &HashMap<String, String>,
        root_dev_deps: &HashMap<String, String>,
        root_optional_deps: &HashMap<String, String>,
        dep_graph: &HashMap<String, Vec<String>>,
        resolved_versions: &HashMap<String, String>,
    ) -> Self {
        let mut classifications = HashMap::new();

        // Step 1: Classify direct deps
        for (name, _) in root_deps {
            let key = format!(
                "{}@{}",
                name,
                resolved_versions.get(name).unwrap_or(&"*".to_string())
            );
            let ctx = if is_build_dep(name) {
                DepContext::Build
            } else {
                DepContext::Production
            };
            classifications.insert(key, ctx);
        }

        for (name, _) in root_optional_deps {
            let key = format!(
                "{}@{}",
                name,
                resolved_versions.get(name).unwrap_or(&"*".to_string())
            );
            classifications.entry(key).or_insert(DepContext::Optional);
        }

        for (name, _) in root_dev_deps {
            let key = format!(
                "{}@{}",
                name,
                resolved_versions.get(name).unwrap_or(&"*".to_string())
            );
            classifications.entry(key).or_insert(DepContext::Dev);
        }

        // Step 2: BFS to classify transitive deps
        let mut queue: Vec<String> = classifications.keys().cloned().collect();
        let mut visited: HashSet<String> = queue.iter().cloned().collect();

        while let Some(parent_key) = queue.pop() {
            let parent_ctx = *classifications
                .get(&parent_key)
                .unwrap_or(&DepContext::Transitive);

            if let Some(children) = dep_graph.get(&parent_key) {
                for child_key in children {
                    let child_ctx = DepContext::Transitive;
                    let effective = higher_priority(
                        *classifications.get(child_key).unwrap_or(&child_ctx),
                        inherit_context(parent_ctx),
                    );
                    classifications.insert(child_key.clone(), effective);

                    if visited.insert(child_key.clone()) {
                        queue.push(child_key.clone());
                    }
                }
            }
        }

        Self { classifications }
    }

    pub fn get(&self, package_key: &str) -> DepContext {
        *self
            .classifications
            .get(package_key)
            .unwrap_or(&DepContext::Transitive)
    }
}

fn is_build_dep(name: &str) -> bool {
    BUILD_PATTERNS.iter().any(|pat| name.starts_with(pat))
}

/// Child of a prod dep is Transitive (still matters).
/// Child of a dev dep is Dev (doesn't matter in prod).
fn inherit_context(parent: DepContext) -> DepContext {
    match parent {
        DepContext::Production => DepContext::Transitive,
        DepContext::Dev => DepContext::Dev,
        DepContext::Build => DepContext::Build,
        DepContext::Optional => DepContext::Transitive,
        DepContext::Transitive => DepContext::Transitive,
    }
}

/// Return whichever context has higher priority for risk purposes.
/// Production > Transitive > Optional > Build > Dev
fn higher_priority(a: DepContext, b: DepContext) -> DepContext {
    let rank = |c: &DepContext| match c {
        DepContext::Production => 5,
        DepContext::Transitive => 4,
        DepContext::Optional => 3,
        DepContext::Build => 2,
        DepContext::Dev => 1,
    };
    if rank(&a) >= rank(&b) { a } else { b }
}
```

```rust
// crates/better-core/src/audit/filter.rs

use super::scoring::{DepContext, ScoredVuln};

/// Filter configuration for smart audit output.
#[derive(Debug, Clone)]
pub struct AuditFilter {
    /// Only show vulnerabilities affecting production deps
    pub prod_only: bool,
    /// Minimum effective score to report (0.0 - 10.0)
    pub min_score: f64,
    /// Exclude dev dependencies entirely
    pub ignore_dev: bool,
    /// Exclude build dependencies
    pub ignore_build: bool,
    /// Only show vulns with available fixes
    pub fixable_only: bool,
    /// Severity floor: only show vulns at or above this level
    pub min_severity: Option<super::scoring::Severity>,
}

impl Default for AuditFilter {
    fn default() -> Self {
        Self {
            prod_only: false,
            min_score: 0.0,
            ignore_dev: false,
            ignore_build: false,
            fixable_only: false,
            min_severity: None,
        }
    }
}

impl AuditFilter {
    /// Parse filter from CLI flags.
    pub fn from_args(
        prod_only: bool,
        min_score: Option<f64>,
        ignore_dev: bool,
        fixable_only: bool,
    ) -> Self {
        let mut f = Self::default();
        f.prod_only = prod_only;
        if let Some(s) = min_score {
            f.min_score = s;
        }
        f.ignore_dev = ignore_dev || prod_only;
        f.ignore_build = prod_only;
        f.fixable_only = fixable_only;
        f
    }

    /// Apply this filter to a list of scored vulnerabilities.
    pub fn apply(&self, vulns: &[ScoredVuln]) -> Vec<ScoredVuln> {
        vulns
            .iter()
            .filter(|v| {
                if self.prod_only
                    && v.context != DepContext::Production
                    && v.context != DepContext::Transitive
                {
                    return false;
                }
                if self.ignore_dev && v.context == DepContext::Dev {
                    return false;
                }
                if self.ignore_build && v.context == DepContext::Build {
                    return false;
                }
                if v.effective_score < self.min_score {
                    return false;
                }
                if self.fixable_only && v.fix_available.is_none() {
                    return false;
                }
                if let Some(ref min_sev) = self.min_severity {
                    if v.severity < *min_sev {
                        return false;
                    }
                }
                true
            })
            .cloned()
            .collect()
    }
}
```

```rust
#[napi]
pub fn smart_audit(input: SmartAuditInput) -> napi::Result<SmartAuditOutput> {
    let classifier = DepClassifier::classify(
        &input.root_deps,
        &input.root_dev_deps,
        &input.root_optional_deps,
        &input.dep_graph,
        &input.resolved_versions,
    );

    let filter = AuditFilter::from_args(
        input.prod_only,
        input.min_score,
        input.ignore_dev,
        input.fixable_only,
    );

    let scored: Vec<ScoredVuln> = input.vulns.iter().map(|v| {
        let ctx = classifier.get(&v.package_key);
        ScoredVuln {
            effective_score: score_vuln(v.severity, ctx),
            context: ctx,
            context_weight: ctx.weight(),
            ..v.clone()
        }
    }).collect();

    let filtered = filter.apply(&scored);

    Ok(SmartAuditOutput {
        total: scored.len(),
        filtered: filtered.len(),
        vulns: filtered,
    })
}
```

```rust
use serde::{Deserialize, Serialize};
use chrono::NaiveDate;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct AuditAllowList {
    pub version: u32,
    pub waivers: Vec<Waiver>,
    #[serde(default)]
    pub global_ignore: Option<GlobalIgnore>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Waiver {
    pub id: String,              // CVE or GHSA ID
    pub reason: String,          // Required: why this is waived
    pub author: String,          // Who approved
    pub expires: Option<String>, // ISO date, None = permanent
    pub packages: Vec<String>,   // Scoped to specific packages, empty = all
    pub created: String,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct GlobalIgnore {
    pub contexts: Vec<String>,   // "dev", "build", "optional"
    pub max_severity: Option<String>,
}

impl AuditAllowList {
    /// Load from .betterauditrc.json in project root.
    pub fn load(project_root: &Path) -> Result<Self, AuditAllowListError> {
        let path = project_root.join(".betterauditrc.json");
        if !path.exists() {
            return Ok(Self::empty());
        }
        let content = std::fs::read_to_string(&path)?;
        let list: Self = serde_json::from_str(&content)?;
        Ok(list)
    }

    /// Check if a vulnerability is waived.
    pub fn is_waived(
        &self,
        vuln_id: &str,
        package_key: &str,
        today: NaiveDate,
    ) -> WaiverResult {
        for waiver in &self.waivers {
            if waiver.id != vuln_id {
                continue;
            }
            // Check package scope
            if !waiver.packages.is_empty()
                && !waiver.packages.iter().any(|p| p == package_key)
            {
                continue;
            }
            // Check expiry
            if let Some(ref exp) = waiver.expires {
                if let Ok(exp_date) = NaiveDate::parse_from_str(exp, "%Y-%m-%d") {
                    if today > exp_date {
                        return WaiverResult::Expired(waiver.clone());
                    }
                }
            }
            return WaiverResult::Waived(waiver.clone());
        }
        WaiverResult::NotWaived
    }

    pub fn empty() -> Self {
        Self {
            version: 1,
            waivers: vec![],
            global_ignore: None,
        }
    }
}

#[derive(Debug)]
pub enum WaiverResult {
    Waived(Waiver),
    Expired(Waiver),
    NotWaived,
}
```

```javascript
if (values["add-ignore"]) {
  const id = values["add-ignore"];
  const reason = values["reason"] || "No reason provided";
  const expires = values["expires"] || null;
  const author = process.env.USER || process.env.USERNAME || "unknown";

  const rcPath = path.join(projectRoot, ".betterauditrc.json");
  let rc = { version: 1, waivers: [] };
  try {
    rc = JSON.parse(await fs.readFile(rcPath, "utf8"));

  // Check for duplicate
  if (rc.waivers.some(w => w.id === id)) {
    printText(`Waiver for ${id} already exists.`);
    return;
  }

  rc.waivers.push({
    id,
    reason,
    author,
    expires,
    packages: [],
    created: new Date().toISOString().split("T")[0],
  });

  await fs.writeFile(rcPath, JSON.stringify(rc, null, 2) + "\n");
  printText(`Added waiver for ${id} (expires: ${expires || "never"})`);
  return;
}
```

```rust
impl StrictLayout {
    pub fn materialize(
        &self,
        cas: &ContentAddressableStore,
    ) -> Result<StrictMaterializeStats, Error> {
        let mut stats = StrictMaterializeStats::default();

        // Phase 1: Create store directories and populate from CAS
        for (pkg_key, pkg_info) in &self.packages {
            let pkg_dir = self
                .store_dir
                .join(pkg_key)
                .join("node_modules")
                .join(&pkg_info.name);
            fs::create_dir_all(&pkg_dir)?;

            // Link files from CAS
            for file_entry in &pkg_info.files {
                let cas_path = cas.path_for(&file_entry.integrity);
                let target = pkg_dir.join(&file_entry.relative_path);
                if let Some(parent) = target.parent() {
                    fs::create_dir_all(parent)?;
                }
                // Try hardlink, fallback to copy
                match fs::hard_link(&cas_path, &target) {
                    Ok(()) => stats.files_linked += 1,
                    Err(_) => {
                        fs::copy(&cas_path, &target)?;
                        stats.files_copied += 1;
                    }
                }
            }
        }

        // Phase 2: Create internal dependency symlinks
        for (pkg_key, links) in &self.internal_links {
            let pkg_nm = self.store_dir.join(pkg_key).join("node_modules");
            fs::create_dir_all(&pkg_nm)?;

            for (dep_name, target) in links {
                let link_path = pkg_nm.join(dep_name);
                if link_path.exists() || link_path.symlink_metadata().is_ok() {
                    continue; // Already the real package dir
                }
                #[cfg(unix)]
                std::os::unix::fs::symlink(target, &link_path)?;
                #[cfg(windows)]
                std::os::windows::fs::symlink_dir(target, &link_path)?;
                stats.symlinks += 1;
            }
        }

        // Phase 3: Create root node_modules symlinks
        let root_nm = self.store_dir.parent().unwrap();
        for (name, target) in &self.root_links {
            let link_path = root_nm.join(name);
            // Handle scoped packages: create @scope/ dir first
            if let Some(parent) = link_path.parent() {
                if parent != root_nm {
                    fs::create_dir_all(parent)?;
                }
            }
            if link_path.symlink_metadata().is_ok() {
                fs::remove_file(&link_path)
                    .or_else(|_| fs::remove_dir_all(&link_path))?;
            }
            #[cfg(unix)]
            std::os::unix::fs::symlink(target, &link_path)?;
            stats.root_symlinks += 1;
        }

        Ok(stats)
    }
}
```

```rust
/// Detect phantom dependencies: packages that are imported
/// but not declared in the importing package's package.json.
pub fn detect_phantom_deps(
    project_root: &Path,
    lock_graph: &LockGraph,
) -> Vec<PhantomDep> {
    let mut phantoms = Vec::new();

    // Scan JS/TS files for require() and import statements
    let imports = scan_imports(project_root);

    // Map bare specifiers to package names
    let imported_packages: HashSet<String> = imports
        .iter()
        .filter_map(|imp| extract_package_name(imp))
        .collect();

    // Load declared deps from package.json
    let declared: HashSet<String> = lock_graph
        .direct_deps()
        .map(|d| d.name.clone())
        .collect();

    // Phantoms = imported but not declared
    for pkg in &imported_packages {
        if !declared.contains(pkg) && !is_builtin_module(pkg) {
            phantoms.push(PhantomDep {
                name: pkg.clone(),
                imported_by: imports
                    .iter()
                    .filter(|i| {
                        extract_package_name(i).as_deref() == Some(pkg.as_str())
                    })
                    .map(|i| i.file.clone())
                    .collect(),
            });
        }
    }

    phantoms
}
```

```rust
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ApprovalList {
    pub version: u32,
    pub mode: ApprovalMode,
    pub packages: HashMap<String, PackageApproval>,
    pub scopes: HashMap<String, ScopeApproval>,
    pub settings: ApprovalSettings,
}

#[derive(Debug, Clone, Copy, Serialize, Deserialize)]
#[serde(rename_all = "lowercase")]
pub enum ApprovalMode {
    /// Only approved packages can be installed (default for enterprise)
    Allowlist,
    /// All packages allowed except explicitly denied
    Denylist,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PackageApproval {
    pub approved_versions: Vec<String>,
    pub approved_by: String,
    pub approved_at: String,
    pub reason: String,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub review_url: Option<String>,
}

impl ApprovalList {
    pub fn is_approved(&self, name: &str, version: &str) -> ApprovalResult {
        // 1. Check scope auto-approve
        if let Some(scope) = extract_scope(name) {
            if let Some(scope_rule) = self.scopes.get(scope) {
                if scope_rule.auto_approve {
                    return ApprovalResult::Approved(ApprovalSource::ScopeAutoApprove);
                }
            }
        }

        // 2. Check package-level approval
        if let Some(approval) = self.packages.get(name) {
            for range in &approval.approved_versions {
                if version_matches(version, range) {
                    return ApprovalResult::Approved(ApprovalSource::Explicit);
                }
            }
            return ApprovalResult::VersionNotApproved {
                approved_ranges: approval.approved_versions.clone(),
            };
        }

        // 3. Not found
        ApprovalResult::NotApproved
    }
}
```

```rust
// crates/better-core/src/analyze/import_scanner.rs

use std::collections::HashSet;
use std::path::{Path, PathBuf};
use regex::Regex;

/// Scan JS/TS files for import/require statements.
/// Returns a set of bare specifier package names.
pub fn scan_imports(project_root: &Path) -> HashSet<String> {
    let mut packages = HashSet::new();

    // Walk src/, lib/, app/, pages/, components/, and root-level JS/TS files
    let scan_dirs = [
        "src", "lib", "app", "pages", "components", "server", "scripts",
    ];
    let extensions = ["js", "jsx", "ts", "tsx", "mjs", "cjs", "mts", "cts"];

    let mut files_to_scan = Vec::new();

    for dir in &scan_dirs {
        let dir_path = project_root.join(dir);
        if dir_path.is_dir() {
            walk_dir(&dir_path, &extensions, &mut files_to_scan);
        }
    }

    // Also scan root-level files
    for entry in std::fs::read_dir(project_root).into_iter().flatten() {
        if let Ok(entry) = entry {
            let path = entry.path();
            if path.is_file() {
                if let Some(ext) = path.extension().and_then(|e| e.to_str()) {
                    if extensions.contains(&ext) {
                        files_to_scan.push(path);
                    }
                }
            }
        }
    }

    // Regex patterns for import detection
    let import_re = Regex::new(
        r#"(?:import|export)\s+.*?from\s+['"]([@\w][^'"]*?)['"]"#
    )
    .unwrap();
    let require_re =
        Regex::new(r#"require\(\s*['"]([@\w][^'"]*?)['"]\s*\)"#).unwrap();
    let dynamic_import_re =
        Regex::new(r#"import\(\s*['"]([@\w][^'"]*?)['"]\s*\)"#).unwrap();

    for file in &files_to_scan {
        if let Ok(content) = std::fs::read_to_string(file) {
            for cap in import_re.captures_iter(&content) {
                if let Some(spec) = cap.get(1) {
                    if let Some(pkg) =
                        extract_package_name_from_specifier(spec.as_str())
                    {
                        packages.insert(pkg);
                    }
                }
            }
            for cap in require_re.captures_iter(&content) {
                if let Some(spec) = cap.get(1) {
                    if let Some(pkg) =
                        extract_package_name_from_specifier(spec.as_str())
                    {
                        packages.insert(pkg);
                    }
                }
            }
            for cap in dynamic_import_re.captures_iter(&content) {
                if let Some(spec) = cap.get(1) {
                    if let Some(pkg) =
                        extract_package_name_from_specifier(spec.as_str())
                    {
                        packages.insert(pkg);
                    }
                }
            }
        }
    }

    packages
}

/// Extract package name from a bare specifier.
/// "lodash/fp" -> "lodash"
/// "@types/node" -> "@types/node"
/// "@scope/pkg/deep" -> "@scope/pkg"
/// "./local" -> None (relative)
fn extract_package_name_from_specifier(spec: &str) -> Option<String> {
    if spec.starts_with('.') || spec.starts_with('/') {
        return None; // relative or absolute path
    }
    if spec.starts_with('@') {
        // Scoped: @scope/name
        let parts: Vec<&str> = spec.splitn(3, '/').collect();
        if parts.len() >= 2 {
            Some(format!("{}/{}", parts[0], parts[1]))
        } else {
            None
        }
    } else {
        // Unscoped: name or name/subpath
        Some(spec.split('/').next().unwrap_or(spec).to_string())
    }
}

/// Walk directory recursively, collecting files with matching extensions.
fn walk_dir(dir: &Path, extensions: &[&str], files: &mut Vec<PathBuf>) {
    let ignore_dirs = [
        "node_modules",
        ".git",
        "dist",
        "build",
        ".next",
```

```rust
// crates/better-core/src/analyze/unused.rs

use super::import_scanner::scan_imports;
use std::collections::{HashMap, HashSet};
use std::path::Path;

/// Packages that should be excluded from unused detection:
/// - @types/* (imported implicitly by TypeScript)
/// - Scripts-only packages (used in package.json scripts but not imported)
/// - Bundler plugins (postcss-*, babel-plugin-*, eslint-*, etc.)
/// - CLI tools (referenced in scripts, not imported)
const EXCLUDE_PREFIXES: &[&str] = &[
    "@types/",
    "eslint-config-",
    "eslint-plugin-",
    "babel-plugin-",
    "babel-preset-",
    "postcss-",
    "@postcss/",
    "prettier-plugin-",
    "typescript",
];

pub struct UnusedResult {
    pub unused: Vec<UnusedPackage>,
    pub maybe_unused: Vec<UnusedPackage>, // might be used in scripts
    pub scanned_files: usize,
    pub total_deps: usize,
}

pub struct UnusedPackage {
    pub name: String,
    pub version: String,
    pub is_dev: bool,
    pub possible_script_use: bool,
}

pub fn detect_unused(
    project_root: &Path,
    dependencies: &HashMap<String, String>,
    dev_dependencies: &HashMap<String, String>,
    scripts: &HashMap<String, String>,
) -> UnusedResult {
    let imported = scan_imports(project_root);

    // Also scan for packages referenced in scripts
    let script_refs: HashSet<String> = scripts
        .values()
        .flat_map(|cmd| extract_package_refs_from_script(cmd))
        .collect();

    let mut unused = Vec::new();
    let mut maybe_unused = Vec::new();

    let all_deps: Vec<(&String, &String, bool)> = dependencies
        .iter()
        .map(|(k, v)| (k, v, false))
        .chain(dev_dependencies.iter().map(|(k, v)| (k, v, true)))
        .collect();

    for (name, version, is_dev) in &all_deps {
        // Skip excluded patterns
        if EXCLUDE_PREFIXES.iter().any(|p| name.starts_with(p)) {
            continue;
        }

        if !imported.contains(*name) {
            let in_scripts = script_refs.contains(*name);
            let pkg = UnusedPackage {
                name: name.to_string(),
                version: version.to_string(),
                is_dev: *is_dev,
                possible_script_use: in_scripts,
            };
            if in_scripts {
                maybe_unused.push(pkg);
            } else {
                unused.push(pkg);
            }
        }
    }

    UnusedResult {
        scanned_files: 0, // filled by caller
        total_deps: all_deps.len(),
        unused,
        maybe_unused,
    }
}

fn extract_package_refs_from_script(cmd: &str) -> Vec<String> {
    // Extract binary names from scripts like "tsc && jest --coverage"
    cmd.split_whitespace()
        .filter(|w| !w.starts_with('-') && !w.starts_with('/') && !w.starts_with('.'))
        .filter(|w| !["&&", "||", "|", ";", "npx", "node", "better"].contains(w))
        .map(|w| w.to_string())
        .collect()
}
```

```rust
// crates/better-core/src/resolve/dedupe.rs

use semver::{Version, VersionReq};
use std::collections::HashMap;

#[derive(Debug, Clone, serde::Serialize)]
pub struct DupeAnalysis {
    pub package_name: String,
    pub versions: Vec<String>,
    pub ranges_requesting: Vec<RangeInfo>,
    pub suggested_version: Option<String>,
    pub can_auto_fix: bool,
    pub savings_bytes: Option<u64>,
}

#[derive(Debug, Clone, serde::Serialize)]
pub struct RangeInfo {
    pub requested_by: String, // parent package
    pub range: String,        // "^1.0.0"
    pub resolved: String,     // "1.2.3"
}

/// Analyze duplicates and suggest overrides.
///
/// For each duplicated package:
/// 1. Collect all version ranges that pull it in
/// 2. Find the newest version that satisfies ALL ranges
/// 3. If found -> can_auto_fix = true, suggest that version
/// 4. If not found -> suggest overrides.
pub fn analyze_dupes(lock_graph: &LockGraph) -> Vec<DupeAnalysis> {
    let mut by_name: HashMap<String, Vec<&PackageInfo>> = HashMap::new();

    for pkg in lock_graph.packages() {
        by_name.entry(pkg.name.clone()).or_default().push(pkg);
    }

    let mut analyses = Vec::new();

    for (name, versions) in &by_name {
        if versions.len() <= 1 {
            continue;
        }

        let version_strings: Vec<String> =
            versions.iter().map(|v| v.version.clone()).collect();

        // Collect all ranges requesting this package
        let ranges: Vec<RangeInfo> = lock_graph
            .dependents_of(name)
            .map(|(parent, range, resolved)| RangeInfo {
                requested_by: parent,
                range,
                resolved,
            })
            .collect();

        // Try to find a single version satisfying all ranges
        let suggested = find_universal_version(&version_strings, &ranges);

        analyses.push(DupeAnalysis {
            package_name: name.clone(),
            versions: version_strings,
            ranges_requesting: ranges,
            can_auto_fix: suggested.is_some(),
            suggested_version: suggested,
            savings_bytes: None, // calculated separately
        });
    }

    analyses.sort_by(|a, b| b.versions.len().cmp(&a.versions.len()));
    analyses
}

fn find_universal_version(
    versions: &[String],
    ranges: &[RangeInfo],
) -> Option<String> {
    // Parse all ranges
    let reqs: Vec<VersionReq> = ranges
        .iter()
        .filter_map(|r| VersionReq::parse(&r.range).ok())
        .collect();

    // Find highest version that satisfies all
    let mut candidates: Vec<Version> = versions
        .iter()
        .filter_map(|v| Version::parse(v).ok())
        .collect();
    candidates.sort();
    candidates.reverse();

    for candidate in &candidates {
        if reqs.iter().all(|req| req.matches(candidate)) {
            return Some(candidate.to_string());
        }
    }

    None
}
```

**SPDX parser (`crates/better-core/src/license/spdx.rs`):**

```rust
/// Parse SPDX license expressions.
///
/// Handles:
/// - Simple: "MIT"
/// - AND: "MIT AND Apache-2.0"
/// - OR: "MIT OR Apache-2.0"
/// - WITH: "Apache-2.0 WITH LLVM-exception"
/// - Parenthesized: "(MIT OR Apache-2.0) AND BSD-3-Clause"

#[derive(Debug, Clone)]
pub enum SpdxExpr {
    License(String),
    And(Box<SpdxExpr>, Box<SpdxExpr>),
    Or(Box<SpdxExpr>, Box<SpdxExpr>),
    With(String, String), // license WITH exception
}

impl SpdxExpr {
    pub fn parse(input: &str) -> Result<Self, SpdxError> {
        let tokens = tokenize(input)?;
        parse_expr(&tokens, 0).map(|(expr, _)| expr)
    }

    /// Check if this expression is satisfied by an allow-list.
    /// For OR: at least one branch must be in the allow-list.
    /// For AND: all branches must be in the allow-list.
    pub fn is_allowed(&self, allowed: &HashSet<String>) -> bool {
        match self {
            Self::License(id) => allowed.contains(id),
            Self::Or(a, b) => a.is_allowed(allowed) || b.is_allowed(allowed),
            Self::And(a, b) => a.is_allowed(allowed) && b.is_allowed(allowed),
            Self::With(license, _) => allowed.contains(license),
        }
    }

    /// Check if this expression contains any denied license.
    pub fn is_denied(&self, denied: &HashSet<String>) -> bool {
        match self {
            Self::License(id) => denied.contains(id),
            Self::Or(a, b) => {
                a.is_denied(denied) && b.is_denied(denied)
            } // both branches denied
            Self::And(a, b) => {
                a.is_denied(denied) || b.is_denied(denied)
            } // any branch denied
            Self::With(license, _) => denied.contains(license),
        }
    }

    /// Extract all individual license IDs.
    pub fn licenses(&self) -> Vec<String> {
        match self {
            Self::License(id) => vec![id.clone()],
            Self::Or(a, b) | Self::And(a, b) => {
                let mut v = a.licenses();
                v.extend(b.licenses());
                v
            }
            Self::With(license, _) => vec![license.clone()],
        }
    }
}

#[derive(Debug)]
pub enum SpdxError {
    UnexpectedToken(String),
    UnexpectedEnd,
    InvalidExpression,
}

#[derive(Debug, Clone, PartialEq)]
enum Token {
    License(String),
    And,
    Or,
    With,
    LParen,
    RParen,
}

fn tokenize(input: &str) -> Result<Vec<Token>, SpdxError> {
    let mut tokens = Vec::new();
    let mut chars = input.chars().peekable();

    while let Some(&ch) = chars.peek() {
        if ch.is_whitespace() {
            chars.next();
            continue;
        }
        if ch == '(' {
            tokens.push(Token::LParen);
            chars.next();
        } else if ch == ')' {
            tokens.push(Token::RParen);
            chars.next();
        } else {
            // Read a word
            let mut word = String::new();
            while let Some(&c) = chars.peek() {
                if c.is_whitespace() || c == '(' || c == ')' {
                    break;
                }
                word.push(c);
                chars.next();
            }
            match word.as_str() {
                "AND" => tokens.push(Token::And),
                "OR" => tokens.push(Token::Or),
                "WITH" => tokens.push(Token::With),
                _ => tokens.push(Token::License(word)),
            }
        }
    }

    Ok(tokens)
}

fn parse_expr(
    tokens: &[Token],
    pos: usize,
) -> Result<(SpdxExpr, usize), SpdxError> {
    let (mut left, mut pos) = parse_primary(tokens, pos)?;

    while pos < tokens.len() {
        match &tokens[pos] {
            Token::And => {
                let (right, new_pos) = parse_primary(tokens, pos + 1)?;
                left = SpdxExpr::And(Box::new(left), Box::new(right));
                pos = new_pos;
            }
            Token::Or => {
                let (right, new_pos) = parse_primary(tokens, pos + 1)?;
                left = SpdxExpr::Or(Box::new(left), Box::new(right));
                pos = new_pos;
            }
            Token::With => {
                if let SpdxExpr::License(license) = left {
                    let (exception, new_pos) = parse_primary(tokens, pos + 1)?;
                    if let SpdxExpr::License(exc_name) = exception {
                        left = SpdxExpr::With(license, exc_name);
                        pos = new_pos;
                    } else {
                        return Err(SpdxError::InvalidExpression);
                    }
                } else {
                    return Err(SpdxError::InvalidExpression);
                }
            }
            _ => break,
        }
    }

    Ok((left, pos))
}

fn parse_primary(
    tokens: &[Token],
    pos: usize,
) -> Result<(SpdxExpr, usize), SpdxError> {
    if pos >= tokens.len() {
        return Err(SpdxError::UnexpectedEnd);
    }
    match &tokens[pos] {
        Token::License(id) => Ok((SpdxExpr::License(id.clone()), pos + 1)),
        Token::LParen => {
            let (expr, new_pos) = parse_expr(tokens, pos + 1)?;
            if new_pos >= tokens.len() || tokens[new_pos] != Token::RParen {
                return Err(SpdxError::InvalidExpression);
            }
            Ok((expr, new_pos + 1))
        }
        other => Err(SpdxError::UnexpectedToken(format!("{:?}", other))),
    }
}
```

```rust
// crates/better-core/src/audit/cache.rs

use std::path::{Path, PathBuf};
use std::time::{Duration, SystemTime};
use serde::{Deserialize, Serialize};

const DEFAULT_TTL_HOURS: u64 = 24;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct CacheMeta {
    pub version: u32,
    pub last_refresh: String, // ISO 8601
    pub ttl_hours: u64,
    pub entries: usize,
}

pub struct VulnCache {
    root: PathBuf,
    ttl: Duration,
}

impl VulnCache {
    pub fn new(cache_root: &Path) -> Self {
        let root = cache_root.join("osv");
        Self {
            root,
            ttl: Duration::from_secs(DEFAULT_TTL_HOURS * 3600),
        }
    }

    pub fn with_ttl(mut self, hours: u64) -> Self {
        self.ttl = Duration::from_secs(hours * 3600);
        self
    }

    /// Look up cached OSV response for a package.
    /// Returns None if not cached or expired.
    pub fn get(
        &self,
        ecosystem: &str,
        name: &str,
        version: &str,
    ) -> Option<CachedResponse> {
        let path = self.entry_path(ecosystem, name, version);
        if !path.exists() {
            return None;
        }

        // Check TTL
        let metadata = std::fs::metadata(&path).ok()?;
        let modified = metadata.modified().ok()?;
        if SystemTime::now().duration_since(modified).ok()? > self.ttl {
            return None; // Expired
        }

        let content = std::fs::read_to_string(&path).ok()?;
        serde_json::from_str(&content).ok()
    }

    /// Store an OSV response in the cache.
    pub fn put(
        &self,
        ecosystem: &str,
        name: &str,
        version: &str,
        response: &CachedResponse,
    ) -> Result<(), std::io::Error> {
        let path = self.entry_path(ecosystem, name, version);
        if let Some(parent) = path.parent() {
            std::fs::create_dir_all(parent)?;
        }
        let json = serde_json::to_string_pretty(response)
            .map_err(|e| std::io::Error::new(std::io::ErrorKind::Other, e))?;
        std::fs::write(&path, json)?;
        Ok(())
    }

    /// Force refresh: delete all cached entries.
    pub fn clear(&self) -> Result<usize, std::io::Error> {
        let mut count = 0;
        if self.root.exists() {
            count = count_files(&self.root);
            std::fs::remove_dir_all(&self.root)?;
        }
        Ok(count)
    }

    /// Get cache statistics.
    pub fn stats(&self) -> CacheStats {
        let entries = if self.root.exists() {
            count_files(&self.root)
        } else {
            0
        };
        let size = if self.root.exists() {
            dir_size(&self.root)
        } else {
            0
        };
        CacheStats {
            entries,
            size_bytes: size,
            ttl_hours: self.ttl.as_secs() / 3600,
        }
    }

    fn entry_path(
        &self,
        ecosystem: &str,
        name: &str,
        version: &str,
    ) -> PathBuf {
        // Handle scoped packages: @scope/name -> @scope/name
        self.root
            .join(ecosystem)
            .join(name)
            .join(format!("{}.json", version))
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct CachedResponse {
    pub vulns: Vec<serde_json::Value>,
    pub queried_at: String,
}

#[derive(Debug, Clone)]
pub struct CacheStats {
    pub entries: usize,
    pub size_bytes: u64,
    pub ttl_hours: u64,
}

fn count_files(dir: &Path) -> usize {
    let mut count = 0;
    if let Ok(entries) = std::fs::read_dir(dir) {
        for entry in entries.flatten() {
            let path = entry.path();
            if path.is_dir() {
                count += count_files(&path);
            } else if path.is_file()
                && path.extension().map_or(false, |e| e == "json")
            {
                count += 1;
            }
        }
    }
    count
}

fn dir_size(dir: &Path) -> u64 {
    let mut size = 0;
    if let Ok(entries) = std::fs::read_dir(dir) {
        for entry in entries.flatten() {
            let path = entry.path();
            if path.is_dir() {
                size += dir_size(&path);
            } else if path.is_file() {
                size += std::fs::metadata(&path).map(|m| m.len()).unwrap_or(0);
            }
        }
    }
    size
}
```

```javascript
// In osv.js, wrap queryBatch:

export async function queryBatchCached(packages, cache, options = {}) {
  const { refresh = false, offline = false } = options;

  if (refresh) {
    cache.clear();
  }

  const uncached = [];
  const results = [];

  for (const pkg of packages) {
    if (!refresh) {
      const cached = cache.get("npm", pkg.name, pkg.version);
      if (cached) {
        results.push({
          name: pkg.name,
          version: pkg.version,
          vulns: cached.vulns,
        });
        continue;
      }
    }

    if (offline) {
      // In offline mode, treat uncached as no-vulns
      results.push({ name: pkg.name, version: pkg.version, vulns: [] });
      continue;
    }

    uncached.push(pkg);
  }

  // Batch query uncached packages
  if (uncached.length > 0) {
    const freshResults = await queryBatch(uncached);
    for (const result of freshResults) {
      cache.put("npm", result.name, result.version, {
        vulns: result.vulns,
        queried_at: new Date().toISOString(),
      });
      results.push(result);
    }
  }

  return results;
}
```

```javascript
    process.env.HOME || process.env.USERPROFILE,
    ".better",
    "cache"
  );

  switch (osvAction) {

      const stats = vulnCache.stats();
      printText(`OSV vulnerability cache:`);
      printText(`  Entries: ${stats.entries}`);
      printText(`  Size: ${formatBytes(stats.size_bytes)}`);
      printText(`  TTL: ${stats.ttl_hours}h`);
      printText(`  Location: ${cacheRoot}/osv/`);
      break;
    }
    case "clear": {
      const count = vulnCache.clear();
      printText(`Cleared ${count} cached entries.`);
      break;
    }
    case "refresh": {
      printText("Refreshing vulnerability cache...");
      vulnCache.clear();
      // Re-run audit to populate cache
      printText("Run 'better audit' to repopulate the cache.");
      break;
    }
    default:
      printText("Usage: better cache osv [stats|clear|refresh]");
  }
  break;
}
```
