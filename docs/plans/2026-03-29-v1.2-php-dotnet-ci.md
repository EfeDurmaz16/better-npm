const PACKAGIST_API: &str = "https://repo.packagist.org";

pub struct PackagistClient {
    client: reqwest::Client,
}

impl PackagistClient {
    pub fn new() -> Self {
        Self {
            client: reqwest::Client::builder()
                .user_agent("better/1.2")
                .build()
                .expect("failed to build HTTP client"),
        }
    }

    /// GET /p2/{vendor}/{package}.json — metadata v2 API
    pub async fn get_package_info(
        &self,
        vendor: &str,
        package: &str,
    ) -> Result<Vec<PackagistVersion>, PhpError> {
        let url = format!("{}/p2/{}/{}.json", PACKAGIST_API, vendor, package);
        let resp = self.client.get(&url).send().await
            .map_err(|e| PhpError::Packagist(e.to_string()))?;
        if !resp.status().is_success() {
            return Err(PhpError::Packagist(format!("{}/{} not found", vendor, package)));
        }
        let data: serde_json::Value = resp.json().await
            .map_err(|e| PhpError::Packagist(e.to_string()))?;
        let key = format!("{}/{}", vendor, package);
        let versions: Vec<PackagistVersion> = data["packages"][&key]
            .as_array()
            .map(|arr| arr.iter()
                .filter_map(|v| serde_json::from_value(v.clone()).ok())
                .collect())
            .unwrap_or_default();
        Ok(versions)
    }

    /// Download dist archive and store in CAS
    pub async fn fetch_to_cas(
        &self,
        package: &ComposerPackage,
        cas_root: &Path,
    ) -> Result<PathBuf, PhpError> {
        let dist = package.dist.as_ref()
            .ok_or_else(|| PhpError::Packagist("No dist info".into()))?;
        let resp = self.client.get(&dist.url).send().await
            .map_err(|e| PhpError::Packagist(e.to_string()))?;
        let bytes = resp.bytes().await
            .map_err(|e| PhpError::Packagist(e.to_string()))?;

        if let Some(ref expected_sha) = dist.shasum {
            if !expected_sha.is_empty() {
                let actual_sha = sha256_hex(&bytes);
                if &actual_sha != expected_sha {
                    return Err(PhpError::Packagist(format!(
                        "SHA mismatch for {}: expected {}, got {}",
                        package.name, expected_sha, actual_sha
                    )));
                }
            }
        }

        let cas_path = cas_root
            .join("php")
            .join(package.name.replace('/', "--"))
            .join(&package.version);
        std::fs::create_dir_all(&cas_path)?;
        extract_zip(&bytes, &cas_path)?;
        Ok(cas_path)
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PackagistVersion {
    pub name: String,
    pub version: String,
    pub version_normalized: Option<String>,
    pub dist: Option<ComposerDist>,
    pub require: Option<HashMap<String, String>>,
    pub license: Option<Vec<String>>,
}

/// Composer version constraint parser: ^, ~, >=, <=, !=, ||, *, dev-*
pub fn parse_composer_constraint(constraint: &str) -> Result<ComposerConstraint, PhpError> {
    let parts: Vec<&str> = constraint.split("||").map(|s| s.trim()).collect();
    let ranges = parts.iter().map(|part| {
        let trimmed = part.trim();
        if trimmed.starts_with('^') {
            ConstraintRange::Caret(trimmed[1..].to_string())
        } else if trimmed.starts_with('~') {
            ConstraintRange::Tilde(trimmed[1..].to_string())
        } else if trimmed.starts_with(">=") {
            ConstraintRange::Gte(trimmed[2..].trim().to_string())
        } else if trimmed.starts_with("<=") {
            ConstraintRange::Lte(trimmed[2..].trim().to_string())
        } else if trimmed == "*" {
            ConstraintRange::Any
        } else if trimmed.starts_with("dev-") {
            ConstraintRange::Dev(trimmed.to_string())
        } else {
            ConstraintRange::Exact(trimmed.to_string())
        }
    }).collect();
    Ok(ComposerConstraint { ranges })
}

#[derive(Debug, Clone)]
pub struct ComposerConstraint { pub ranges: Vec<ConstraintRange> }

#[derive(Debug, Clone)]
pub enum ConstraintRange {
    Exact(String), Caret(String), Tilde(String),
    Gte(String), Lte(String), Any, Dev(String),
}

// crates/better-core/src/engines/dotnet.rs

use serde::{Deserialize, Serialize};
use std::path::{Path, PathBuf};

pub struct DotNetEngine {
    pub cas_root: PathBuf,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct CsprojManifest {
    pub project_name: String,
    pub target_framework: String,
    pub packages: Vec<NuGetReference>,
    pub project_references: Vec<String>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct NuGetReference {
    pub name: String,
    pub version: String,
    pub include_assets: Option<String>,
    pub exclude_assets: Option<String>,
    pub private_assets: Option<String>,
}

pub fn parse_csproj(path: &Path) -> Result<CsprojManifest, DotNetError> {
    let content = std::fs::read_to_string(path)?;
    let doc = roxmltree::Document::parse(&content)
        .map_err(|e| DotNetError::ParseFailed(e.to_string()))?;
    let root = doc.root_element();

    let target_framework = root.descendants()
        .find(|n| n.has_tag_name("TargetFramework"))
        .and_then(|n| n.text())
        .unwrap_or("net8.0")
        .to_string();

    let project_name = path.file_stem()
        .and_then(|s| s.to_str())
        .unwrap_or("Unknown")
        .to_string();

    let packages: Vec<NuGetReference> = root.descendants()
        .filter(|n| n.has_tag_name("PackageReference"))
        .filter_map(|node| {
            let name = node.attribute("Include")?;
            let version = node.attribute("Version")
                .or_else(|| node.children()
                    .find(|c| c.has_tag_name("Version"))
                    .and_then(|c| c.text()))
                .unwrap_or("*");
            Some(NuGetReference {
                name: name.to_string(),
                version: version.to_string(),
                include_assets: node.attribute("IncludeAssets").map(String::from),
                exclude_assets: node.attribute("ExcludeAssets").map(String::from),
                private_assets: node.attribute("PrivateAssets").map(String::from),
            })
        })
        .collect();

    let project_references = root.descendants()
        .filter(|n| n.has_tag_name("ProjectReference"))
        .filter_map(|node| node.attribute("Include").map(String::from))
        .collect();

    Ok(CsprojManifest { project_name, target_framework, packages, project_references })
}

/// Parse legacy packages.config XML
pub fn parse_packages_config(path: &Path) -> Result<Vec<NuGetReference>, DotNetError> {
    let content = std::fs::read_to_string(path)?;
    let doc = roxmltree::Document::parse(&content)
        .map_err(|e| DotNetError::ParseFailed(e.to_string()))?;
    Ok(doc.root_element().children()
        .filter(|n| n.has_tag_name("package"))
        .filter_map(|node| {
            Some(NuGetReference {
                name: node.attribute("id")?.to_string(),
                version: node.attribute("version")?.to_string(),
                include_assets: None,
                exclude_assets: None,
                private_assets: None,
            })
        })
        .collect())
}

impl DotNetEngine {
    pub fn new(cas_root: PathBuf) -> Self { Self { cas_root } }
    pub fn detect(project_root: &Path) -> bool {
        walkdir::WalkDir::new(project_root)
            .max_depth(2)
            .into_iter()
            .filter_map(|e| e.ok())
            .any(|e| {
                let name = e.file_name().to_string_lossy();
                name.ends_with(".csproj") || name.ends_with(".fsproj")
                    || name == "packages.config"
            })
    }
}

#[derive(Debug, thiserror::Error)]
pub enum DotNetError {
    #[error("Failed to parse: {0}")]
    ParseFailed(String),
    #[error("NuGet error: {0}")]
    NuGet(String),
    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),
}

            client: reqwest::Client::builder()
                .user_agent("better/1.2").build()
                .expect("failed to build HTTP client"),
            service_index: None,
        }
    }

    /// Lazy-load service index to discover API endpoints
    async fn ensure_service_index(&mut self) -> Result<(), DotNetError> {
        if self.service_index.is_some() { return Ok(()); }
        let resp = self.client.get(NUGET_V3_INDEX).send().await
            .map_err(|e| DotNetError::NuGet(e.to_string()))?;
        self.service_index = Some(resp.json().await
            .map_err(|e| DotNetError::NuGet(e.to_string()))?);
        Ok(())
    }

    /// Download .nupkg (zip) and store in CAS
    pub async fn fetch_to_cas(
        &self,
        package_id: &str,
        version: &str,
        cas_root: &Path,
    ) -> Result<PathBuf, DotNetError> {
        let url = format!(
            "https://api.nuget.org/v3-flatcontainer/{}/{}/{}.{}.nupkg",
            package_id.to_lowercase(), version.to_lowercase(),
            package_id.to_lowercase(), version.to_lowercase(),
        );
        let resp = self.client.get(&url).send().await
            .map_err(|e| DotNetError::NuGet(e.to_string()))?;
        let bytes = resp.bytes().await
            .map_err(|e| DotNetError::NuGet(e.to_string()))?;

        let cas_path = cas_root.join("nuget").join(package_id.to_lowercase()).join(version);
        std::fs::create_dir_all(&cas_path)?;
        extract_zip(&bytes, &cas_path)?;
        Ok(cas_path)
    }
}

/// NuGet version range parser: "[1.0.0, 2.0.0)", "1.0.0", "[1.0.0,)"
pub fn parse_nuget_version_range(range: &str) -> Result<NuGetVersionRange, DotNetError> {
    let trimmed = range.trim();
    if !trimmed.contains(',') && !trimmed.starts_with('[') && !trimmed.starts_with('(') {
        return Ok(NuGetVersionRange {
            min: Some(trimmed.to_string()), max: None,
            min_inclusive: true, max_inclusive: false,
        });
    }
    let min_inclusive = trimmed.starts_with('[');
    let max_inclusive = trimmed.ends_with(']');
    let inner = trimmed.trim_matches(|c| c == '[' || c == '(' || c == ']' || c == ')');
    let parts: Vec<&str> = inner.split(',').map(|s| s.trim()).collect();
    Ok(NuGetVersionRange {
        min: if parts[0].is_empty() { None } else { Some(parts[0].to_string()) },
        max: parts.get(1).and_then(|s| if s.is_empty() { None } else { Some(s.to_string()) }),
        min_inclusive, max_inclusive,
    })
}

#[derive(Debug, Clone)]
pub struct NuGetVersionRange {
    pub min: Option<String>,
    pub max: Option<String>,
    pub min_inclusive: bool,
    pub max_inclusive: bool,
}

// crates/better-core/src/ci.rs

use serde::Serialize;
use std::path::Path;
use std::time::Instant;

#[derive(Debug, Clone, Serialize)]
pub struct CiResult {
    pub steps: Vec<CiStepResult>,
    pub total_duration_ms: u64,
    pub all_passed: bool,
    pub sbom_path: Option<String>,
    pub receipt_path: Option<String>,
}

#[derive(Debug, Clone, Serialize)]
pub struct CiStepResult {
    pub name: String,
    pub status: CiStepStatus,
    pub duration_ms: u64,
    pub details: serde_json::Value,
}

#[derive(Debug, Clone, Serialize)]
pub enum CiStepStatus { Passed, Failed, Skipped, Warning }

pub struct CiConfig {
    pub skip_audit: bool,
    pub skip_policy: bool,
    pub skip_sbom: bool,
    pub audit_severity: String,
    pub sbom_format: String,
}

/// Pipeline: install --frozen -> verify provenance -> audit -> policy -> sbom -> receipt
pub async fn run_ci(project_root: &Path, config: CiConfig) -> CiResult {
    let start = Instant::now();
    let mut steps = vec![];
    let mut all_passed = true;

    // Step 1: frozen install (fail fast if lockfile mismatch)
    let s1 = run_step("install --frozen", || run_frozen_install(project_root)).await;
    if matches!(s1.status, CiStepStatus::Failed) { all_passed = false; }
    let install_failed = matches!(s1.status, CiStepStatus::Failed);
    steps.push(s1);
    if install_failed {
        return CiResult { steps, total_duration_ms: start.elapsed().as_millis() as u64,
            all_passed: false, sbom_path: None, receipt_path: None };
    }

    // Step 2: verify provenance
    let s2 = run_step("verify-provenance", || verify_provenance(project_root)).await;
    if matches!(s2.status, CiStepStatus::Failed) { all_passed = false; }
    steps.push(s2);

    // Step 3: audit
    if !config.skip_audit {
        let s3 = run_step("audit", || run_audit(project_root, &config.audit_severity)).await;
        if matches!(s3.status, CiStepStatus::Failed) { all_passed = false; }
        steps.push(s3);
    }

    // Step 4: policy
    if !config.skip_policy {
        let s4 = run_step("policy", || check_policy(project_root)).await;
        if matches!(s4.status, CiStepStatus::Failed) { all_passed = false; }
        steps.push(s4);
    }

    // Step 5: SBOM
    let sbom_path = if !config.skip_sbom {
        let s5 = run_step("sbom", || generate_sbom(project_root, &config.sbom_format)).await;
        steps.push(s5);
        Some(format!("{}/better-sbom.json", project_root.display()))
    } else { None };

    // Step 6: receipt
    let receipt_path = format!("{}/.better-receipt.json", project_root.display());
    let s6 = run_step("receipt", || generate_receipt(project_root, &steps)).await;
    steps.push(s6);

    CiResult {
        steps,
        total_duration_ms: start.elapsed().as_millis() as u64,
        all_passed,
        sbom_path,
        receipt_path: Some(receipt_path),
    }
}

// crates/better-core/src/diff.rs

use serde::Serialize;
use std::collections::HashMap;

#[derive(Debug, Clone, Serialize)]
pub struct DepDiff {
    pub added: Vec<DepChange>,
    pub removed: Vec<DepChange>,
    pub updated: Vec<DepUpdate>,
    pub unchanged: usize,
    pub security_impact: SecurityImpact,
}

#[derive(Debug, Clone, Serialize)]
pub struct DepChange {
    pub name: String,
    pub version: String,
    pub ecosystem: String,
    pub license: String,
    pub size_bytes: u64,
}

#[derive(Debug, Clone, Serialize)]
pub struct DepUpdate {
    pub name: String,
    pub from_version: String,
    pub to_version: String,
    pub ecosystem: String,
    pub change_type: UpdateType,  // Major, Minor, Patch, PreRelease
    pub new_vulns: Vec<String>,
    pub fixed_vulns: Vec<String>,
}

#[derive(Debug, Clone, Serialize)]
pub struct SecurityImpact {
    pub new_vulnerabilities: usize,
    pub fixed_vulnerabilities: usize,
    pub new_high_risk_packages: Vec<String>,
    pub removed_maintainers: Vec<String>,
}

/// Compute diff between two dependency snapshots
pub fn compute_dep_diff(
    before: &HashMap<String, (String, String)>,  // name -> (version, ecosystem)
    after: &HashMap<String, (String, String)>,
) -> DepDiff {
    let mut added = vec![];
    let mut removed = vec![];
    let mut updated = vec![];
    let mut unchanged = 0;

    for (name, (after_ver, eco)) in after {
        match before.get(name) {
            None => added.push(DepChange {
                name: name.clone(), version: after_ver.clone(),
                ecosystem: eco.clone(), license: String::new(), size_bytes: 0,
            }),
            Some((before_ver, _)) if before_ver != after_ver => {
                updated.push(DepUpdate {
                    name: name.clone(), from_version: before_ver.clone(),
                    to_version: after_ver.clone(), ecosystem: eco.clone(),
                    change_type: classify_update(before_ver, after_ver),
                    new_vulns: vec![], fixed_vulns: vec![],
                });
            }
            _ => { unchanged += 1; }
        }
    }

    for (name, (ver, eco)) in before {
        if !after.contains_key(name) {
            removed.push(DepChange {
                name: name.clone(), version: ver.clone(),
                ecosystem: eco.clone(), license: String::new(), size_bytes: 0,
            });
        }
    }

    DepDiff { added, removed, updated, unchanged,
        security_impact: SecurityImpact {
            new_vulnerabilities: 0, fixed_vulnerabilities: 0,
            new_high_risk_packages: vec![], removed_maintainers: vec![],
        },
    }
}

fn classify_update(from: &str, to: &str) -> UpdateType {
    match (semver::Version::parse(from), semver::Version::parse(to)) {
        (Ok(f), Ok(t)) if f.major != t.major => UpdateType::Major,
        (Ok(f), Ok(t)) if f.minor != t.minor => UpdateType::Minor,
        (Ok(f), Ok(t)) if f.patch != t.patch => UpdateType::Patch,
        _ => UpdateType::PreRelease,
    }
}

#[derive(Debug, Clone, Serialize)]
pub enum UpdateType { Major, Minor, Patch, PreRelease }
```

// crates/better-core/src/lock_merge.rs

/// Three-way merge: git calls `better lock merge %O %A %B`
///   %O = base (ancestor), %A = ours, %B = theirs
pub fn merge_lockfiles(
    base_path: &Path,
    ours_path: &Path,
    theirs_path: &Path,
) -> Result<MergeResult, MergeError> {
    let base = read_lockfile_json(base_path)?;
    let ours = read_lockfile_json(ours_path)?;
    let theirs = read_lockfile_json(theirs_path)?;

    let base_pkgs = extract_packages(&base);
    let our_pkgs = extract_packages(&ours);
    let their_pkgs = extract_packages(&theirs);

    let mut merged = ours.clone();
    let mut conflicts = vec![];
    let mut auto_resolved = 0;

    // Strategy:
    // - New in theirs, not in base/ours -> add
    // - Changed in theirs, unchanged in ours -> take theirs
    // - Changed in both -> take newer semver version
    // - Removed in theirs, unchanged in ours -> remove
    // - Removed in theirs, changed in ours -> keep ours (conflict warning)

    for (name, their_info) in &their_pkgs {
        let base_info = base_pkgs.get(name);
        let our_info = our_pkgs.get(name);

        match (base_info, our_info) {
            (None, None) => {
                insert_package(&mut merged, name, their_info);
                auto_resolved += 1;
            }
            (Some(b), Some(o)) if b == o && b != their_info => {
                update_package(&mut merged, name, their_info);
                auto_resolved += 1;
            }
            (Some(_b), Some(o)) if o != their_info => {
                // Both changed: take newer version
                let resolved = if semver_gt(&extract_version(their_info), &extract_version(o)) {
                    update_package(&mut merged, name, their_info);
                    "theirs (newer)"
                } else { "ours (newer)" };
                auto_resolved += 1;
                conflicts.push(MergeConflict {
                    package: name.clone(),
                    resolution: resolved.to_string(),
                });
            }
            _ => {}
        }
    }

    // Handle removals
    for (name, _) in &base_pkgs {
        if !their_pkgs.contains_key(name) {
            if our_pkgs.get(name) == base_pkgs.get(name) {
                remove_package(&mut merged, name);
                auto_resolved += 1;
            }
        }
    }

    std::fs::write(ours_path, serde_json::to_string_pretty(&merged)?)?;

    Ok(MergeResult { success: true, conflicts, auto_resolved })
}

#[derive(Debug)]
pub struct MergeResult {
    pub success: bool,
    pub conflicts: Vec<MergeConflict>,
    pub auto_resolved: usize,
}

#[derive(Debug)]
pub struct MergeConflict {
    pub package: String,
    pub resolution: String,
}
```
