/// Compute canonical JSON for signature verification.
/// Per OSP spec: serialize all fields EXCEPT `provider_signature`,
/// keys sorted alphabetically, no extra whitespace.
pub fn canonical_json(manifest: &ServiceManifest) -> Result<String, serde_json::Error> {
    let mut value = serde_json::to_value(manifest)?;
    if let Some(obj) = value.as_object_mut() {
        obj.remove("provider_signature");
    }
    // serde_json with sorted keys
    let sorted = sort_json_keys(&value);
    serde_json::to_string(&sorted)
}

fn sort_json_keys(value: &serde_json::Value) -> serde_json::Value {
    match value {
        serde_json::Value::Object(map) => {
            let mut sorted: serde_json::Map<String, serde_json::Value> = serde_json::Map::new();
            let mut keys: Vec<&String> = map.keys().collect();
            keys.sort();
            for key in keys {
                sorted.insert(key.clone(), sort_json_keys(&map[key]));
            }
            serde_json::Value::Object(sorted)
        }
        serde_json::Value::Array(arr) => {
            serde_json::Value::Array(arr.iter().map(sort_json_keys).collect())
        }
        other => other.clone(),
    }
}

/// Verify the Ed25519 signature on a ServiceManifest.
/// 1. Extract provider_public_key from manifest
/// 2. Compute canonical JSON (excluding provider_signature)
/// 3. Verify signature using ed25519-dalek
pub fn verify_manifest_signature(manifest: &ServiceManifest) -> Result<bool, OspError> {
    use ed25519_dalek::{Signature, VerifyingKey};
    use super::manifest::canonical_json;

    let pubkey_b64 = manifest.provider_public_key.as_ref()
        .ok_or(OspError::MissingPublicKey)?;

    let pubkey_bytes = base64_url_decode(pubkey_b64)?;
    let verifying_key = VerifyingKey::from_bytes(
        &pubkey_bytes.try_into().map_err(|_| OspError::InvalidPublicKey)?
    ).map_err(|_| OspError::InvalidPublicKey)?;

    let canonical = canonical_json(manifest)
        .map_err(|e| OspError::SerializationError(e.to_string()))?;

    let sig_bytes = base64_url_decode(&manifest.provider_signature)?;
    let signature = Signature::from_bytes(
        &sig_bytes.try_into().map_err(|_| OspError::InvalidSignature)?
    );

    use ed25519_dalek::Verifier;
    verifying_key.verify(canonical.as_bytes(), &signature)
        .map_err(|_| OspError::SignatureVerificationFailed)?;

    Ok(true)
}

fn base64_url_decode(input: &str) -> Result<Vec<u8>, OspError> {
    use base64::Engine;
    base64::engine::general_purpose::URL_SAFE_NO_PAD
        .decode(input)
        .map_err(|e| OspError::Base64Error(e.to_string()))
}

/// Generate a cryptographic nonce (32 bytes, base64url).

    let mut bytes = [0u8; 32];

/// Generate an idempotency key from offering+tier+project (SHA-256, base64url).
fn generate_idempotency_key(offering_id: &str, tier_id: &str, project_name: &str) -> String {
    use sha2::{Sha256, Digest};
    let mut hasher = Sha256::new();
    hasher.update(offering_id.as_bytes());
    hasher.update(tier_id.as_bytes());
    hasher.update(project_name.as_bytes());
    hasher.update(chrono::Utc::now().to_rfc3339().as_bytes());
    base64_url_encode(&hasher.finalize())
}

fn base64_url_encode(bytes: &[u8]) -> String {
    use base64::Engine;
    base64::engine::general_purpose::URL_SAFE_NO_PAD.encode(bytes)
}

/// Decrypted credential payload — the actual secrets.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct DecryptedCredentials {
    /// Raw key-value pairs from the provider (e.g., connection_string, api_key, etc.)
    pub fields: serde_json::Map<String, serde_json::Value>,
}

/// Decrypt a CredentialBundle using the agent's X25519 secret key.
///
/// Flow:
/// 1. Decode ephemeral_public_key (provider's ephemeral X25519 pubkey)
/// 2. Perform X25519 Diffie-Hellman with agent's secret key -> shared secret
/// 3. Derive symmetric key from shared secret (HKDF or direct)
/// 4. Decrypt encrypted_payload using AES-256-GCM with the nonce
/// 5. Parse decrypted JSON into DecryptedCredentials
pub fn decrypt_credential_bundle(
    bundle: &CredentialBundle,
    agent_secret_key: &[u8; 32],  // X25519 secret key
) -> Result<DecryptedCredentials, super::discovery::OspError> {
    use x25519_dalek::{StaticSecret, PublicKey};

    let ephemeral_pubkey_bytes = base64_url_decode(
        bundle.ephemeral_public_key.as_ref()
            .ok_or(super::discovery::OspError::DecryptionFailed)?
    )?;

    let agent_secret = StaticSecret::from(*agent_secret_key);
    let ephemeral_pubkey = PublicKey::from(
        <[u8; 32]>::try_from(ephemeral_pubkey_bytes.as_slice())
            .map_err(|_| super::discovery::OspError::DecryptionFailed)?
    );

    // X25519 DH -> shared secret
    let shared_secret = agent_secret.diffie_hellman(&ephemeral_pubkey);

    // Decrypt with AES-256-GCM
    let ciphertext = base64_url_decode(&bundle.encrypted_payload)?;
    let nonce_bytes = base64_url_decode(
        bundle.nonce.as_ref()
            .ok_or(super::discovery::OspError::DecryptionFailed)?
    )?;

    let plaintext = aes_256_gcm_decrypt(
        shared_secret.as_bytes(),
        &nonce_bytes,
        &ciphertext,
    )?;

    let fields: serde_json::Map<String, serde_json::Value> =
        serde_json::from_slice(&plaintext)
            .map_err(|e| super::discovery::OspError::ParseError(e.to_string()))?;

    Ok(DecryptedCredentials { fields })
}

fn aes_256_gcm_decrypt(
    key: &[u8],
    nonce: &[u8],
    ciphertext: &[u8],
) -> Result<Vec<u8>, super::discovery::OspError> {
    use aes_gcm::{Aes256Gcm, Key, Nonce};
    use aes_gcm::aead::Aead;
    use aes_gcm::KeyInit;

    let key = Key::<Aes256Gcm>::from_slice(key);
    let cipher = Aes256Gcm::new(key);
    let nonce = Nonce::from_slice(nonce);

    cipher.decrypt(nonce, ciphertext)
        .map_err(|_| super::discovery::OspError::DecryptionFailed)
}

fn base64_url_decode(input: &str) -> Result<Vec<u8>, super::discovery::OspError> {
    use base64::Engine;
    base64::engine::general_purpose::URL_SAFE_NO_PAD
        .decode(input)
        .map_err(|e| super::discovery::OspError::Base64Error(e.to_string()))
}

/// Resolve an osp:// URI to a decrypted credential field value.
///
/// Format: osp://provider_domain/offering_id_suffix/field_name
/// Example: osp://supabase.com/postgres/connection_string
///
/// 1. Parse URI -> extract provider, offering suffix, field
/// 2. Look up in vault -> decrypt -> return field value
pub fn resolve_osp_uri(
    uri: &str,
    vault: &super::vault::Vault,
    agent_secret_key: &[u8; 32],
) -> Result<String, super::discovery::OspError> {
    let parsed = parse_osp_uri(uri)?;

    let entry = vault.get_entry(&parsed.provider, &parsed.offering_suffix)
        .ok_or_else(|| super::discovery::OspError::ResourceNotFound(
            format!("No vault entry for {}:{}", parsed.provider, parsed.offering_suffix)
        ))?;

    let decrypted = decrypt_credential_bundle(&entry.credential_bundle, agent_secret_key)?;

    decrypted.fields.get(&parsed.field)
        .and_then(|v| v.as_str())
        .map(|s| s.to_string())
        .ok_or_else(|| super::discovery::OspError::ResourceNotFound(
            format!("Field '{}' not found in credentials", parsed.field)
        ))
}

struct OspUri {
    provider: String,
    offering_suffix: String,
    field: String,
}

fn parse_osp_uri(uri: &str) -> Result<OspUri, super::discovery::OspError> {
    // osp://supabase.com/postgres/connection_string
    let stripped = uri.strip_prefix("osp://")
        .ok_or_else(|| super::discovery::OspError::ParseError(
            format!("Invalid osp:// URI: {}", uri)
        ))?;

    let parts: Vec<&str> = stripped.splitn(3, '/').collect();
    if parts.len() != 3 {
        return Err(super::discovery::OspError::ParseError(
            format!("osp:// URI must have format osp://provider/offering/field, got: {}", uri)
        ));
    }

    Ok(OspUri {
        provider: parts[0].to_string(),
        offering_suffix: parts[1].to_string(),
        field: parts[2].to_string(),
    })
}

/// Derive a machine-local encryption key from hostname + username.
/// NOT a security boundary against root — protects against casual file copy.
fn derive_machine_key() -> [u8; 32] {
    use sha2::{Sha256, Digest};
    let hostname = hostname::get().unwrap_or_default();
    let username = whoami::username();
    let mut hasher = Sha256::new();
    hasher.update(b"better-vault-v1:");
    hasher.update(hostname.as_encoded_bytes());
    hasher.update(b":");
    hasher.update(username.as_bytes());
    let result = hasher.finalize();
    let mut key = [0u8; 32];
    key.copy_from_slice(&result);
    key
}

    use aes_gcm::{Aes256Gcm, Key, Nonce};
    use aes_gcm::aead::Aead;
    use aes_gcm::KeyInit;
    use rand::RngCore;

    let key = Key::<Aes256Gcm>::from_slice(key);
    let cipher = Aes256Gcm::new(key);
    let mut nonce_bytes = [0u8; 12];
    rand::thread_rng().fill_bytes(&mut nonce_bytes);
    let nonce = Nonce::from_slice(&nonce_bytes);

    let ciphertext = cipher.encrypt(nonce, plaintext)
        .map_err(|_| super::discovery::OspError::VaultError("Encryption failed".into()))?;

    // Prepend nonce to ciphertext
    let mut result = Vec::with_capacity(12 + ciphertext.len());
    result.extend_from_slice(&nonce_bytes);
    result.extend_from_slice(&ciphertext);
    Ok(result)
}

fn aes_256_gcm_encrypt_with_embedded_nonce(key: &[u8; 32], plaintext: &[u8]) -> Result<Vec<u8>, super::discovery::OspError> {
    aes_256_gcm_encrypt(key, plaintext)
}

fn aes_256_gcm_decrypt_with_embedded_nonce(key: &[u8; 32], data: &[u8]) -> Result<Vec<u8>, super::discovery::OspError> {
    use aes_gcm::{Aes256Gcm, Key, Nonce};
    use aes_gcm::aead::Aead;
    use aes_gcm::KeyInit;

    if data.len() < 12 {
        return Err(super::discovery::OspError::VaultError("Data too short".into()));
    }
    let (nonce_bytes, ciphertext) = data.split_at(12);
    let key = Key::<Aes256Gcm>::from_slice(key);
    let cipher = Aes256Gcm::new(key);
    let nonce = Nonce::from_slice(nonce_bytes);

    cipher.decrypt(nonce, ciphertext)
        .map_err(|_| super::discovery::OspError::DecryptionFailed)
}

### Task 58: Environment Generation (osp:// URI resolution, .env.osp template, .env output)

**What:** Read `.env.osp` template with `osp://` URIs, resolve each from vault, write `.env`.

**Rust module:** `crates/better-core/src/osp/env_gen.rs` (new file in osp module)

```rust
// crates/better-core/src/osp/env_gen.rs

use std::collections::BTreeMap;
use std::path::Path;

/// Template entry from .env.osp file
#[derive(Debug, Clone)]
pub struct EnvTemplate {
    pub entries: Vec<EnvTemplateEntry>,
}

#[derive(Debug, Clone)]
pub struct EnvTemplateEntry {
    pub key: String,
    pub value: EnvTemplateValue,
    pub comment: Option<String>,
}

#[derive(Debug, Clone)]
pub enum EnvTemplateValue {
    /// An osp:// URI that needs vault resolution
    OspUri(String),
    /// A static value that passes through
    Static(String),
    /// A reference to another env var
    Ref(String),
}

/// Parse a .env.osp template file.
///
/// Format:
/// ```text
/// # Database
/// DATABASE_URL=osp://supabase.com/postgres/connection_string
/// SUPABASE_ANON_KEY=osp://supabase.com/postgres/anon_key
///
/// # Static values
/// NODE_ENV=production
/// PORT=3000
///
/// # References
/// SHADOW_DB_URL=${DATABASE_URL}?schema=shadow
/// ```
pub fn parse_env_template(content: &str) -> EnvTemplate {
    let mut entries = Vec::new();
    let mut current_comment = None;

    for line in content.lines() {
        let trimmed = line.trim();

        if trimmed.is_empty() {
            current_comment = None;
            continue;
        }

        if trimmed.starts_with('#') {
            current_comment = Some(trimmed[1..].trim().to_string());
            continue;
        }

        if let Some((key, raw_value)) = trimmed.split_once('=') {
            let key = key.trim().to_string();
            let raw_value = raw_value.trim().to_string();

            let value = if raw_value.starts_with("osp://") {
                EnvTemplateValue::OspUri(raw_value)
            } else if raw_value.contains("${") {
                EnvTemplateValue::Ref(raw_value)
            } else {
                EnvTemplateValue::Static(raw_value)
            };

            entries.push(EnvTemplateEntry {
                key,
                value,
                comment: current_comment.take(),
            });
        }
    }

    EnvTemplate { entries }
}

/// Resolve all osp:// URIs in a template and generate the final .env content.
pub fn generate_env(
    template: &EnvTemplate,
    vault: &super::vault::Vault,
    agent_secret_key: &[u8; 32],
) -> Result<String, super::discovery::OspError> {
    let mut resolved: BTreeMap<String, String> = BTreeMap::new();
    let mut output = String::new();

    // First pass: resolve all osp:// URIs and static values
    for entry in &template.entries {
        if let Some(comment) = &entry.comment {
            output.push_str(&format!("# {}\n", comment));
        }

        let value = match &entry.value {
            EnvTemplateValue::OspUri(uri) => {
                super::credentials::resolve_osp_uri(uri, vault, agent_secret_key)?
            }
            EnvTemplateValue::Static(s) => s.clone(),
            EnvTemplateValue::Ref(s) => s.clone(), // Resolved in second pass
        };

        resolved.insert(entry.key.clone(), value.clone());
        output.push_str(&format!("{}={}\n", entry.key, value));
    }

    // Second pass: resolve ${REFERENCES}
    for entry in &template.entries {
        if let EnvTemplateValue::Ref(ref_value) = &entry.value {
            let mut final_value = ref_value.clone();
            for (k, v) in &resolved {
                final_value = final_value.replace(&format!("${{{}}}", k), v);
            }
            resolved.insert(entry.key.clone(), final_value);
        }
    }

    Ok(output)
}

/// Read .env.osp, resolve, write .env.
pub fn generate_env_file(
    project_root: &Path,
    vault: &super::vault::Vault,
    agent_secret_key: &[u8; 32],
) -> Result<std::path::PathBuf, super::discovery::OspError> {
    let template_path = project_root.join(".env.osp");
    if !template_path.exists() {
        return Err(super::discovery::OspError::ParseError(
            "No .env.osp template found in project root".into()
        ));
    }

    let content = std::fs::read_to_string(&template_path)
        .map_err(|e| super::discovery::OspError::VaultError(e.to_string()))?;

    let template = parse_env_template(&content);
    let env_content = generate_env(&template, vault, agent_secret_key)?;

    let output_path = project_root.join(".env");
    std::fs::write(&output_path, &env_content)
        .map_err(|e| super::discovery::OspError::VaultError(e.to_string()))?;

    // Ensure .env is in .gitignore
    ensure_gitignore(project_root, ".env");

    Ok(output_path)
}

fn ensure_gitignore(project_root: &Path, pattern: &str) {
    let gitignore_path = project_root.join(".gitignore");
    let content = std::fs::read_to_string(&gitignore_path).unwrap_or_default();
    if !content.lines().any(|l| l.trim() == pattern) {
        let mut new_content = content;
        if !new_content.ends_with('\n') && !new_content.is_empty() {
            new_content.push('\n');
        }
        new_content.push_str(&format!("{}\n", pattern));
        let _ = std::fs::write(&gitignore_path, new_content);
    }
}

    pub minimum_amount: Option<String>,

    PayWhatYouWant,
    PerInstall,

}

    pub name: String,
    pub amount: String,
    pub currency: String,
    pub features: Vec<String>,
}

/// Validate and inject pricing metadata into package.json before publish.
pub fn inject_pricing(
    package_json: &mut serde_json::Value,
    pricing: &PackagePricing,
) -> Result<(), String> {
    // Add under "better" key in package.json:
    // { "better": { "pricing": { ... } } }
    let better = package_json.as_object_mut()
        .ok_or("Invalid package.json")?
        .entry("better")
        .or_insert(serde_json::json!({}));

    better.as_object_mut()
        .ok_or("Invalid better field")?
        .insert("pricing".into(), serde_json::to_value(pricing).map_err(|e| e.to_string())?);

    Ok(())
}

/// Validate pricing before publish.
pub fn validate_pricing(pricing: &PackagePricing) -> Result<(), Vec<String>> {
    let mut errors = vec![];

    if pricing.sardis_wallet_id.is_empty() {
        errors.push("sardis_wallet_id is required".into());
    }

    if let Some(ref amount) = pricing.suggested_amount {
        if amount.parse::<f64>().is_err() {
            errors.push(format!("Invalid suggested_amount: {}", amount));
        }
    }

    if errors.is_empty() { Ok(()) } else { Err(errors) }
}
