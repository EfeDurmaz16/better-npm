```rust
// Global flag handling in main.rs argument parser
struct GlobalFlags {
    json: bool,
    no_color: bool,
    no_interactive: bool,
    verbose: bool,
}

fn parse_global_flags(args: &[String]) -> (GlobalFlags, Vec<String>) {
    let mut flags = GlobalFlags {
        json: false,
        no_color: false,
        no_interactive: false,
        verbose: false,
    };
    let mut remaining = Vec::new();

    for arg in args {
        match arg.as_str() {
            "--json" => flags.json = true,
            "--no-color" => flags.no_color = true,
            "--no-interactive" => flags.no_interactive = true,
            "--verbose" | "-v" => flags.verbose = true,
            _ => remaining.push(arg.clone()),
        }
    }

    (flags, remaining)
}
```

```rust
/// Extract context from a JS/TS package
pub fn extract_js_context(
    package_root: &Path,
    name: &str,
    version: &str,
) -> Result<PackageContext, String> {
    let mut exports = Vec::new();
    let mut patterns = Vec::new();
    let mut gotchas = Vec::new();
    let mut description = String::new();
    let mut types_summary = None;

    // 1. Parse index.d.ts (primary source for TypeScript types)
    let dts_candidates = [
        package_root.join("index.d.ts"),
        package_root.join("dist/index.d.ts"),
        package_root.join("types/index.d.ts"),
    ];
    for dts_path in &dts_candidates {
        if dts_path.exists() {
            let dts_exports = parse_dts(dts_path)?;
            exports.extend(dts_exports);
            break;
        }
    }

    // 2. Parse JSDoc from source files
    let main_file = find_main_file(package_root);
    if let Some(main) = main_file {
        let jsdoc_info = parse_jsdoc(&main)?;
        for info in jsdoc_info {
            // Merge with dts exports or add new
            if let Some(existing) = exports.iter_mut().find(|e| e.name == info.name) {
                if existing.description.is_none() {
                    existing.description = info.description;
                }
            }
        }
    }

    // 3. Parse README for quick_start, patterns, gotchas
    let readme = find_readme(package_root);
    let (quick_start, readme_patterns, readme_gotchas) = if let Some(readme_path) = readme {
        extract_from_readme(&readme_path)?
    } else {
        (String::new(), Vec::new(), Vec::new())
    };
    patterns.extend(readme_patterns);
    gotchas.extend(readme_gotchas);

    // 4. Generate types summary (compact view of all exported types)
    if !exports.is_empty() {
        types_summary = Some(generate_types_summary(&exports));
    }

    // 5. Read description from package.json
    let pkg_json_path = package_root.join("package.json");
    if pkg_json_path.exists() {
        let content = std::fs::read_to_string(&pkg_json_path).unwrap_or_default();
        if let Some(desc) = extract_json_field(&content, "description") {
            description = desc;
        }
    }

    let ctx = PackageContext {
        name: name.to_string(),
        version: version.to_string(),
        ecosystem: "npm".to_string(),
        description,
        exports,
        quick_start,
        patterns,
        gotchas,
        types_summary,
        dependencies: Vec::new(),
        generated_at: chrono_now(),
        markdown: String::new(),
    };

    let markdown = template::render_context(&ctx);
    Ok(PackageContext { markdown, ..ctx })
}

/// Parse .d.ts file for exported symbols
fn parse_dts(path: &Path) -> Result<Vec<ExportedSymbol>, String> {
    let content = std::fs::read_to_string(path)
        .map_err(|e| format!("failed to read {}: {}", path.display(), e))?;

    let mut symbols = Vec::new();

    for line in content.lines() {
        let line = line.trim();
        if !line.starts_with("export ") { continue; }

        let rest = &line["export ".len()..];

        if rest.starts_with("function ") {
            if let Some(sym) = parse_function_signature(rest) {
                symbols.push(sym);
            }
        } else if rest.starts_with("class ") {
            if let Some(sym) = parse_class_declaration(rest) {
                symbols.push(sym);
            }
        } else if rest.starts_with("interface ") {
            if let Some(sym) = parse_interface_declaration(rest) {
                symbols.push(sym);
            }
        } else if rest.starts_with("const ") || rest.starts_with("let ") {
            if let Some(sym) = parse_variable_declaration(rest) {
                symbols.push(sym);
            }
        } else if rest.starts_with("type ") {
            if let Some(sym) = parse_type_alias(rest) {
                symbols.push(sym);
            }
        } else if rest.starts_with("enum ") {
            if let Some(sym) = parse_enum_declaration(rest) {
                symbols.push(sym);
            }
        }
    }

    Ok(symbols)
}
```

```rust
use super::*;

/// Extract context from a Python package
pub fn extract_python_context(
    package_root: &Path,
    name: &str,
    version: &str,
) -> Result<PackageContext, String> {
    let mut exports = Vec::new();
    let mut patterns = Vec::new();
    let mut gotchas = Vec::new();

    // 1. Parse __init__.py for exported symbols and docstrings
    let init_path = package_root.join("__init__.py");
    if init_path.exists() {
        let init_exports = parse_python_init(&init_path)?;
        exports.extend(init_exports);
    }

    // 2. Parse __all__ if defined
    let _all_exports = parse_dunder_all(&init_path)?;

    // 3. Parse type hints from .pyi stub files
    let pyi_path = package_root.join("__init__.pyi");
    if pyi_path.exists() {
        let stub_exports = parse_pyi_stubs(&pyi_path)?;
        for stub in stub_exports {
            if let Some(existing) = exports.iter_mut().find(|e| e.name == stub.name) {
                if stub.signature.is_some() {
                    existing.signature = stub.signature;
                }
                if stub.return_type.is_some() {
                    existing.return_type = stub.return_type;
                }
            } else {
                exports.push(stub);
            }
        }
    }

    // 4. Parse py.typed marker (PEP 561)
    let py_typed = package_root.join("py.typed").exists();

    // 5. Parse README
    let readme_candidates = ["README.md", "README.rst", "README.txt", "README"];
    let readme_path = readme_candidates.iter()
        .map(|f| package_root.join(f))
        .find(|p| p.exists());

    let (quick_start, readme_patterns, readme_gotchas) = if let Some(rp) = readme_path {
        extract_from_readme(&rp)?
    } else {
        (String::new(), Vec::new(), Vec::new())
    };
    patterns.extend(readme_patterns);
    gotchas.extend(readme_gotchas);

    // 6. Read metadata from PKG-INFO or METADATA
    let description = read_package_description(package_root)?;

    let ctx = PackageContext {
        name: name.to_string(),
        version: version.to_string(),
        ecosystem: "python".to_string(),
        description,
        exports,
        quick_start,
        patterns,
        gotchas,
        types_summary: if py_typed { Some("Inline type hints (PEP 561)".to_string()) } else { None },
        dependencies: Vec::new(),
        generated_at: chrono_now(),
        markdown: String::new(),
    };

    let markdown = super::template::render_context(&ctx);
    Ok(PackageContext { markdown, ..ctx })
}

/// Parse Python source for class/function definitions with docstrings
fn parse_python_init(path: &Path) -> Result<Vec<ExportedSymbol>, String> {
    let content = std::fs::read_to_string(path)
        .map_err(|e| format!("failed to read {}: {}", path.display(), e))?;

    let mut symbols = Vec::new();
    let mut lines = content.lines().peekable();

    while let Some(line) = lines.next() {
        let trimmed = line.trim();

        if trimmed.starts_with("def ") {
            if let Some(sym) = parse_python_function(trimmed, &mut lines) {
                symbols.push(sym);
            }
        } else if trimmed.starts_with("class ") {
            if let Some(sym) = parse_python_class(trimmed, &mut lines) {
                symbols.push(sym);
            }
        } else if trimmed.contains('=') && !trimmed.starts_with('#') && !trimmed.starts_with(' ') {
            if let Some((name, _)) = trimmed.split_once('=') {
                let name = name.trim();
                if name.chars().all(|c| c.is_ascii_uppercase() || c == '_') {
                    symbols.push(ExportedSymbol {
                        name: name.to_string(),
                        kind: SymbolKind::Constant,
                        signature: None,
                        description: None,
                        params: Vec::new(),
                        return_type: None,
                    });
                }
            }
        }
    }

    Ok(symbols)
}
```

```rust
// crates/better-core/src/context/template.rs
use super::*;

/// Render a PackageContext into LLM-optimized markdown
pub fn render_context(ctx: &PackageContext) -> String {
    let mut md = String::with_capacity(4096);

    // Header
    md.push_str(&format!("# {} v{}\n\n", ctx.name, ctx.version));
    md.push_str(&format!("> {}\n\n", ctx.description));
    md.push_str(&format!("**Ecosystem:** {}  \n", ctx.ecosystem));
    md.push_str(&format!("**Generated:** {}  \n\n", ctx.generated_at));

    // Exports section
    if !ctx.exports.is_empty() {
        md.push_str("## Exports\n\n");

        let functions: Vec<_> = ctx.exports.iter().filter(|e| matches!(e.kind, SymbolKind::Function)).collect();
        let classes: Vec<_> = ctx.exports.iter().filter(|e| matches!(e.kind, SymbolKind::Class)).collect();
        let types: Vec<_> = ctx.exports.iter().filter(|e| matches!(e.kind, SymbolKind::Type | SymbolKind::Interface)).collect();
        let constants: Vec<_> = ctx.exports.iter().filter(|e| matches!(e.kind, SymbolKind::Constant)).collect();

        if !functions.is_empty() {
            md.push_str("### Functions\n\n");
            for f in &functions {
                render_symbol(&mut md, f, &ctx.ecosystem);
            }
        }

        if !classes.is_empty() {
            md.push_str("### Classes\n\n");
            for c in &classes {
                render_symbol(&mut md, c, &ctx.ecosystem);
            }
        }

        if !types.is_empty() {
            md.push_str("### Types\n\n");
            for t in &types {
                render_symbol(&mut md, t, &ctx.ecosystem);
            }
        }

        if !constants.is_empty() {
            md.push_str("### Constants\n\n");
            for c in &constants {
                md.push_str(&format!("- `{}`", c.name));
                if let Some(desc) = &c.description {
                    md.push_str(&format!(" -- {}", desc));
                }
                md.push('\n');
            }
            md.push('\n');
        }
    }

    // Quick Start
    if !ctx.quick_start.is_empty() {
        md.push_str("## Quick Start\n\n");
        md.push_str(&ctx.quick_start);
        md.push_str("\n\n");
    }

    // Patterns
    if !ctx.patterns.is_empty() {
        md.push_str("## Common Patterns\n\n");
        for pattern in &ctx.patterns {
            md.push_str(&format!("### {}\n\n", pattern.title));
            md.push_str(&format!("```{}\n{}\n```\n\n", pattern.language, pattern.code));
        }
    }

    // Gotchas
    if !ctx.gotchas.is_empty() {
        md.push_str("## Gotchas\n\n");
        for gotcha in &ctx.gotchas {
            md.push_str(&format!("- {}\n", gotcha));
        }
        md.push('\n');
    }

    // Types summary
    if let Some(types) = &ctx.types_summary {
        md.push_str("## Types Summary\n\n");
        md.push_str(types);
        md.push_str("\n\n");
    }

    md
}

fn render_symbol(md: &mut String, sym: &ExportedSymbol, ecosystem: &str) {
    let lang = match ecosystem {
        "npm" => "typescript",
        "python" => "python",
        _ => "text",
    };

    if let Some(sig) = &sym.signature {
        md.push_str(&format!("```{}\n{}\n```\n", lang, sig));
    } else {
        md.push_str(&format!("**`{}`**\n", sym.name));
    }

    if let Some(desc) = &sym.description {
        md.push_str(&format!("{}\n", desc));
    }

    if !sym.params.is_empty() {
        md.push_str("\n| Param | Type | Description |\n|-------|------|-------------|\n");
        for p in &sym.params {
            md.push_str(&format!("| `{}{}` | `{}` | {} |\n",
                p.name,
                if p.optional { "?" } else { "" },
                p.type_str.as_deref().unwrap_or("any"),
                p.description.as_deref().unwrap_or(""),
            ));
        }
    }
    md.push('\n');
}
```

```rust
pub struct BulkContextResult {

    pub failed: Vec<(String, String)>,  // (package, error)

/// Generate context for all installed packages in parallel
pub fn generate_all_context(
    project_root: &Path,
    cache_root: &Path,
    force: bool,
) -> Result<BulkContextResult, String> {
    let start = std::time::Instant::now();

    // 1. Read lockfile to get all installed packages
    let lockfile_path = project_root.join("better.lock");
    let lockfile = super::super::lockfile::binary::Lockfile::read(&lockfile_path)?;

    // 2. Build list of (ecosystem, name, version, package_root)
    let mut work_items: Vec<ContextWorkItem> = Vec::new();

    for section in &lockfile.sections {
        for pkg in &section.packages {
            let package_root = resolve_package_root(project_root, section.ecosystem, &pkg.name);
            let cache_path = context_cache_path(cache_root, section.ecosystem, &pkg.name, &pkg.version);

            if !force && cache_path.exists() {
                continue; // Skip cached
            }

            work_items.push(ContextWorkItem {
                ecosystem: section.ecosystem,
                name: pkg.name.clone(),
                version: pkg.version.clone(),
                package_root,
                cache_path,
            });
        }
    }

    let cached = lockfile.sections.iter()
        .map(|s| s.packages.len())
        .sum::<usize>() - work_items.len();

    // 3. Generate in parallel
    use rayon::prelude::*;
    let results: Vec<_> = work_items.par_iter()
        .map(|item| {
            let result = match item.ecosystem {
                super::super::lockfile::binary::EcosystemByte::Npm =>
                    super::js::extract_js_context(&item.package_root, &item.name, &item.version),
                super::super::lockfile::binary::EcosystemByte::Python =>
                    super::python::extract_python_context(&item.package_root, &item.name, &item.version),
                _ => Err(format!("unsupported ecosystem for context: {:?}", item.ecosystem)),
            };

            match result {
                Ok(ctx) => {
                    if let Some(parent) = item.cache_path.parent() {
                        std::fs::create_dir_all(parent).ok();
                    }
                    std::fs::write(&item.cache_path, &ctx.markdown).ok();
                    Ok(item.name.clone())
                }
                Err(e) => Err((item.name.clone(), e)),
            }
        })
        .collect();

    let generated = results.iter().filter(|r| r.is_ok()).count();
    let failed: Vec<_> = results.into_iter()
        .filter_map(|r| r.err())
        .collect();

    let project_context_dir = project_root.join(".better").join("context");
    std::fs::create_dir_all(&project_context_dir).ok();

    Ok(BulkContextResult {
        generated,
        cached,
        failed,
        total_ms: start.elapsed().as_millis() as u64,
        output_dir: project_context_dir.to_string_lossy().to_string(),
    })
}

struct ContextWorkItem {
    ecosystem: super::super::lockfile::binary::EcosystemByte,
    name: String,
    version: String,
    package_root: std::path::PathBuf,
    cache_path: std::path::PathBuf,
}

fn resolve_package_root(
    project_root: &Path,
    ecosystem: super::super::lockfile::binary::EcosystemByte,
    name: &str,
) -> std::path::PathBuf {
    match ecosystem {
        super::super::lockfile::binary::EcosystemByte::Npm =>
            project_root.join("node_modules").join(name),
        super::super::lockfile::binary::EcosystemByte::Python =>
            project_root.join(".venv").join("lib").join("python3").join("site-packages").join(name),
        _ => project_root.join(name),
    }
}

fn context_cache_path(
    cache_root: &Path,
    ecosystem: super::super::lockfile::binary::EcosystemByte,
    name: &str,
    version: &str,
) -> std::path::PathBuf {
    let eco_str = match ecosystem {
        super::super::lockfile::binary::EcosystemByte::Npm => "npm",
        super::super::lockfile::binary::EcosystemByte::Python => "python",
        _ => "other",
    };
    cache_root
        .join("context")
        .join(eco_str)
        .join(format!("{}@{}.md", name, version))
}
```

```rust
// crates/better-core/src/mcp/transport.rs
use super::protocol::*;
use std::io::{BufRead, Write};

pub trait McpTransport {
    fn read_message(&mut self) -> Result<JsonRpcRequest, String>;
    fn write_message(&mut self, response: &JsonRpcResponse) -> Result<(), String>;
}

/// stdio transport (default)
pub struct StdioTransport {
    reader: std::io::BufReader<std::io::Stdin>,
}

impl StdioTransport {
    pub fn new() -> Self {
        Self {
            reader: std::io::BufReader::new(std::io::stdin()),
        }
    }
}

impl McpTransport for StdioTransport {
    fn read_message(&mut self) -> Result<JsonRpcRequest, String> {
        let mut header_line = String::new();
        self.reader.read_line(&mut header_line)
            .map_err(|e| format!("failed to read header: {}", e))?;

        let content_length: usize = header_line
            .trim()
            .strip_prefix("Content-Length: ")
            .ok_or_else(|| "missing Content-Length header".to_string())?
            .parse()
            .map_err(|e| format!("invalid Content-Length: {}", e))?;

        let mut sep = String::new();
        self.reader.read_line(&mut sep).ok();

        let mut body = vec![0u8; content_length];
        std::io::Read::read_exact(&mut self.reader, &mut body)
            .map_err(|e| format!("failed to read body: {}", e))?;

        serde_json::from_slice(&body)
            .map_err(|e| format!("invalid JSON-RPC: {}", e))
    }

    fn write_message(&mut self, response: &JsonRpcResponse) -> Result<(), String> {
        let body = serde_json::to_string(response)
            .map_err(|e| format!("failed to serialize response: {}", e))?;

        let mut stdout = std::io::stdout().lock();
        write!(stdout, "Content-Length: {}\r\n\r\n{}", body.len(), body)
            .map_err(|e| format!("failed to write response: {}", e))?;
        stdout.flush()
            .map_err(|e| format!("failed to flush stdout: {}", e))
    }
}

/// SSE transport
pub struct SseTransport {
    pub port: u16,
```

```rust
pub struct McpServer<T: McpTransport> {
    transport: T,
    initialized: bool,
}

impl<T: McpTransport> McpServer<T> {
    pub fn new(transport: T) -> Self {
        Self {
            transport,
            initialized: false,
        }
    }

    pub fn run(&mut self) -> Result<(), String> {
        loop {
            let request = match self.transport.read_message() {
                Ok(req) => req,
                Err(e) => {
                    eprintln!("transport error: {}", e);
                    break;
                }
            };

            let response = self.handle_request(&request);
            self.transport.write_message(&response)?;
        }
        Ok(())
    }

    fn handle_request(&mut self, request: &JsonRpcRequest) -> JsonRpcResponse {
        match request.method.as_str() {
            "initialize" => {
                self.initialized = true;
                JsonRpcResponse {
                    jsonrpc: "2.0".to_string(),
                    id: request.id.clone(),
                    result: Some(initialize_result()),
                    error: None,
                }
            }
            "initialized" => {
                JsonRpcResponse {
                    jsonrpc: "2.0".to_string(),
                    id: request.id.clone(),
                    result: Some(serde_json::json!({})),
                    error: None,
                }
            }
            "tools/list" => {
                let tool_list = tools::list_tools();
                JsonRpcResponse {
                    jsonrpc: "2.0".to_string(),
                    id: request.id.clone(),
                    result: Some(serde_json::json!({ "tools": tool_list })),
                    error: None,
                }
            }
            "tools/call" => {
                let params = request.params.as_ref()
                    .and_then(|p| p.as_object());

                let (tool_name, args) = match params {
                    Some(p) => {
                        let name = p.get("name")
                            .and_then(|n| n.as_str())
                            .unwrap_or("");
                        let arguments = p.get("arguments")
                            .cloned()
                            .unwrap_or(serde_json::json!({}));
                        (name.to_string(), arguments)
                    }
                    None => {
                        return JsonRpcResponse {
                            jsonrpc: "2.0".to_string(),
                            id: request.id.clone(),
                            result: None,
                            error: Some(JsonRpcError {
                                code: -32602,
                                message: "Invalid params".to_string(),
                                data: None,
                            }),
                        };
                    }
                };

                let tool_result = tools::execute_tool(&tool_name, &args);
                JsonRpcResponse {
                    jsonrpc: "2.0".to_string(),
                    id: request.id.clone(),
                    result: Some(serde_json::to_value(tool_result).unwrap()),
                    error: None,
                }
            }
            _ => {
                JsonRpcResponse {
                    jsonrpc: "2.0".to_string(),
                    id: request.id.clone(),
                    result: None,
                    error: Some(JsonRpcError {
                        code: -32601,
                        message: format!("Method not found: {}", request.method),
                        data: None,
                    }),
                }
```

```rust
pub fn rank(packages: &mut [super::SearchPackage], query: &str) {
    for pkg in packages.iter_mut() {
        pkg.score = compute_score(pkg, query);
    }
    packages.sort_by(|a, b| b.score.partial_cmp(&a.score).unwrap_or(std::cmp::Ordering::Equal));
}

fn compute_score(pkg: &super::SearchPackage, query: &str) -> f64 {
    let mut score = 0.0;

    // Name relevance
    if pkg.name == query { score += 0.30; }
    else if pkg.name.starts_with(query) { score += 0.20; }
    else if pkg.name.contains(query) { score += 0.10; }

    // Popularity (log-scaled)
    let pop = (pkg.downloads_weekly as f64 + 1.0).log10() / 10.0;
    score += pop.min(0.25);

    // Maintenance freshness
    score += 0.10; // placeholder based on last_publish

    // Security
    score += pkg.security_score * 0.15;

    // Type safety
    if pkg.has_types { score += 0.05; }

    // Context availability
    if pkg.has_context { score += 0.10; }

    // Sardis enabled
    if pkg.sardis_enabled { score += 0.05; }

    score.min(1.0)
}
```

```rust
    if has_npm {
        let imports = js_imports::extract_js_imports(project_root)?;
        let declared = read_npm_deps(project_root)?;
        let builtins: HashSet<&str> = NODE_BUILTINS.iter().copied().collect();

        for import in &imports {
            if builtins.contains(import.as_str()) { continue; }
            if import.starts_with('.') { continue; }
            if !declared.contains(import) {
                report.missing.push(MissingDep {
                    name: import.clone(),
                    imported_in: Vec::new(),
                    import_style: "import".to_string(),
                    suggested_version: None,
                    confidence: 0.9,
                });
            }
        }

        for dep in &declared {
            if !imports.contains(dep) {
                report.unused.push(UnusedDep {
                    name: dep.clone(),
                    declared_in: "dependencies".to_string(),
                    version: String::new(),
                    confidence: 0.7,
                });
            }
        }
    }

    if has_python {
        let imports = python_imports::extract_python_imports(project_root)?;
        let declared = read_python_deps(project_root)?;
        let stdlib: HashSet<&str> = PYTHON_STDLIB.iter().copied().collect();
        let import_map: std::collections::HashMap<&str, &str> = PYTHON_IMPORT_TO_PACKAGE.iter().copied().collect();

        for import in &imports {
            if stdlib.contains(import.as_str()) { continue; }

            let package_name = import_map.get(import.as_str())
                .map(|s| s.to_string())
                .unwrap_or_else(|| import.clone());

            if !declared.contains(&package_name) && !declared.contains(import) {
                report.missing.push(MissingDep {
                    name: package_name,
                    imported_in: Vec::new(),
                    import_style: "import".to_string(),
                    suggested_version: None,
                    confidence: 0.85,
                });
            }
        }

        for dep in &declared {
            let import_name = PYTHON_IMPORT_TO_PACKAGE.iter()
                .find(|(_, pkg)| *pkg == dep.as_str())
                .map(|(imp, _)| imp.to_string())
                .unwrap_or_else(|| dep.replace('-', "_"));

            if !imports.contains(&import_name) && !imports.contains(dep) {
                report.unused.push(UnusedDep {
                    name: dep.clone(),
                    declared_in: "dependencies".to_string(),
                    version: String::new(),
                    confidence: 0.7,
                });
            }
        }
    }

    report.scan_ms = start.elapsed().as_millis() as u64;
    Ok(report)
}
```

```rust
use std::path::{Path, PathBuf};

/// Cache layout:
/// ~/.better/context/
///   npm/
///     lodash@4.17.21.md
///     express@4.18.2.md
///   python/
///     requests@2.31.0.md
///     flask@3.1.0.md
///   index.json

#[derive(Debug, serde::Serialize, serde::Deserialize)]
pub struct CacheIndex {
    pub entries: Vec<CacheEntry>,
    pub total_size_bytes: u64,
    pub last_gc: Option<String>,
}

#[derive(Debug, serde::Serialize, serde::Deserialize)]
pub struct CacheEntry {
    pub ecosystem: String,
    pub name: String,
    pub version: String,
    pub path: String,
    pub size_bytes: u64,
    pub generated_at: String,
    pub last_accessed: String,
    pub authored: bool,
}

pub fn cache_root() -> PathBuf {
    let home = std::env::var("HOME").unwrap_or_else(|_| "/tmp".to_string());
    PathBuf::from(home).join(".better").join("context")
}

pub fn cache_path(ecosystem: &str, name: &str, version: &str) -> PathBuf {
    cache_root()
        .join(ecosystem)
        .join(format!("{}@{}.md", name.replace('/', "__"), version))
}

pub fn read_cached(ecosystem: &str, name: &str, version: &str) -> Option<String> {
    let path = cache_path(ecosystem, name, version);
    if path.exists() {
        update_access_time(ecosystem, name, version);
        std::fs::read_to_string(&path).ok()
    } else {
        None
    }
}

pub fn write_cached(
    ecosystem: &str,
    name: &str,
    version: &str,
    content: &str,
    authored: bool,
) -> Result<(), String> {
    let path = cache_path(ecosystem, name, version);
    if let Some(parent) = path.parent() {
        std::fs::create_dir_all(parent)
            .map_err(|e| format!("failed to create cache dir: {}", e))?;
    }
    std::fs::write(&path, content)
        .map_err(|e| format!("failed to write cache: {}", e))?;

    update_index(CacheEntry {
        ecosystem: ecosystem.to_string(),
        name: name.to_string(),
        version: version.to_string(),
        path: path.to_string_lossy().to_string(),
        size_bytes: content.len() as u64,
        generated_at: now_iso(),
        last_accessed: now_iso(),
        authored,
    });

    Ok(())
}

pub fn gc(max_age_days: u64, dry_run: bool) -> Result<GcResult, String> {
    let root = cache_root();
    let index_path = root.join("index.json");

    let index = if index_path.exists() {
        let content = std::fs::read_to_string(&index_path)
            .map_err(|e| format!("failed to read index: {}", e))?;
        serde_json::from_str::<CacheIndex>(&content)
            .unwrap_or(CacheIndex { entries: Vec::new(), total_size_bytes: 0, last_gc: None })
    } else {
        CacheIndex { entries: Vec::new(), total_size_bytes: 0, last_gc: None }
    };

    let cutoff = now_minus_days(max_age_days);
    let mut removed = 0u64;
    let mut freed_bytes = 0u64;
    let mut kept = 0u64;

    let remaining: Vec<CacheEntry> = index.entries.into_iter()
        .filter(|entry| {
            if entry.last_accessed < cutoff {
                if !dry_run {
                    std::fs::remove_file(&entry.path).ok();
                }
                removed += 1;
                freed_bytes += entry.size_bytes;
                false
            } else {
                kept += 1;
                true
            }
        })
        .collect();

    if !dry_run {
        let new_index = CacheIndex {
            total_size_bytes: remaining.iter().map(|e| e.size_bytes).sum(),
            entries: remaining,
            last_gc: Some(now_iso()),
        };
        let json = serde_json::to_string_pretty(&new_index)
            .map_err(|e| format!("failed to serialize index: {}", e))?;
        std::fs::write(&index_path, json)
            .map_err(|e| format!("failed to write index: {}", e))?;
    }

    Ok(GcResult {
        removed,
        freed_bytes,
        kept,
        dry_run,
    })
}

pub fn stats() -> Result<CacheStats, String> {
    let root = cache_root();
    let index_path = root.join("index.json");

    if !index_path.exists() {
        return Ok(CacheStats::default());
    }

    let content = std::fs::read_to_string(&index_path)
        .map_err(|e| format!("failed to read index: {}", e))?;
    let index: CacheIndex = serde_json::from_str(&content)
        .unwrap_or(CacheIndex { entries: Vec::new(), total_size_bytes: 0, last_gc: None });

    let npm_count = index.entries.iter().filter(|e| e.ecosystem == "npm").count();
    let python_count = index.entries.iter().filter(|e| e.ecosystem == "python").count();
    let authored_count = index.entries.iter().filter(|e| e.authored).count();

    Ok(CacheStats {
        total_entries: index.entries.len(),
        total_size_bytes: index.total_size_bytes,
        npm_entries: npm_count,
        python_entries: python_count,
        authored_entries: authored_count,
        last_gc: index.last_gc,
    })
}
```
