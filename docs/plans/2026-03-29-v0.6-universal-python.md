// crates/better-core/src/engine/registry.rs
use super::{Ecosystem, PackageEngine};
use std::path::Path;

pub struct EngineRegistry {
    engines: Vec<Box<dyn PackageEngine>>,
}

#[derive(Debug)]
pub struct DetectionResult {
    pub ecosystems: Vec<Ecosystem>,
    pub is_polyglot: bool,
    pub manifests: Vec<(Ecosystem, std::path::PathBuf)>,
}

impl EngineRegistry {
    pub fn new() -> Self {
        Self { engines: Vec::new() }
    }

    pub fn register(&mut self, engine: Box<dyn PackageEngine>) {
        self.engines.push(engine);
    }

    /// Scan project root for known manifest files, return detected ecosystems
    pub fn detect(&self, project_root: &Path) -> DetectionResult {
        let mut ecosystems = Vec::new();
        let mut manifests = Vec::new();

        for engine in &self.engines {
            let eco = engine.ecosystem();
            for manifest_file in eco.manifest_files() {
                let manifest_path = project_root.join(manifest_file);
                if manifest_path.exists() {
                    if !ecosystems.contains(&eco) {
                        ecosystems.push(eco);
                    }
                    manifests.push((eco, manifest_path));
                }
            }
        }

        let is_polyglot = ecosystems.len() > 1;
        DetectionResult { ecosystems, is_polyglot, manifests }
    }

    pub fn get(&self, ecosystem: Ecosystem) -> Option<&dyn PackageEngine> {
        self.engines.iter()
            .find(|e| e.ecosystem() == ecosystem)
            .map(|e| e.as_ref())
    }
}

/// Build the default registry with all compiled-in engines
pub fn default_registry() -> EngineRegistry {
    let mut reg = EngineRegistry::new();
    reg.register(Box::new(super::npm::NpmEngine));
    reg.register(Box::new(super::python::PythonEngine::new()));
    reg
}

// crates/better-core/src/engine/polyglot.rs
use super::{Ecosystem, EngineError, PackageEngine};
use super::registry::EngineRegistry;
use std::path::Path;

pub struct PolyglotInstall {
    pub results: Vec<(Ecosystem, Result<InstallResult, EngineError>)>,
    pub total_packages: usize,
    pub total_ms: u64,
}

/// Run install across all detected ecosystems in parallel (rayon)
pub fn polyglot_install(
    registry: &EngineRegistry,
    ecosystems: &[Ecosystem],
    project_root: &Path,
    cache_root: &Path,
) -> PolyglotInstall {
    use rayon::prelude::*;

    let results: Vec<_> = ecosystems.par_iter()
        .map(|eco| {
            let engine = registry.get(*eco).unwrap();
            let result = engine.resolve(project_root)
                .and_then(|resolved| {
                    engine.fetch(&resolved.packages, cache_root)?;
                    engine.materialize(&resolved.artifacts, project_root)
                });
            (*eco, result.map(|r| InstallResult { packages: r.count, ms: r.elapsed_ms }))
        })
        .collect();

    let total_packages = results.iter()
        .filter_map(|(_, r)| r.as_ref().ok())
        .map(|r| r.packages)
        .sum();

    PolyglotInstall { results, total_packages, total_ms: 0 }
}

        // Verify it's a valid venv
        let info = VenvInfo::detect(project_root);
        if info.exists {
            return Ok(info);
        }
        // Invalid venv — recreate
        std::fs::remove_dir_all(&venv_path)
            .map_err(|e| format!("failed to remove invalid .venv: {}", e))?;
    }

    // Create venv using the Python interpreter

        .args(["-m", "venv", ".venv", "--without-pip"])
        .current_dir(project_root)
        .status()
        .map_err(|e| format!("failed to create venv: {}", e))?;

    if !status.success() {
        return Err("venv creation failed".to_string());
    }

    // We manage packages ourselves — no pip needed in venv
    Ok(VenvInfo::detect(project_root))

// crates/better-core/src/lockfile/binary.rs

/// Binary lockfile format v2
///
/// Header:
///   magic: [u8; 4]  = b"BTLK"
///   version: u16     = 2
///   flags: u16       = bitfield (compressed, signed, ...)
///   ecosystems: u8   = number of ecosystem sections
///   reserved: [u8; 7]
///
/// For each ecosystem section:
///   discriminator: u8  (0=npm, 1=python, 2=cargo, 3=go, ...)
///   section_len: u64   (byte length of this section)
///   package_count: u32
///   For each package:
///     name_len: u16
///     name: [u8; name_len]
///     version_len: u16
///     version: [u8; version_len]
///     integrity_len: u16
///     integrity: [u8; integrity_len]  (sha256 hex or sri)
///     dep_count: u16
///     deps: [(u16, u16); dep_count]  (name_idx, version_idx)
///     flags: u8  (dev, optional, peer, ...)

pub const MAGIC: [u8; 4] = *b"BTLK";
pub const FORMAT_VERSION: u16 = 2;

#[derive(Debug, Clone, Copy)]
#[repr(u8)]
pub enum EcosystemByte {
    Npm = 0,
    Python = 1,
    Cargo = 2,
    Go = 3,
}

#[derive(Debug)]
pub struct LockfileHeader {
    pub version: u16,
    pub flags: u16,
    pub ecosystem_count: u8,
}

#[derive(Debug)]
pub struct EcosystemSection {
    pub ecosystem: EcosystemByte,
    pub packages: Vec<LockedPackage>,
}

#[derive(Debug, Clone)]
pub struct LockedPackage {
    pub name: String,
    pub version: String,
    pub integrity: String,
    pub dependencies: Vec<(String, String)>,  // (name, version)
    pub dev: bool,
    pub optional: bool,
}

#[derive(Debug)]
pub struct Lockfile {
    pub header: LockfileHeader,
    pub sections: Vec<EcosystemSection>,
}

impl Lockfile {
    /// Read binary lockfile from disk
    pub fn read(path: &std::path::Path) -> Result<Self, String> {
        let data = std::fs::read(path)
            .map_err(|e| format!("failed to read lockfile: {}", e))?;

        if data.len() < 16 {
            return Err("lockfile too small".to_string());
        }
        if &data[0..4] != &MAGIC {
            return Err("invalid lockfile magic".to_string());
        }

        let version = u16::from_le_bytes([data[4], data[5]]);

        // v1 compatibility: single npm section, no ecosystem discriminator
        if version == 1 {
            return Self::read_v1(&data[16..]);
        }

        Self::read_v2(&data)
    }

    /// Write binary lockfile to disk
    pub fn write(&self, path: &std::path::Path) -> Result<(), String> { todo!() }

    fn read_v1(_data: &[u8]) -> Result<Self, String> { todo!() }
    fn read_v2(_data: &[u8]) -> Result<Self, String> { todo!() }

    /// Get packages for a specific ecosystem
    pub fn packages_for(&self, ecosystem: EcosystemByte) -> &[LockedPackage] {
        self.sections.iter()
            .find(|s| s.ecosystem as u8 == ecosystem as u8)
            .map(|s| s.packages.as_slice())
            .unwrap_or(&[])
    }

    /// Merge a new ecosystem section into an existing lockfile
    pub fn merge_section(&mut self, section: EcosystemSection) {
        // Remove existing section for this ecosystem
        self.sections.retain(|s| s.ecosystem as u8 != section.ecosystem as u8);
        self.sections.push(section);
        self.header.ecosystem_count = self.sections.len() as u8;
    }

use super::binary::{Lockfile, EcosystemByte, LockedPackage};

/// Generate JSON sidecar from binary lockfile
/// Format:
/// {
///   "version": 2,
///   "npm": {
///     "packages": { "lodash": { "version": "4.17.21", "integrity": "...", ... } }
///   },
///   "python": {
///     "packages": { "requests": { "version": "2.31.0", "integrity": "...", ... } }
///   }
/// }
pub fn generate_sidecar(lockfile: &Lockfile) -> String {
    use crate::JsonWriter;
    let mut w = JsonWriter::new();
    w.begin_object();
    w.key("version");
    w.number_u64(lockfile.header.version as u64);

    for section in &lockfile.sections {
        let eco_name = match section.ecosystem {
            EcosystemByte::Npm => "npm",
            EcosystemByte::Python => "python",
            EcosystemByte::Cargo => "cargo",
            EcosystemByte::Go => "go",
        };
        w.key(eco_name);
        w.begin_object();
        w.key("packages");
        w.begin_object();
        for pkg in &section.packages {
            w.key(&pkg.name);
            w.begin_object();
            w.key("version");
            w.string(&pkg.version);
            w.key("integrity");
            w.string(&pkg.integrity);
            if pkg.dev { w.key("dev"); w.bool_val(true); }
            if !pkg.dependencies.is_empty() {
                w.key("dependencies");
                w.begin_object();
                for (name, ver) in &pkg.dependencies {
                    w.key(name);
                    w.string(ver);
                }
                w.end_object();
            }
            w.end_object();
        }
        w.end_object(); // packages
        w.end_object(); // ecosystem
    }

    w.end_object();
    w.finish()
}

#[derive(Debug, Clone, Copy)]

    RequirementsTxt,
    PipfileLock,
    PoetryLock,
    UvLock,
}

impl MigrationSource {
    pub fn as_str(&self) -> &'static str {
        match self {
            Self::RequirementsTxt => "requirements.txt",
            Self::PipfileLock => "Pipfile.lock",
            Self::PoetryLock => "poetry.lock",
            Self::UvLock => "uv.lock",
        }
    }

/// Parse requirements.txt (pinned versions after pip freeze)
pub fn parse_requirements_txt(path: &Path) -> Result<Vec<MigratedPackage>, String> {
    let content = std::fs::read_to_string(path)
        .map_err(|e| format!("failed to read {}: {}", path.display(), e))?;

    let mut packages = Vec::new();
    for line in content.lines() {
        let line = line.trim();
        if line.is_empty() || line.starts_with('#') || line.starts_with('-') {
            continue;
        }
        // Parse: package==version or package==version --hash=sha256:xxx
        if let Some((name_part, rest)) = line.split_once("==") {
            let name = name_part.trim().to_lowercase();
            let (version, hash) = if let Some((v, h)) = rest.split_once("--hash=") {
                (v.trim().trim_end_matches('\\').trim(), Some(h.trim().to_string()))
            } else {
                (rest.trim(), None)
            };
            packages.push(MigratedPackage {
                name,
                version: version.to_string(),
                integrity: hash,
                source: MigrationSource::RequirementsTxt,
            });
        }
    }
    Ok(packages)
}

/// Auto-detect which migration source to use
pub fn detect_migration_source(project_root: &Path) -> Option<(MigrationSource, std::path::PathBuf)> {
    let candidates = [
        ("uv.lock", MigrationSource::UvLock),
        ("poetry.lock", MigrationSource::PoetryLock),
        ("Pipfile.lock", MigrationSource::PipfileLock),
        ("requirements.txt", MigrationSource::RequirementsTxt),

/// Run migration: parse source, resolve missing info, generate lockfile section
pub fn migrate(
    project_root: &Path,
    from: Option<MigrationSource>,
) -> Result<MigrationResult, String> {
    let (source, path) = if let Some(from) = from {
        let filename = match from {
            MigrationSource::RequirementsTxt => "requirements.txt",
            MigrationSource::PipfileLock => "Pipfile.lock",
            MigrationSource::PoetryLock => "poetry.lock",
            MigrationSource::UvLock => "uv.lock",
        };
        (from, project_root.join(filename))
    } else {
        detect_migration_source(project_root)
            .ok_or_else(|| "no Python lockfile found to migrate from".to_string())?
    };

    let packages = match source {
        MigrationSource::RequirementsTxt => parse_requirements_txt(&path)?,
        MigrationSource::PipfileLock => parse_pipfile_lock(&path)?,
        MigrationSource::PoetryLock => parse_poetry_lock(&path)?,
        MigrationSource::UvLock => parse_uv_lock(&path)?,
    };

    Ok(MigrationResult {
        source,
        source_path: path,
        packages_count: packages.len(),
        packages,
    })
}

    pub source: MigrationSource,
    pub source_path: std::path::PathBuf,
    pub packages_count: usize,

fn load_dotenv(project_root: &Path) -> Vec<(String, String)> {
    let dotenv_path = project_root.join(".env");
    if !dotenv_path.exists() {
        return Vec::new();
    }
    // Parse .env file: KEY=VALUE lines, skip comments
    let content = match std::fs::read_to_string(&dotenv_path) {
        Ok(c) => c,
        Err(_) => return Vec::new(),
    };
    content.lines()
        .filter(|l| !l.trim().is_empty() && !l.trim().starts_with('#'))
        .filter_map(|l| {
            let (k, v) = l.split_once('=')?;
            Some((k.trim().to_string(), v.trim().trim_matches('"').to_string()))
        })
        .collect()
