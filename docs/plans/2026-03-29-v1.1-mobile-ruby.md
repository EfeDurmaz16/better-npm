/// Swift Package Manager engine — parses Package.swift via `swift package dump-package`
pub struct SwiftEngine {
    pub swift_path: PathBuf,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SwiftManifest {
    pub name: String,
    pub platforms: Vec<SwiftPlatform>,
    pub dependencies: Vec<SwiftDependency>,
    pub targets: Vec<SwiftTarget>,
    pub swift_tools_version: String,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SwiftDependency {
    pub identity: String,
    pub url: Option<String>,         // .package(url:...)
    pub registry: Option<String>,    // .package(id:...)
    pub requirement: SwiftVersionReq,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(tag = "type")]
pub enum SwiftVersionReq {
    #[serde(rename = "exact")]
    Exact { version: String },
    #[serde(rename = "range")]
    Range { lower_bound: String, upper_bound: String },
    #[serde(rename = "branch")]
    Branch { name: String },
    #[serde(rename = "revision")]
    Revision { hash: String },
    #[serde(rename = "upToNextMajor")]
    UpToNextMajor { from: String },
    #[serde(rename = "upToNextMinor")]
    UpToNextMinor { from: String },
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SwiftPlatform {
    pub name: String,       // "iOS", "macOS", "watchOS", "tvOS", "visionOS"
    pub version: String,    // "17.0"
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SwiftTarget {
    pub name: String,
    pub target_type: String,  // "regular", "test", "executable"
    pub dependencies: Vec<String>,
}

impl SwiftEngine {
    pub fn new() -> Self {
        Self {
            swift_path: which::which("swift").unwrap_or_else(|_| PathBuf::from("swift")),
        }
    }

    /// Parse Package.swift using `swift package dump-package` (outputs JSON)
    pub fn parse_manifest(&self, project_root: &Path) -> Result<SwiftManifest, SwiftError> {
        let output = std::process::Command::new(&self.swift_path)
            .args(["package", "dump-package"])
            .current_dir(project_root)
            .output()
            .map_err(|e| SwiftError::SwiftNotFound(e.to_string()))?;

        if !output.status.success() {
            return Err(SwiftError::ParseFailed(
                String::from_utf8_lossy(&output.stderr).to_string()
            ));
        }

        let manifest: SwiftManifest = serde_json::from_slice(&output.stdout)
            .map_err(|e| SwiftError::ParseFailed(e.to_string()))?;

        Ok(manifest)
    }

    /// Detect if this is a Swift project
    pub fn detect(project_root: &Path) -> bool {
        project_root.join("Package.swift").exists()
    }
}

#[derive(Debug, thiserror::Error)]
pub enum SwiftError {
    #[error("Swift toolchain not found: {0}")]
    SwiftNotFound(String),
    #[error("Failed to parse Package.swift: {0}")]
    ParseFailed(String),
    #[error("Resolution failed: {0}")]
    ResolutionFailed(String),
    #[error("Registry error: {0}")]
    Registry(String),
    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),
}

// Added to swift.rs

/// Swift Package Registry API
/// Spec: https://github.com/swiftlang/swift-package-manager/blob/main/Documentation/PackageRegistry/Registry.md
pub struct SwiftRegistryClient {
    client: reqwest::Client,
    registries: Vec<String>,  // e.g., ["https://packages.swift.org"]
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SwiftRegistryRelease {
    pub id: String,
    pub version: String,
    pub url: String,
    pub checksum: String,
    pub metadata: SwiftReleaseMetadata,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SwiftReleaseMetadata {
    pub description: Option<String>,
    pub license: Option<String>,
    pub repository_urls: Vec<String>,
}

impl SwiftRegistryClient {
    pub fn new(registries: Vec<String>) -> Self {
        Self {
            client: reqwest::Client::builder()
                .user_agent("better/1.1")
                .build()
                .expect("failed to build HTTP client"),
            registries,
        }
    }

    /// GET /{scope}/{name} — list versions
    pub async fn list_versions(
        &self,
        scope: &str,
        name: &str,
    ) -> Result<Vec<String>, SwiftError> {
        for registry in &self.registries {
            let url = format!("{}/{}/{}", registry, scope, name);
            let resp = self.client
                .get(&url)
                .header("Accept", "application/vnd.swift.registry.v1+json")
                .send()
                .await
                .map_err(|e| SwiftError::Registry(e.to_string()))?;

            if resp.status().is_success() {
                let data: serde_json::Value = resp.json().await
                    .map_err(|e| SwiftError::Registry(e.to_string()))?;
                let versions = data["releases"]
                    .as_object()
                    .map(|obj| obj.keys().cloned().collect())
                    .unwrap_or_default();
                return Ok(versions);
            }
        }
        Err(SwiftError::Registry(format!("Package {}/{} not found", scope, name)))
    }

    /// GET /{scope}/{name}/{version} — get release
    pub async fn get_release(
        &self,
        scope: &str,
        name: &str,
        version: &str,
    ) -> Result<SwiftRegistryRelease, SwiftError> {
        for registry in &self.registries {
            let url = format!("{}/{}/{}/{}", registry, scope, name, version);
            let resp = self.client
                .get(&url)
                .header("Accept", "application/vnd.swift.registry.v1+json")
                .send()
                .await
                .map_err(|e| SwiftError::Registry(e.to_string()))?;

            if resp.status().is_success() {
                let release: SwiftRegistryRelease = resp.json().await
                    .map_err(|e| SwiftError::Registry(e.to_string()))?;
                return Ok(release);
            }
        }
        Err(SwiftError::Registry(format!("{}/{}@{} not found", scope, name, version)))
    }

    /// Download source archive and store in CAS
    pub async fn fetch_to_cas(
        &self,
        release: &SwiftRegistryRelease,
        cas_root: &Path,
    ) -> Result<PathBuf, SwiftError> {
        let resp = self.client
            .get(&release.url)
            .send()
            .await
            .map_err(|e| SwiftError::Registry(e.to_string()))?;

        let bytes = resp.bytes().await
            .map_err(|e| SwiftError::Registry(e.to_string()))?;

        // Verify checksum
        let actual_checksum = sha256_hex(&bytes);
        if actual_checksum != release.checksum {
            return Err(SwiftError::Registry(format!(
                "Checksum mismatch for {}: expected {}, got {}",
                release.id, release.checksum, actual_checksum
            )));
        }

        // Store in CAS: ~/.better/cas/swift/{checksum}
        let cas_path = cas_root.join("swift").join(&release.checksum);
        std::fs::create_dir_all(&cas_path)?;
        Ok(cas_path)
    }
}

fn sha256_hex(data: &[u8]) -> String {
    use sha2::{Sha256, Digest};
    let hash = Sha256::digest(data);
    hex::encode(hash)
}

### Subtask 90.3: Git-Based Resolution (for URL dependencies)

**Files to modify:**

Most Swift packages use git URLs, not the registry. This subtask adds `SwiftGitResolver` which:
1. Clones/fetches bare repos to `~/.better/cas/swift-git/{url-hash}/`
2. Lists tags and matches against `SwiftVersionReq` (upToNextMajor, upToNextMinor, exact, range, branch, revision)
3. Returns `ResolvedSwiftPackage` with identity, version, source, and cache path

#[derive(Debug, Clone)]
pub struct ResolvedSwiftPackage {
    pub identity: String,
    pub version: String,
    pub source: PackageSource,
    pub cache_path: PathBuf,
}

#[derive(Debug, Clone)]
pub enum PackageSource {
    Registry(String),
    Git(String),
}

fn resolve_version_from_tags(
    tags: &[String],
    req: &SwiftVersionReq,
) -> Result<String, SwiftError> {
    let semver_tags: Vec<semver::Version> = tags.iter()
        .filter_map(|t| semver::Version::parse(t.trim_start_matches('v')).ok())
        .collect();

    match req {
        SwiftVersionReq::Exact { version } => {
            let target = semver::Version::parse(version)
                .map_err(|e| SwiftError::ResolutionFailed(e.to_string()))?;
            if semver_tags.contains(&target) { Ok(version.clone()) }
            else { Err(SwiftError::ResolutionFailed(format!("Version {} not found", version))) }
        }
        SwiftVersionReq::UpToNextMajor { from } => {
            let from_ver = semver::Version::parse(from)
                .map_err(|e| SwiftError::ResolutionFailed(e.to_string()))?;
            semver_tags.iter()
                .filter(|v| v.major == from_ver.major && *v >= &from_ver)
                .max()
                .map(|v| v.to_string())
                .ok_or_else(|| SwiftError::ResolutionFailed(format!("No version >= {}", from)))
        }
        SwiftVersionReq::UpToNextMinor { from } => {
            let from_ver = semver::Version::parse(from)
                .map_err(|e| SwiftError::ResolutionFailed(e.to_string()))?;
            semver_tags.iter()
                .filter(|v| v.major == from_ver.major && v.minor == from_ver.minor && *v >= &from_ver)
                .max()
                .map(|v| v.to_string())
                .ok_or_else(|| SwiftError::ResolutionFailed(format!("No version >= {}", from)))
        }
        SwiftVersionReq::Branch { name } => Ok(name.clone()),
        SwiftVersionReq::Revision { hash } => Ok(hash.clone()),
        SwiftVersionReq::Range { lower_bound, upper_bound } => {
            let lower = semver::Version::parse(lower_bound)
                .map_err(|e| SwiftError::ResolutionFailed(e.to_string()))?;
            let upper = semver::Version::parse(upper_bound)
                .map_err(|e| SwiftError::ResolutionFailed(e.to_string()))?;
            semver_tags.iter()
                .filter(|v| *v >= &lower && *v < &upper)
                .max()
                .map(|v| v.to_string())
                .ok_or_else(|| SwiftError::ResolutionFailed(
                    format!("No version in range {}..{}", lower_bound, upper_bound)
                ))
        }
    }
}

/// Parse Podfile.lock (YAML format) into structured data.
/// Handles PODS, DEPENDENCIES, SPEC REPOS, EXTERNAL SOURCES, SPEC CHECKSUMS sections.
pub fn parse_podfile_lock(path: &Path) -> Result<PodfileLock, CocoaPodsError> {
    let content = std::fs::read_to_string(path)?;
    let yaml: serde_yaml::Value = serde_yaml::from_str(&content)
        .map_err(|e| CocoaPodsError::ParseFailed(e.to_string()))?;

    // Parse PODS section: entries are either "Name (version)" strings
    // or mappings of "Name (version)" -> [subdependency list]
    let pods = parse_pods_section(&yaml["PODS"])?;

    let spec_checksums: HashMap<String, String> = yaml["SPEC CHECKSUMS"]
        .as_mapping()
        .map(|m| m.iter()
            .filter_map(|(k, v)| Some((k.as_str()?.to_string(), v.as_str()?.to_string())))
            .collect())
        .unwrap_or_default();

    Ok(PodfileLock {
        pods,
        dependencies: vec![],
        spec_repos: HashMap::new(),
        external_sources: HashMap::new(),
        spec_checksums,
        cocoapods_version: yaml["COCOAPODS"].as_str().unwrap_or("unknown").to_string(),
    })
}

    #[error("IO error: {0}")]

const COCOAPODS_CDN: &str = "https://cdn.cocoapods.org";

pub struct CocoaPodsCdnClient {
    client: reqwest::Client,
    cdn_url: String,
}

/// CocoaPods CDN shard: md5(name)[0..1] / md5(name)[1..2]
fn pod_shard(name: &str) -> (String, String) {
    let digest = md5::compute(name.as_bytes());
    let hex = format!("{:x}", digest);
    (hex[0..1].to_string(), hex[1..2].to_string())
}

impl CocoaPodsCdnClient {
    pub fn new() -> Self {
        Self {
            client: reqwest::Client::new(),
            cdn_url: COCOAPODS_CDN.to_string(),
        }
    }

    /// Fetch podspec.json from CDN
    pub async fn fetch_podspec(
        &self,
        name: &str,
        version: &str,
    ) -> Result<PodSpec, CocoaPodsError> {
        let shard = pod_shard(name);
        let url = format!(
            "{}/Specs/{}/{}/{}/{}/{}.podspec.json",
            self.cdn_url, shard.0, shard.1, name, version, name
        );
        let resp = self.client.get(&url).send().await
            .map_err(|e| CocoaPodsError::Cdn(e.to_string()))?;
        if !resp.status().is_success() {
            return Err(CocoaPodsError::Cdn(format!("Not found: {} {}", name, version)));
        }
        resp.json().await.map_err(|e| CocoaPodsError::Cdn(e.to_string()))
    }
}

// crates/better-core/src/engines/ruby.rs

use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use std::path::{Path, PathBuf};

pub struct RubyEngine {
    pub cas_root: PathBuf,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct GemfileLock {
    pub gems: Vec<GemEntry>,
    pub platforms: Vec<String>,
    pub ruby_version: Option<String>,
    pub bundled_with: Option<String>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct GemEntry {
    pub name: String,
    pub version: String,
    pub source: GemSource,
    pub dependencies: Vec<GemDependency>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum GemSource {
    Rubygems(String),
    Git { url: String, ref_: String },
    Path(String),
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct GemDependency {
    pub name: String,
    pub requirement: String,   // "~> 2.0", ">= 1.0, < 2.0"
}

/// Parse Gemfile.lock section by section: GEM, GIT, PATH, PLATFORMS, RUBY VERSION, BUNDLED WITH, DEPENDENCIES.
/// GEM entries are indented: 4 spaces = package entry "name (version)", 6 spaces = subdependency.
pub fn parse_gemfile_lock(path: &Path) -> Result<GemfileLock, RubyError> {
    let content = std::fs::read_to_string(path)?;
    let mut gems: Vec<GemEntry> = vec![];
    let mut platforms: Vec<String> = vec![];
    let mut ruby_version = None;
    let mut bundled_with = None;
    let mut current_section = "";
    let mut current_source = GemSource::Rubygems("https://rubygems.org".into());

    for line in content.lines() {
        let trimmed = line.trim();
        match trimmed {
            "GEM" | "GIT" | "PATH" => { current_section = trimmed; continue; }
            "PLATFORMS" => { current_section = "PLATFORMS"; continue; }
            "RUBY VERSION" => { current_section = "RUBY_VERSION"; continue; }
            "BUNDLED WITH" => { current_section = "BUNDLED_WITH"; continue; }
            "DEPENDENCIES" => { current_section = "DEPENDENCIES"; continue; }
            "" => continue,
            _ => {}
        }

        match current_section {
            "GEM" | "GIT" | "PATH" => {
                if line.starts_with("  remote:") {
                    let remote = line.trim_start_matches("  remote:").trim();
                    current_source = match current_section {
                        "GEM" => GemSource::Rubygems(remote.to_string()),
                        "GIT" => GemSource::Git { url: remote.to_string(), ref_: String::new() },
                        "PATH" => GemSource::Path(remote.to_string()),
                        _ => current_source.clone(),
                    };
                } else if line.starts_with("  revision:") {
                    if let GemSource::Git { ref mut ref_, .. } = current_source {
                        *ref_ = line.trim_start_matches("  revision:").trim().to_string();
                    }
                } else if line.starts_with("    ") && !line.starts_with("      ") {
                    if let Some((name, ver_paren)) = trimmed.split_once(' ') {
                        let version = ver_paren.trim_matches(|c| c == '(' || c == ')');
                        gems.push(GemEntry {
                            name: name.to_string(),
                            version: version.to_string(),
                            source: current_source.clone(),
                            dependencies: vec![],
                        });
                    }
                } else if line.starts_with("      ") {
                    if let Some(last_gem) = gems.last_mut() {
                        if let Some((name, req)) = trimmed.split_once(' ') {
                            last_gem.dependencies.push(GemDependency {
                                name: name.to_string(),
                                requirement: req.trim_matches(|c| c == '(' || c == ')').to_string(),
                            });
                        } else {
                            last_gem.dependencies.push(GemDependency {
                                name: trimmed.to_string(),
                                requirement: ">= 0".to_string(),
                            });
                        }
                    }
                }
            }
            "PLATFORMS" => { platforms.push(trimmed.to_string()); }
            "RUBY_VERSION" => { ruby_version = Some(trimmed.trim_start_matches("ruby ").to_string()); }
            "BUNDLED_WITH" => { bundled_with = Some(trimmed.to_string()); }
            _ => {}
        }
    }

    Ok(GemfileLock { gems, platforms, ruby_version, bundled_with })
}

#[derive(Debug, thiserror::Error)]
pub enum RubyError {
    #[error("Failed to parse Gemfile.lock: {0}")]
    ParseFailed(String),
    #[error("Registry error: {0}")]
    Registry(String),
    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),
}

impl RubyGemsClient {
    pub fn new() -> Self {
        Self {
            client: reqwest::Client::builder()
                .user_agent("better/1.1")
                .build()
                .expect("failed to build HTTP client"),
        }
    }

    /// GET /api/v1/gems/{name}.json
    pub async fn get_gem_info(&self, name: &str) -> Result<GemInfo, RubyError> {
        let url = format!("{}/gems/{}.json", RUBYGEMS_API, name);
        let resp = self.client.get(&url).send().await
            .map_err(|e| RubyError::Registry(e.to_string()))?;
        if !resp.status().is_success() {
            return Err(RubyError::Registry(format!("Gem '{}' not found", name)));
        }
        resp.json().await.map_err(|e| RubyError::Registry(e.to_string()))
    }

    /// GET /api/v1/versions/{name}.json — all versions
    pub async fn get_versions(&self, name: &str) -> Result<Vec<GemVersionInfo>, RubyError> {
        let url = format!("{}/versions/{}.json", RUBYGEMS_API, name);
        let resp = self.client.get(&url).send().await
            .map_err(|e| RubyError::Registry(e.to_string()))?;
        resp.json().await.map_err(|e| RubyError::Registry(e.to_string()))
    }

    /// Download .gem and store in CAS with SHA256 verification
    pub async fn fetch_to_cas(
        &self,
        gem: &GemInfo,
        cas_root: &Path,
    ) -> Result<PathBuf, RubyError> {
        let resp = self.client.get(&gem.gem_uri).send().await
            .map_err(|e| RubyError::Registry(e.to_string()))?;
        let bytes = resp.bytes().await
            .map_err(|e| RubyError::Registry(e.to_string()))?;

        let actual_sha = sha256_hex(&bytes);
        if actual_sha != gem.sha {
            return Err(RubyError::Registry(format!(
                "SHA mismatch for {}: expected {}, got {}", gem.name, gem.sha, actual_sha
            )));
        }

        let cas_path = cas_root.join("ruby").join(&gem.sha);
        std::fs::create_dir_all(&cas_path)?;
        let gem_path = cas_path.join(format!("{}-{}.gem", gem.name, gem.version));
        std::fs::write(&gem_path, &bytes)?;
        Ok(cas_path)
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct GemVersionInfo {
    pub number: String,
    pub created_at: String,
    pub sha: String,
    pub platform: String,
}
