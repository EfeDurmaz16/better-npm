### Subtask 3.2: Create `scripts/postinstall.js` — download prebuilt Rust binary

**File to create:** `scripts/postinstall.js`

**What to do:** Write a Node.js postinstall script that:
1. Detects platform (linux/darwin) and arch (x64/arm64)
2. Downloads the matching prebuilt `better-core` binary from GitHub Releases
3. Places it in a known location (e.g., `node_modules/.better-core/better-core`)
4. Falls back gracefully if download fails (better works without Rust core for many commands)

**Code to write:**
```javascript
#!/usr/bin/env node
/**
 * postinstall.js — Download prebuilt better-core binary for the current platform.
 *
 * This runs automatically after `npm install better`.
 * If the download fails, better still works for JS-only commands.
 */

import { createWriteStream, mkdirSync, chmodSync, existsSync, readFileSync, unlinkSync } from 'node:fs';
import { join, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';
import { get } from 'node:https';
import { execFileSync } from 'node:child_process';

const __dirname = dirname(fileURLToPath(import.meta.url));
const PACKAGE_ROOT = join(__dirname, '..');

function getPlatformTarget() {
  const platform = process.platform;
  const arch = process.arch;

  const targets = {
    'darwin-arm64': 'aarch64-apple-darwin',
    'darwin-x64': 'x86_64-apple-darwin',
    'linux-x64': 'x86_64-unknown-linux-gnu',
    'linux-arm64': 'aarch64-unknown-linux-gnu',
  };

  const key = `${platform}-${arch}`;
  return targets[key] || null;
}

function getVersion() {
  const pkgPath = join(PACKAGE_ROOT, 'package.json');
  const pkg = JSON.parse(readFileSync(pkgPath, 'utf8'));
  return pkg.version;
}

function download(url, dest) {
  return new Promise((resolve, reject) => {
    const file = createWriteStream(dest);
    get(url, (response) => {
      // Handle redirects
      if (response.statusCode >= 300 && response.statusCode < 400 && response.headers.location) {
        file.close();
        return download(response.headers.location, dest).then(resolve, reject);
      }
      if (response.statusCode !== 200) {
        file.close();
        return reject(new Error(`HTTP ${response.statusCode} for ${url}`));
      }
      response.pipe(file);
      file.on('finish', () => { file.close(); resolve(); });
    }).on('error', (err) => {
      file.close();
      reject(err);
    });
  });
}

async function main() {
  const target = getPlatformTarget();
  if (!target) {
    console.log(`[better] No prebuilt binary available for ${process.platform}-${process.arch}`);
    console.log('[better] Rust-powered commands will be unavailable. JS commands still work.');
    return;
  }

  const version = getVersion();
  const binDir = join(PACKAGE_ROOT, '.native');
  const binaryPath = join(binDir, 'better-core');

  // Skip if already exists
  if (existsSync(binaryPath)) {
    console.log('[better] Native binary already present, skipping download.');
    return;
  }

  mkdirSync(binDir, { recursive: true });

  const tarUrl = `https://github.com/EfeDurmaz16/better-npm/releases/download/v${version}/better-${target}.tar.gz`;
  const tarPath = join(binDir, 'better-core.tar.gz');

  console.log(`[better] Downloading native binary for ${target}...`);

  try {
    await download(tarUrl, tarPath);

    // Extract using execFileSync (safe — no shell, no user input)
    execFileSync('tar', ['-xzf', tarPath, '-C', binDir], { stdio: 'pipe' });

    // The tarball contains `better-core` at root
    if (existsSync(binaryPath)) {
      chmodSync(binaryPath, 0o755);
      console.log('[better] Native binary installed successfully.');
    } else {
      console.log('[better] Warning: binary not found after extraction.');
    }

    // Clean up tarball
    try { unlinkSync(tarPath); } catch {}

  } catch (err) {
    console.log(`[better] Could not download native binary: ${err.message}`);
    console.log('[better] Rust-powered commands will be unavailable. JS commands still work.');
  }
}

main();
```

### Subtask 4.2: Add auto-update to release workflow

**File to modify:** `.github/workflows/release.yml`

**What to do:** Add a job at the end of the release workflow that:
1. Downloads the release assets
2. Computes SHA256 hashes
3. Updates the formula file
4. Commits and pushes to the tap repo (or this repo's `homebrew/` dir)

**Code to add to `release.yml`** (new job after `release`):
```yaml
  update-homebrew:
    needs: release
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - name: Download release assets for checksums
        run: |
          VERSION="${GITHUB_REF#refs/tags/v}"
          BASE_URL="https://github.com/EfeDurmaz16/better-npm/releases/download/v${VERSION}"

          mkdir -p /tmp/release
          for target in aarch64-apple-darwin x86_64-apple-darwin aarch64-unknown-linux-gnu x86_64-unknown-linux-gnu; do
            curl -fsSL "${BASE_URL}/better-${target}.tar.gz" -o "/tmp/release/better-${target}.tar.gz"
          done

      - name: Update formula
        run: |
          VERSION="${GITHUB_REF#refs/tags/v}"
          SHA_ARM64=$(sha256sum /tmp/release/better-aarch64-apple-darwin.tar.gz | cut -d' ' -f1)
          SHA_X64=$(sha256sum /tmp/release/better-x86_64-apple-darwin.tar.gz | cut -d' ' -f1)
          SHA_LINUX_ARM64=$(sha256sum /tmp/release/better-aarch64-unknown-linux-gnu.tar.gz | cut -d' ' -f1)
          SHA_LINUX_X64=$(sha256sum /tmp/release/better-x86_64-unknown-linux-gnu.tar.gz | cut -d' ' -f1)

          sed -i "s/version \".*\"/version \"${VERSION}\"/" homebrew/better.rb
          sed -i "s/PLACEHOLDER_SHA256_ARM64/${SHA_ARM64}/" homebrew/better.rb
          sed -i "s/PLACEHOLDER_SHA256_X64/${SHA_X64}/" homebrew/better.rb
          sed -i "s/PLACEHOLDER_SHA256_LINUX_ARM64/${SHA_LINUX_ARM64}/" homebrew/better.rb
          sed -i "s/PLACEHOLDER_SHA256_LINUX_X64/${SHA_LINUX_X64}/" homebrew/better.rb

      - name: Commit and push formula
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "github-actions[bot]@users.noreply.github.com"
          git add homebrew/better.rb
          git commit -m "brew: update formula to v${GITHUB_REF#refs/tags/v}" || true
          git push
```

```javascript
/**
 * Simple concurrency limiter — like p-limit but zero-dependency.
 * Returns a function that wraps async fns to run at most `concurrency` at a time.
 */
function createLimiter(concurrency) {
  let active = 0;
  const queue = [];

  function next() {
    if (queue.length === 0 || active >= concurrency) return;
    active++;
    const { fn, resolve, reject } = queue.shift();
    fn().then(resolve, reject).finally(() => {
      active--;
      next();
    });
  }

  return function limit(fn) {
    return new Promise((resolve, reject) => {
      queue.push({ fn, resolve, reject });
      next();
    });
  };
}
```

```javascript
/**
 * Parse pnpm-lock.yaml (v6/v9 format) — minimal YAML parser for known structure.
 * We parse: importers, packages, snapshots sections.
 */
function parsePnpmLock(yamlContent) {
  const graph = new Map();
  const lines = yamlContent.split("\n");

  // State machine for section parsing
  let section = null;        // 'importers' | 'packages' | 'snapshots' | null
  let currentPkg = null;     // current package identifier (e.g., "express@4.18.2")
  let inDeps = false;        // inside a dependencies: block
  let rootDeps = {};
  const packageDeps = {};    // pkg -> { deps: {name: version}, version }

  for (let i = 0; i < lines.length; i++) {
    const line = lines[i];

    // Top-level sections
    if (/^importers:/.test(line)) { section = "importers"; currentPkg = null; inDeps = false; continue; }
    if (/^packages:/.test(line)) { section = "packages"; currentPkg = null; inDeps = false; continue; }
    if (/^snapshots:/.test(line)) { section = "snapshots"; currentPkg = null; inDeps = false; continue; }
    if (/^\w/.test(line) && !/^(importers|packages|snapshots):/.test(line)) {
      section = null; currentPkg = null; inDeps = false; continue;
    }

    if (section === "importers") {
      // Parse root deps from importers.'.'.dependencies/devDependencies
      if (/^    dependencies:/.test(line) || /^    devDependencies:/.test(line)) {
        inDeps = true; continue;
      }
      if (inDeps && /^      (\S+):/.test(line)) {
        const match = line.match(/^      ([^:]+):/);
        if (match) {
          // The specifier is on the next indented line, or same line
          const specMatch = line.match(/specifier:\s*(.+)/);
          if (specMatch) {
            rootDeps[match[1]] = specMatch[1].trim();
          } else {
            // pnpm v9: value is on next lines as specifier/version
            rootDeps[match[1]] = "*"; // Will be refined below
          }
        }
      }
      // Refine: pnpm v9 has specifier on child line
      if (inDeps && /^        specifier:\s*(.+)/.test(line)) {
        const keys = Object.keys(rootDeps);
        const lastKey = keys[keys.length - 1];
        if (lastKey) {
          rootDeps[lastKey] = line.match(/specifier:\s*(.+)/)[1].trim();
        }
      }
      if (/^  \S/.test(line) && !/^    /.test(line)) { inDeps = false; }
    }

    if (section === "packages" || section === "snapshots") {
      // Package entry: "  express@4.18.2:" or "  express@4.18.2({...}):"
      const pkgMatch = line.match(/^  ['"]?([^'"({\s][^'"\s]*?)(?:\(.*\))?['"]?:\s*$/);
      if (pkgMatch) {
        currentPkg = pkgMatch[1]; // e.g., "express@4.18.2"
        if (!packageDeps[currentPkg]) {
          packageDeps[currentPkg] = { deps: {}, version: null };
        }
        // Extract version from identifier
        const atIdx = currentPkg.lastIndexOf("@");
        if (atIdx > 0) {
          packageDeps[currentPkg].version = currentPkg.slice(atIdx + 1);
        }
        inDeps = false;
        continue;
      }

      if (currentPkg) {
        if (/^    dependencies:/.test(line)) { inDeps = true; continue; }
        if (/^    [a-zA-Z]/.test(line) && !/^    dependencies:/.test(line)) { inDeps = false; }

        if (inDeps) {
          const depMatch = line.match(/^      ([^:]+):\s*(.+)/);
          if (depMatch) {
            packageDeps[currentPkg].deps[depMatch[1].trim()] = depMatch[2].trim();
          }
        }
      }

      // New top-level entry resets
      if (/^  \S/.test(line) && !line.startsWith("    ")) {
        // handled by pkgMatch above
      }
    }
  }

  // Build graph
  graph.set("__ROOT__", { deps: rootDeps, version: "0.0.0" });

  for (const [pkgId, data] of Object.entries(packageDeps)) {
    const atIdx = pkgId.lastIndexOf("@");
    const name = atIdx > 0 ? pkgId.slice(0, atIdx) : pkgId;
    const version = data.version || "unknown";

    if (!graph.has(name)) {
      graph.set(name, { deps: data.deps, version });
    }
  }

  return graph;
}
```

```javascript
/**
 * Parse yarn.lock — supports v1 format (indentation-based) and v2 format (YAML-ish).
 * Also reads package.json for root deps since yarn.lock doesn't store the root.
 */
function parseYarnLock(lockContent, packageJson) {
  const graph = new Map();

  // Root deps from package.json (passed in)
  const rootDeps = {
    ...(packageJson?.dependencies || {}),
    ...(packageJson?.devDependencies || {}),
    ...(packageJson?.optionalDependencies || {})
  };
  graph.set("__ROOT__", { deps: rootDeps, version: packageJson?.version || "0.0.0" });

  const lines = lockContent.split("\n");

  // Detect format
  const isV2 = lines.some(l => l.startsWith("__metadata:"));

  if (isV2) {
    return parseYarnLockV2(lines, graph);
  }

  // v1 format parsing
  let currentNames = [];
  let currentVersion = null;
  let inDeps = false;
  let currentDeps = {};

  function flushEntry() {
    if (currentNames.length === 0 || !currentVersion) return;

    for (const fullSpec of currentNames) {
      // "express@^4.18.0" -> name="express"
      const atIdx = fullSpec.lastIndexOf("@");
      if (atIdx <= 0) continue;
      const name = fullSpec.slice(0, atIdx);

      if (!graph.has(name)) {
        graph.set(name, { deps: { ...currentDeps }, version: currentVersion });
      }
    }

    currentNames = [];
    currentVersion = null;
    currentDeps = {};
    inDeps = false;
  }

  for (const line of lines) {
    // Skip comments and empty lines
    if (line.startsWith("#") || line.trim() === "") continue;

    // New entry: non-indented line with quotes or package spec
    if (!line.startsWith(" ") && !line.startsWith("\t")) {
      flushEntry();

      // Parse entry header: "express@^4.18.0, express@~4.18.0:"
      const header = line.replace(/:$/, "").trim();
      currentNames = header.split(",").map(s => s.trim().replace(/^"/, "").replace(/"$/, ""));
      continue;
    }

    const trimmed = line.trim();

    // version "4.18.2"
    if (trimmed.startsWith("version ")) {
      const match = trimmed.match(/version\s+"([^"]+)"/);
      if (match) currentVersion = match[1];
      continue;
    }

    // dependencies:
    if (trimmed === "dependencies:") {
      inDeps = true;
      continue;
    }

    // End of dependencies block (new non-dep key at same indent)
    if (inDeps && /^\s{2}\S/.test(line) && !line.startsWith("    ")) {
      inDeps = false;
    }

    // Dependency line: "    lodash "^4.17.21""
    if (inDeps) {
      const depMatch = trimmed.match(/^([^\s]+)\s+"([^"]+)"/);
      if (depMatch) {
        currentDeps[depMatch[1]] = depMatch[2];
      }
    }
  }

  flushEntry();
  return graph;
}
```

```rust
fn determine_severity(vuln: &OsvVulnerability) -> String {
    // Try CVSS score first
    for sev in &vuln.severity {
        if sev.severity_type == "CVSS_V3" || sev.severity_type == "CVSS_V2" {
            // Parse CVSS score to severity level
            if let Some(score) = extract_cvss_score(&sev.score) {
                return if score >= 9.0 { "CRITICAL" }
                    else if score >= 7.0 { "HIGH" }
                    else if score >= 4.0 { "MEDIUM" }
                    else { "LOW" }.to_string();
            }
        }
    }

    // Try database_specific.severity
    if let Some(ref db) = vuln.database_specific {
        if let Some(sev) = db.get("severity").and_then(|v| v.as_str()) {
            return sev.to_uppercase();
        }
    }

    "UNKNOWN".to_string()
}

fn extract_cvss_score(cvss_string: &str) -> Option<f64> {
    // Try to find a numeric score at the end or in the string
    // Format examples: "CVSS:3.1/AV:N/AC:L/PR:N/UI:N/S:U/C:N/I:N/A:H" (no score)
    //                  or a plain number like "7.5"
    cvss_string.parse::<f64>().ok().or_else(|| {
        // Sometimes the score is the last part after /
        cvss_string.rsplit('/').next().and_then(|s| s.parse::<f64>().ok())
    })
}
```

```bash
# bash completion for better

_better_completions() {
    local cur prev commands
    COMPREPLY=()
    cur="${COMP_WORDS[COMP_CWORD]}"
    prev="${COMP_WORDS[COMP_CWORD-1]}"

    commands="install analyze cache doctor serve benchmark lock policy workspace audit dashboard run why dedupe license outdated scripts lint test dev build"
    global_opts="--json --cache-root --log-level --config --version --help"

    case "${prev}" in
        better)
            COMPREPLY=( $(compgen -W "${commands} ${global_opts}" -- ${cur}) )
            return 0
            ;;
        cache)
            COMPREPLY=( $(compgen -W "stats gc explain" -- ${cur}) )
            return 0
            ;;
        policy)
            COMPREPLY=( $(compgen -W "check init" -- ${cur}) )
            return 0
            ;;
        workspace)
            COMPREPLY=( $(compgen -W "list info graph changed run" -- ${cur}) )
            return 0
            ;;
        scripts)
            COMPREPLY=( $(compgen -W "list allow block" -- ${cur}) )
            return 0
            ;;
        --log-level)
            COMPREPLY=( $(compgen -W "debug info warn error silent" -- ${cur}) )
            return 0
            ;;
        *)
            COMPREPLY=( $(compgen -W "${global_opts} --project-root" -- ${cur}) )
            return 0
            ;;
    esac
}

complete -F _better_completions better
```

```javascript
  if (first === "completions") {
    const shell = argv[1] || "bash";
    const { readFileSync } = await import("node:fs");
    const { join, dirname } = await import("node:path");
    const { fileURLToPath } = await import("node:url");
    const __dirname = dirname(fileURLToPath(import.meta.url));
    const completionsDir = join(__dirname, "..", "scripts");

    const files = {
      bash: "completions.sh",
      zsh: "completions.zsh",
      fish: "completions.fish"
    };

    const file = files[shell];
    if (!file) {
      printText(`Unknown shell: ${shell}. Supported: bash, zsh, fish`);
      process.exitCode = 1;
      return;
    }

    const content = readFileSync(join(completionsDir, file), "utf8");
    printText(content);
    return;
  }
```
