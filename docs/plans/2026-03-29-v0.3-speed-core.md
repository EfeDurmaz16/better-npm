### Subtask 11.1: Remove JS Fallback from analyzeFacade.js

**File:** `src/lib/analyzeFacade.js`

**Current code (lines 38-41):**

**New code:**
```js
export async function analyzeWithBestEngine(projectRoot, opts = {}) {
  const { includeGraph = true, coreMode = "auto" } = opts;

  // Try napi first (for "auto" or "napi" mode)
  if (coreMode === "napi" || coreMode === "auto") {
    const addon = tryLoadNapiAddon();
    if (addon) {
      try {
        const analysis = runBetterCoreAnalyzeNapi(projectRoot, { includeGraph });
        return { analysis, engine: "napi", corePath: null };
      } catch (err) {
        if (coreMode === "napi") throw err;
        // fall through to binary
      }
    } else if (coreMode === "napi") {
      throw new Error("napi addon not found (build via `npm run napi:build`)");
    }
  }

  if (coreMode !== "off") {
    const corePath = await findBetterCore();
    if (corePath) {
      try {
        const analysis = await runBetterCoreAnalyze(corePath, projectRoot, { includeGraph });
        return { analysis, engine: "core", corePath };
      } catch (err) {
        if (coreMode === "force") throw err;
        // fall through — but NO JS fallback anymore
      }
    } else if (coreMode === "force") {
      throw new Error("better-core not found (set BETTER_CORE_PATH or build via `npm run core:build`)");
    }
  }

  // v0.3: No more JS fallback — Rust is mandatory
  throw new Error(
    "No Rust engine available. Install better-core binary or build the NAPI addon:\n" +
    "  npm run napi:build    (recommended)\n" +
    "  npm run core:build    (binary fallback)\n" +
    "See: https://github.com/user/better-npm#building-from-source"
  );
}
```

## Task 13: better.lock Binary Format

**Goal:** Replace JSON lockfiles with a compact binary format. Faster to read (memory-mapped), faster to write, smaller on disk. Include a JSON sidecar for human readability and git diff.

### Binary Format Specification

```
better.lock binary layout:

Offset  Size     Description
─────────────────────────────────────────────
0       4        Magic bytes: "BTLK" (0x42 0x54 0x4C 0x4B)
4       2        Format version (u16 LE) — starts at 1
6       2        Flags (u16 LE) — bit 0: has integrity, bit 1: has scripts
8       4        Package count (u32 LE)
12      4        String table offset (u32 LE)
16      4        String table size (u32 LE)
20      4        Dependency table offset (u32 LE)
24      4        Dependency table size (u32 LE)
28      32       SHA-256 checksum of everything after this header
60      -        Package index entries (fixed-size records)
-       -        String table (length-prefixed UTF-8 strings)
-       -        Dependency table (varint-encoded dep lists)

Package index entry (40 bytes each):
  0     4        name_offset (u32 LE, into string table)
  4     4        version_offset (u32 LE, into string table)
  8     4        resolved_url_offset (u32 LE, into string table)
  12    4        integrity_offset (u32 LE, into string table)
  16    4        rel_path_offset (u32 LE, into string table)
  20    4        deps_offset (u32 LE, into dependency table)
  24    2        deps_count (u16 LE)
  26    2        dev_deps_count (u16 LE)
  28    2        optional_deps_count (u16 LE)
  30    2        peer_deps_count (u16 LE)
  32    4        unpacked_size (u32 LE)
  36    1        flags (has_scripts, is_optional, is_dev, is_bundled)
  37    3        reserved

String table entry:
  0     2        length (u16 LE)
  2     N        UTF-8 bytes

Dependency table entry:
  0     4        package_index (u32 LE, index into package array)
  4     4        version_range_offset (u32 LE, into string table)
```

### Subtask 13.1: Implement LockfileWriter in Rust

**File:** `crates/better-core/src/lockfile_binary.rs` (NEW)

```rust
use std::collections::HashMap;
use std::io::{Write, Cursor};
use sha2::{Sha256, Digest};

pub const MAGIC: &[u8; 4] = b"BTLK";
pub const FORMAT_VERSION: u16 = 1;
pub const HEADER_SIZE: usize = 60;
pub const PACKAGE_ENTRY_SIZE: usize = 40;

pub struct StringTable {
    data: Vec<u8>,
    index: HashMap<String, u32>,
}

impl StringTable {
    pub fn new() -> Self {
        Self { data: Vec::new(), index: HashMap::new() }
    }

    pub fn intern(&mut self, s: &str) -> u32 {
        if let Some(&offset) = self.index.get(s) {
            return offset;
        }
        let offset = self.data.len() as u32;
        let bytes = s.as_bytes();
        assert!(bytes.len() <= u16::MAX as usize, "string too long for lockfile");
        self.data.extend_from_slice(&(bytes.len() as u16).to_le_bytes());
        self.data.extend_from_slice(bytes);
        self.index.insert(s.to_string(), offset);
        offset
    }

    pub fn bytes(&self) -> &[u8] { &self.data }
    pub fn len(&self) -> usize { self.data.len() }
}

pub struct PackageEntry {
    pub name: String,
    pub version: String,
    pub resolved_url: String,
    pub integrity: String,
    pub rel_path: String,
    pub dependencies: Vec<(u32, String)>,       // (pkg_index, version_range)
    pub dev_dependencies: Vec<(u32, String)>,
    pub optional_dependencies: Vec<(u32, String)>,
    pub peer_dependencies: Vec<(u32, String)>,
    pub unpacked_size: u32,
    pub flags: u8,
}

pub struct LockfileWriter {
    packages: Vec<PackageEntry>,
    flags: u16,
}

impl LockfileWriter {
    pub fn new() -> Self {
        Self { packages: Vec::new(), flags: 0 }
    }

    pub fn add_package(&mut self, entry: PackageEntry) -> u32 {
        let idx = self.packages.len() as u32;
        self.packages.push(entry);
        idx
    }

    pub fn set_flag(&mut self, flag: u16) {
        self.flags |= flag;
    }

    pub fn write(&self) -> Vec<u8> {
        let mut strings = StringTable::new();
        let mut dep_table: Vec<u8> = Vec::new();

        // Pre-intern all strings and build dep table
        struct IndexedEntry {
            name_off: u32, version_off: u32, url_off: u32,
            integrity_off: u32, rel_path_off: u32,
            deps_off: u32, deps_count: u16, dev_deps_count: u16,
            opt_deps_count: u16, peer_deps_count: u16,
            unpacked_size: u32, flags: u8,
        }

        let mut indexed: Vec<IndexedEntry> = Vec::with_capacity(self.packages.len());

        for pkg in &self.packages {
            let name_off = strings.intern(&pkg.name);
            let version_off = strings.intern(&pkg.version);
            let url_off = strings.intern(&pkg.resolved_url);
            let integrity_off = strings.intern(&pkg.integrity);
            let rel_path_off = strings.intern(&pkg.rel_path);

            let deps_off = dep_table.len() as u32;
            let mut write_deps = |deps: &[(u32, String)]| {
                for (pkg_idx, range) in deps {
                    dep_table.extend_from_slice(&pkg_idx.to_le_bytes());
                    let range_off = strings.intern(range);
                    dep_table.extend_from_slice(&range_off.to_le_bytes());
                }
            };
            write_deps(&pkg.dependencies);
            write_deps(&pkg.dev_dependencies);
            write_deps(&pkg.optional_dependencies);
            write_deps(&pkg.peer_dependencies);

            indexed.push(IndexedEntry {
                name_off, version_off, url_off, integrity_off, rel_path_off,
                deps_off,
                deps_count: pkg.dependencies.len() as u16,
                dev_deps_count: pkg.dev_dependencies.len() as u16,
                opt_deps_count: pkg.optional_dependencies.len() as u16,
                peer_deps_count: pkg.peer_dependencies.len() as u16,
                unpacked_size: pkg.unpacked_size,
                flags: pkg.flags,
            });
        }

        let pkg_count = self.packages.len() as u32;
        let index_size = pkg_count as usize * PACKAGE_ENTRY_SIZE;
        let string_table_offset = HEADER_SIZE + index_size;
        let dep_table_offset = string_table_offset + strings.len();

        let total_size = dep_table_offset + dep_table.len();
        let mut buf = Vec::with_capacity(total_size);

        // Header (will be patched with checksum)
        buf.extend_from_slice(MAGIC);                                    // 0..4
        buf.extend_from_slice(&FORMAT_VERSION.to_le_bytes());            // 4..6
        buf.extend_from_slice(&self.flags.to_le_bytes());                // 6..8
        buf.extend_from_slice(&pkg_count.to_le_bytes());                 // 8..12
        buf.extend_from_slice(&(string_table_offset as u32).to_le_bytes()); // 12..16
        buf.extend_from_slice(&(strings.len() as u32).to_le_bytes());   // 16..20
        buf.extend_from_slice(&(dep_table_offset as u32).to_le_bytes()); // 20..24
        buf.extend_from_slice(&(dep_table.len() as u32).to_le_bytes()); // 24..28
        buf.extend_from_slice(&[0u8; 32]);                               // 28..60 checksum placeholder

        // Package index entries
        for e in &indexed {
            buf.extend_from_slice(&e.name_off.to_le_bytes());
            buf.extend_from_slice(&e.version_off.to_le_bytes());
            buf.extend_from_slice(&e.url_off.to_le_bytes());
            buf.extend_from_slice(&e.integrity_off.to_le_bytes());
            buf.extend_from_slice(&e.rel_path_off.to_le_bytes());
            buf.extend_from_slice(&e.deps_off.to_le_bytes());
            buf.extend_from_slice(&e.deps_count.to_le_bytes());
            buf.extend_from_slice(&e.dev_deps_count.to_le_bytes());
            buf.extend_from_slice(&e.opt_deps_count.to_le_bytes());
            buf.extend_from_slice(&e.peer_deps_count.to_le_bytes());
            buf.extend_from_slice(&e.unpacked_size.to_le_bytes());
            buf.push(e.flags);
            buf.extend_from_slice(&[0u8; 3]); // reserved
        }

        // String table
        buf.extend_from_slice(strings.bytes());

        // Dependency table
        buf.extend_from_slice(&dep_table);

        // Compute checksum over everything after the header
        let mut hasher = Sha256::new();
        hasher.update(&buf[HEADER_SIZE..]);
        let hash = hasher.finalize();
        buf[28..60].copy_from_slice(&hash);

        buf
    }
}
```

```rust
use memmap2::Mmap;
use std::fs::File;

pub struct LockfileReader {
    mmap: Mmap,
}

pub struct PackageRef<'a> {
    data: &'a [u8],
    string_table: &'a [u8],
    offset: usize,
}

impl<'a> PackageRef<'a> {
    pub fn name(&self) -> &str {
        let off = u32::from_le_bytes(self.data[self.offset..self.offset+4].try_into().unwrap()) as usize;
        read_string(self.string_table, off)
    }

    pub fn version(&self) -> &str {
        let off = u32::from_le_bytes(self.data[self.offset+4..self.offset+8].try_into().unwrap()) as usize;
        read_string(self.string_table, off)
    }

    pub fn resolved_url(&self) -> &str {
        let off = u32::from_le_bytes(self.data[self.offset+8..self.offset+12].try_into().unwrap()) as usize;
        read_string(self.string_table, off)
    }

    pub fn integrity(&self) -> &str {
        let off = u32::from_le_bytes(self.data[self.offset+12..self.offset+16].try_into().unwrap()) as usize;
        read_string(self.string_table, off)
    }
}

fn read_string(table: &[u8], offset: usize) -> &str {
    let len = u16::from_le_bytes(table[offset..offset+2].try_into().unwrap()) as usize;
    std::str::from_utf8(&table[offset+2..offset+2+len]).unwrap_or("")
}

impl LockfileReader {
    pub fn open(path: &Path) -> Result<Self, String> {
        let file = File::open(path).map_err(|e| format!("open lockfile: {}", e))?;
        let mmap = unsafe { Mmap::map(&file) }.map_err(|e| format!("mmap lockfile: {}", e))?;

        // Validate magic
        if mmap.len() < HEADER_SIZE || &mmap[0..4] != MAGIC {
            return Err("not a valid better.lock file".to_string());
        }

        // Validate checksum
        let stored_hash: [u8; 32] = mmap[28..60].try_into().unwrap();
        let mut hasher = Sha256::new();
        hasher.update(&mmap[HEADER_SIZE..]);
        let computed = hasher.finalize();
        if stored_hash != computed.as_slice() {
            return Err("better.lock checksum mismatch — file may be corrupted".to_string());
        }

        Ok(Self { mmap })
    }

    pub fn package_count(&self) -> u32 {
        u32::from_le_bytes(self.mmap[8..12].try_into().unwrap())
    }

    pub fn package(&self, index: u32) -> Option<PackageRef> {
        let count = self.package_count();
        if index >= count { return None; }
        let str_table_off = u32::from_le_bytes(self.mmap[12..16].try_into().unwrap()) as usize;
        let str_table_size = u32::from_le_bytes(self.mmap[16..20].try_into().unwrap()) as usize;
        let offset = HEADER_SIZE + (index as usize * PACKAGE_ENTRY_SIZE);
        Some(PackageRef {
            data: &self.mmap,
            string_table: &self.mmap[str_table_off..str_table_off + str_table_size],
            offset,
        })
    }

    pub fn iter_packages(&self) -> impl Iterator<Item = PackageRef> + '_ {
        let count = self.package_count();
        (0..count).filter_map(move |i| self.package(i))
    }
}
```

```rust
    pub updated: usize,
}

/// Three-way merge: base (common ancestor), ours (current), theirs (incoming).
pub fn merge_lockfiles(
    base_path: &Path,
    ours_path: &Path,
    theirs_path: &Path,
) -> Result<MergeResult, String> {
    let base = LockfileReader::open(base_path)?;
    let ours = LockfileReader::open(ours_path)?;
    let theirs = LockfileReader::open(theirs_path)?;

    // Index all packages by name
    let base_pkgs = index_packages(&base);
    let ours_pkgs = index_packages(&ours);
    let theirs_pkgs = index_packages(&theirs);

    let all_names: HashSet<&str> = base_pkgs.keys()
        .chain(ours_pkgs.keys())
        .chain(theirs_pkgs.keys())
        .copied()
        .collect();

    let mut writer = LockfileWriter::new();
    let mut conflicts = Vec::new();
    let mut added = 0usize;
    let mut removed = 0usize;
    let mut updated = 0usize;

    for name in all_names {
        let in_base = base_pkgs.get(name);
        let in_ours = ours_pkgs.get(name);
        let in_theirs = theirs_pkgs.get(name);

        match (in_base, in_ours, in_theirs) {
            // Both kept it unchanged — use either
            (Some(b), Some(o), Some(t)) if o.version == b.version && t.version == b.version => {
                writer.add_package(o.clone());
            },
            // Only ours changed it — use ours
            (Some(b), Some(o), Some(t)) if t.version == b.version => {
                writer.add_package(o.clone());
                updated += 1;
            },
            // Only theirs changed it — use theirs
            (Some(b), Some(o), Some(t)) if o.version == b.version => {
                writer.add_package(t.clone());
                updated += 1;
            },
            // Both changed to SAME version — use either
            (Some(_), Some(o), Some(t)) if o.version == t.version => {
                writer.add_package(o.clone());
                updated += 1;
            },
            // Both changed to DIFFERENT versions — conflict
            (Some(b), Some(o), Some(t)) => {
                conflicts.push(MergeConflict::VersionConflict {
                    package: name.to_string(),
                    base_version: b.version.clone(),
                    ours_version: o.version.clone(),
                    theirs_version: t.version.clone(),
                });
                // Default: pick theirs (incoming) — user can override
                writer.add_package(t.clone());
            },
            // Ours removed it, theirs didn't change — remove
            (Some(b), None, Some(t)) if t.version == b.version => { removed += 1; },
            // Theirs removed it, ours didn't change — remove
            (Some(b), Some(o), None) if o.version == b.version => { removed += 1; },
            // Added only by ours — keep
            (None, Some(o), None) => { writer.add_package(o.clone()); added += 1; },
            // Added only by theirs — keep
            (None, None, Some(t)) => { writer.add_package(t.clone()); added += 1; },
            // Both added same package, same version — keep
            (None, Some(o), Some(t)) if o.version == t.version => {
                writer.add_package(o.clone()); added += 1;
            },
            // Both added same package, different versions — conflict
            (None, Some(o), Some(t)) => {
                conflicts.push(MergeConflict::AddConflict {
                    package: name.to_string(),
                    ours_version: o.version.clone(),
                    theirs_version: t.version.clone(),
                });
                writer.add_package(t.clone());
                added += 1;
            },
            // All other cases
            _ => {},
        }
    }

    let merged = writer.write();
    Ok(MergeResult { merged, conflicts, added, removed, updated })
}
```

```rust
    pub references: Option<Vec<OsvReference>>,

pub fn batch_query_osv(packages: &[(String, String)]) -> Result<Vec<Vec<OsvVuln>>, String> {
    let agent = registry_agent();
    let mut all_results: Vec<Vec<OsvVuln>> = vec![Vec::new(); packages.len()];

    // Split into batches of MAX_BATCH_SIZE
    for (batch_idx, chunk) in packages.chunks(MAX_BATCH_SIZE).enumerate() {
        let queries: Vec<OsvQuery> = chunk.iter().map(|(name, version)| OsvQuery {
            package: OsvPackage {
                name: name.clone(),
                ecosystem: "npm".to_string(),
            },
            version: version.clone(),
        }).collect();

        let body = serde_json::to_string(&OsvBatchRequest { queries })
            .map_err(|e| format!("serialize: {}", e))?;

        let response = agent.post(OSV_BATCH_URL)
            .set("Content-Type", "application/json")
            .send_string(&body)
            .map_err(|e| format!("OSV batch query failed: {}", e))?;

        let resp_body = response.into_string()
            .map_err(|e| format!("read response: {}", e))?;
        let parsed: OsvBatchResponse = serde_json::from_str(&resp_body)
            .map_err(|e| format!("parse response: {}", e))?;

        for (i, result) in parsed.results.into_iter().enumerate() {
            let global_idx = batch_idx * MAX_BATCH_SIZE + i;
            if let Some(vulns) = result.vulns {
                all_results[global_idx] = vulns;
            }
        }
    }

    Ok(all_results)
}
```

## Task 18: Streaming Extraction

**Goal:** Stream tarballs directly from download -> GzDecoder -> tar::Archive -> CAS without writing intermediate temp files. Compute SHA-512 on the fly.

### Subtask 18.1: Streaming Download + Extract Pipeline

**File:** `crates/better-core/src/streaming.rs` (NEW)

```rust
use flate2::read::GzDecoder;
use sha2::{Sha512, Digest};
use std::io::{Read, Write, BufReader};
use std::path::Path;
use tar::Archive;

/// Download a tarball URL and extract directly to CAS, computing integrity on the fly.
/// Returns (sha512_hex, file_count, total_bytes).
pub fn stream_fetch_and_extract(
    url: &str,
    cas_dir: &Path,
) -> Result<StreamResult, String> {
    let response = crate::http::client()
        .get(url)
        .send()
        .map_err(|e| format!("fetch {}: {}", url, e))?;

    if !response.status().is_success() {
        return Err(format!("HTTP {} for {}", response.status(), url));
    }

    // Wrap the response body in a SHA-512 hasher
    let reader = response;
    let mut hasher = Sha512::new();
    let mut tee_reader = TeeReader::new(reader, &mut hasher);

    // GzDecoder -> tar::Archive
    let gz = GzDecoder::new(BufReader::new(&mut tee_reader));
    let mut archive = Archive::new(gz);

    let mut file_count: u64 = 0;
    let mut total_bytes: u64 = 0;

    for entry in archive.entries().map_err(|e| format!("tar entries: {}", e))? {
        let mut entry = entry.map_err(|e| format!("tar entry: {}", e))?;
        let path = entry.path().map_err(|e| format!("entry path: {}", e))?;

        // Strip the leading "package/" prefix (npm convention)
        let rel_path = path.strip_prefix("package/").unwrap_or(&path);
        let dest = cas_dir.join(rel_path);

        if entry.header().entry_type().is_dir() {
            std::fs::create_dir_all(&dest).ok();
            continue;
        }

        if let Some(parent) = dest.parent() {
            std::fs::create_dir_all(parent).ok();
        }

        let size = std::io::copy(&mut entry, &mut std::fs::File::create(&dest)
            .map_err(|e| format!("create {}: {}", dest.display(), e))?)
            .map_err(|e| format!("extract {}: {}", dest.display(), e))?;

        file_count += 1;
        total_bytes += size;
    }

    let hash = hasher.finalize();
    let sha512_hex = hex::encode(hash);

    Ok(StreamResult { sha512_hex, file_count, total_bytes })
}

pub struct StreamResult {
    pub sha512_hex: String,
    pub file_count: u64,
    pub total_bytes: u64,
}

/// A reader that tees all bytes to a writer (for hashing).
struct TeeReader<R: Read, W: Write> {
    reader: R,
    writer: W,
}

impl<R: Read, W: Write> TeeReader<R, W> {
    fn new(reader: R, writer: W) -> Self {
        Self { reader, writer }
    }
}

impl<R: Read, W: Write> Read for TeeReader<R, W> {
    fn read(&mut self, buf: &mut [u8]) -> std::io::Result<usize> {
        let n = self.reader.read(buf)?;
        if n > 0 {
            self.writer.write_all(&buf[..n])?;
        }
        Ok(n)
    }
}
```

### Subtask 20.1: File-Level SHA-256 CAS Store

**File:** `crates/better-core/src/file_cas_v2.rs` (NEW)

```rust
use sha2::{Sha256, Digest};
use std::collections::HashMap;
use std::fs;
use std::io::Read;
use std::path::{Path, PathBuf};

/// Store layout:
/// ~/.better/cas/v2/sha256/ab/cd/abcdef1234...
///                         ^first 2 hex chars
///                            ^next 2 hex chars
///                               ^full sha256 hex

pub struct FileCasV2 {
    root: PathBuf,
}

pub struct IngestReport {
    pub files_ingested: u64,
    pub files_already_present: u64,
    pub bytes_stored: u64,
    pub bytes_deduped: u64,
}

impl FileCasV2 {
    pub fn new(root: &Path) -> Self {
        Self { root: root.join("cas").join("v2").join("sha256") }
    }

    fn file_path(&self, hex: &str) -> PathBuf {
        let a = &hex[..2];
        let b = &hex[2..4];
        self.root.join(a).join(b).join(hex)
    }

    /// Hash a file and return its SHA-256 hex digest.
    pub fn hash_file(path: &Path) -> Result<String, String> {
        let mut file = fs::File::open(path).map_err(|e| format!("open: {}", e))?;
        let mut hasher = Sha256::new();
        let mut buf = [0u8; 8192];
        loop {
            let n = file.read(&mut buf).map_err(|e| format!("read: {}", e))?;
            if n == 0 { break; }
            hasher.update(&buf[..n]);
        }
        Ok(hex::encode(hasher.finalize()))
    }

    /// Ingest a file into the CAS. Returns the hash.
    /// If the file already exists in CAS, this is a no-op.
    pub fn ingest(&self, source: &Path) -> Result<(String, bool), String> {
        let hash = Self::hash_file(source)?;
        let dest = self.file_path(&hash);

        if dest.exists() {
            return Ok((hash, false)); // already present
        }

        // Ensure directory exists
        if let Some(parent) = dest.parent() {
            fs::create_dir_all(parent).map_err(|e| format!("mkdir: {}", e))?;
        }

        // Copy to CAS (atomic: write to temp, then rename)
        let tmp = dest.with_extension("tmp");
        fs::copy(source, &tmp).map_err(|e| format!("copy: {}", e))?;
        // Make read-only
        let meta = fs::metadata(&tmp).map_err(|e| format!("metadata: {}", e))?;
        let mut perms = meta.permissions();
        perms.set_readonly(true);
        fs::set_permissions(&tmp, perms).ok();
        fs::rename(&tmp, &dest).map_err(|e| format!("rename: {}", e))?;

        Ok((hash, true))
    }

    /// Materialize a file from CAS to a destination using hardlink.
    pub fn materialize(&self, hash: &str, dest: &Path) -> Result<(), String> {
        let source = self.file_path(hash);
        if !source.exists() {
            return Err(format!("CAS entry not found: {}", hash));
        }

        if let Some(parent) = dest.parent() {
            fs::create_dir_all(parent).ok();
        }

        // Try hardlink first
        match fs::hard_link(&source, dest) {
            Ok(()) => Ok(()),
            Err(_) => {
                // Fallback to copy
                fs::copy(&source, dest).map_err(|e| format!("copy: {}", e))?;
                Ok(())
            }
        }
    }

    /// Ingest all files in a directory tree, return manifest.
    pub fn ingest_tree(&self, root: &Path) -> Result<(FileManifest, IngestReport), String> {
        let mut manifest = FileManifest::new();
        let mut report = IngestReport {
            files_ingested: 0,
            files_already_present: 0,
            bytes_stored: 0,
            bytes_deduped: 0,
        };

        walk_tree(root, root, &mut |rel_path, abs_path, metadata| {
            if metadata.is_dir() { return Ok(()); }

            let size = metadata.len();
            let (hash, is_new) = self.ingest(abs_path)?;

            manifest.entries.push(FileManifestEntry {
                path: rel_path.to_string_lossy().to_string(),
                hash: hash.clone(),
                size,
                mode: metadata.permissions().mode(),
            });

            if is_new {
                report.files_ingested += 1;
                report.bytes_stored += size;
            } else {
                report.files_already_present += 1;
                report.bytes_deduped += size;
            }
            Ok(())
        })?;

        Ok((manifest, report))
    }
}

#[derive(Serialize, Deserialize)]
pub struct FileManifest {
    pub entries: Vec<FileManifestEntry>,
}

#[derive(Serialize, Deserialize)]
pub struct FileManifestEntry {
    pub path: String,
    pub hash: String,
    pub size: u64,
    pub mode: u32,
}

impl FileManifest {
    pub fn new() -> Self { Self { entries: Vec::new() } }
}
```

```rust
impl FileCasV2 {
    /// Materialize an entire package from its manifest using hardlinks.
    /// Files with identical content across packages share the same CAS entry,
    /// so they are hardlinked to the same inode.
    pub fn materialize_from_manifest(
        &self,
        manifest: &FileManifest,
        dest: &Path,
    ) -> Result<MaterializeReport, String> {
        let mut linked = 0u64;
        let mut copied = 0u64;
        let mut dirs = 0u64;

        for entry in &manifest.entries {
            let file_dest = dest.join(&entry.path);
            if let Some(parent) = file_dest.parent() {
                if !parent.exists() {
                    fs::create_dir_all(parent).ok();
                    dirs += 1;
                }
            }

            let source = self.file_path(&entry.hash);
            match fs::hard_link(&source, &file_dest) {
                Ok(()) => { linked += 1; },
                Err(_) => {
                    fs::copy(&source, &file_dest)
                        .map_err(|e| format!("copy {}: {}", entry.path, e))?;
                    copied += 1;
                }
            }
        }

        Ok(MaterializeReport {
            stats: MaterializeStats {
                files: linked + copied,
                files_linked: linked,
                files_copied: copied,
                directories: dirs,
                ..Default::default()
            },
            phases: PhaseDurations::default(),
        })
    }

    /// Report CAS stats: total files, total bytes, dedup ratio.
    pub fn stats(&self) -> Result<CasV2Stats, String> {
        let mut total_files = 0u64;
        let mut total_bytes = 0u64;
        let mut total_links = 0u64;

        walk_cas_store(&self.root, &mut |path, metadata| {
            total_files += 1;
            total_bytes += metadata.len();
            total_links += metadata.nlink() as u64;
        })?;

        Ok(CasV2Stats {
            total_files,
            total_bytes,
            total_links,
            dedup_ratio: if total_links > 0 {
                total_files as f64 / total_links as f64
            } else {
                1.0
            },
        })
    }
}
```

```rust
use std::path::{Path, PathBuf};

/// The trait every package manager engine must implement.
pub trait PackageEngine: Send + Sync {
    /// Unique engine identifier: "npm", "yarn", "pnpm", "bun"
    fn name(&self) -> &str;

    /// Detect if this engine should be used for the given project root.
    /// Returns a confidence score 0-100.
    fn detect(&self, project_root: &Path) -> u32;

    /// Parse the lockfile and return resolved packages.
    fn resolve(&self, project_root: &Path) -> Result<ResolveResult, String>;

    /// Fetch packages to the CAS.
    fn fetch(&self, packages: &[ResolvedPackage], cache: &Path) -> Result<FetchResult, String>;

    /// Materialize packages from CAS to node_modules.
    fn materialize(
        &self,
        packages: &[ResolvedPackage],
        cache: &Path,
        project_root: &Path,
        strategy: LinkStrategy,
    ) -> Result<MaterializeReport, String>;

    /// Run a full install (resolve + fetch + materialize + scripts).
    fn install(&self, project_root: &Path, opts: InstallOpts) -> Result<InstallReport, String>;

    /// Check for outdated packages.
    fn outdated(&self, project_root: &Path) -> Result<OutdatedReport, String>;

    /// Audit packages for vulnerabilities.
    fn audit(&self, project_root: &Path, min_severity: &str) -> Result<AuditReport, String>;

    /// Scan licenses.
    fn licenses(&self, project_root: &Path, allow: &[String], deny: &[String]) -> Result<LicenseReport, String>;

    /// Check for duplicate packages.
    fn dedupe(&self, project_root: &Path) -> Result<DedupeReport, String>;

    /// Trace why a package is installed.
    fn why(&self, project_root: &Path, package: &str) -> Result<WhyReport, String>;

    /// Generate SBOM.
    fn sbom(&self, project_root: &Path, format: &str) -> Result<SbomReport, String>;

    /// Return the lockfile name this engine uses.
    fn lockfile_name(&self) -> &str;

    /// Return the expected node_modules layout.
    fn layout(&self) -> NodeModulesLayout;
}

#[derive(Debug, Clone, Copy)]
pub enum NodeModulesLayout {
    /// Flat: node_modules/package-name (npm default)
    Flat,
    /// Nested: node_modules/.pnpm/package@version/node_modules/package-name
    Pnpm,
    /// PnP: .pnp.cjs (yarn berry)
    PlugNPlay,
}

pub struct InstallOpts {
    pub link_strategy: LinkStrategy,
    pub jobs: usize,
    pub scripts: bool,
    pub frozen: bool,
    pub dedup: bool,
    pub lazy: bool,
}
```

```rust
use std::sync::OnceLock;

/// Registry of available engines.
static ENGINES: OnceLock<Vec<Box<dyn PackageEngine>>> = OnceLock::new();

pub fn registered_engines() -> &'static [Box<dyn PackageEngine>] {
    ENGINES.get_or_init(|| {
        vec![
            Box::new(crate::engines::npm::NpmEngine),
            // Future: Box::new(crate::engines::yarn::YarnEngine),
            // Future: Box::new(crate::engines::pnpm::PnpmEngine),
            // Future: Box::new(crate::engines::bun::BunEngine),
        ]
    })
}

/// Auto-detect the best engine for a project root.
/// Returns the engine with the highest confidence score.
pub fn detect_engine(project_root: &Path) -> Result<&'static dyn PackageEngine, String> {
    let engines = registered_engines();

    let mut best: Option<(&dyn PackageEngine, u32)> = None;

    for engine in engines.iter() {
        let score = engine.detect(project_root);
        if score == 0 { continue; }
        match &best {
            None => { best = Some((engine.as_ref(), score)); },
            Some((_, best_score)) if score > *best_score => {
                best = Some((engine.as_ref(), score));
            },
            _ => {},
        }
    }

    best.map(|(e, _)| e)
        .ok_or_else(|| format!("No package engine detected for {}", project_root.display()))
}

/// Get a specific engine by name.
pub fn engine_by_name(name: &str) -> Option<&'static dyn PackageEngine> {
    registered_engines().iter()
        .find(|e| e.name() == name)
        .map(|e| e.as_ref())
}
```
