```rust
use serde::{Deserialize, Serialize};

#[derive(Debug, Clone)]
pub struct AiEngine {
    pub provider: AiProvider,
    pub model: String,
    pub context_budget: usize,   // Max tokens for context
}

#[derive(Debug, Clone)]
pub enum AiProvider {
    Claude { api_key: String },
    OpenAI { api_key: String },
    Local { endpoint: String },
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct AiRequest {
    pub intent: String,           // User's natural language request
    pub project_context: ProjectContext,
    pub available_tools: Vec<AiTool>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ProjectContext {
    pub ecosystems: Vec<String>,
    pub existing_deps: Vec<DepSummary>,
    pub framework: Option<String>,
    pub osp_services: Vec<String>,
    pub file_structure: Vec<String>,  // Top-level dirs/files
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct DepSummary {
    pub name: String,
    pub version: String,
    pub ecosystem: String,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct AiTool {
    pub name: String,             // "install", "provision", "search", "context"
    pub description: String,
    pub parameters: serde_json::Value,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct AiPlan {
    pub explanation: String,
    pub steps: Vec<AiStep>,
    pub estimated_time: String,
    pub requires_confirmation: bool,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct AiStep {
    pub step_number: u32,
    pub description: String,
    pub tool_call: ToolCall,
    pub expected_result: String,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ToolCall {
    pub tool: String,
    pub arguments: serde_json::Value,
}

impl AiEngine {
    pub fn new(provider: AiProvider) -> Self {
        let model = match &provider {
            AiProvider::Claude { .. } => "claude-sonnet-4-20250514".to_string(),
            AiProvider::OpenAI { .. } => "gpt-4o".to_string(),
            AiProvider::Local { .. } => "local".to_string(),
        };
        Self {
            provider,
            model,
            context_budget: 50_000,
        }
    }

    /// Convert natural language intent to an executable plan
    pub async fn plan(
        &self,
        request: &AiRequest,
    ) -> Result<AiPlan, AiError> {
        let system_prompt = build_system_prompt(&request.project_context, &request.available_tools);
        let user_prompt = &request.intent;

        let response = self.call_llm(&system_prompt, user_prompt).await?;
        let plan: AiPlan = serde_json::from_str(&response)
            .map_err(|e| AiError::PlanParseFailed(e.to_string()))?;

        Ok(plan)
    }

    /// Execute a plan step by step with confirmation gates
    pub async fn execute_plan(
        &self,
        plan: &AiPlan,
        confirm_fn: impl Fn(&AiStep) -> bool,
    ) -> Result<Vec<StepResult>, AiError> {
        let mut results = vec![];

        for step in &plan.steps {
            if plan.requires_confirmation && !confirm_fn(step) {
                results.push(StepResult {
                    step_number: step.step_number,
                    status: StepStatus::Skipped,
                    output: None,
                });
                continue;
            }

            let result = execute_tool_call(&step.tool_call).await;
            results.push(StepResult {
                step_number: step.step_number,
                status: if result.is_ok() { StepStatus::Success } else { StepStatus::Failed },
                output: result.ok(),
            });
        }

        Ok(results)
    }

    async fn call_llm(
        &self,
        system: &str,
        user: &str,
    ) -> Result<String, AiError> {
        match &self.provider {
            AiProvider::Claude { api_key } => {
                let client = reqwest::Client::new();
                let resp = client
                    .post("https://api.anthropic.com/v1/messages")
                    .header("x-api-key", api_key)
                    .header("anthropic-version", "2023-06-01")
                    .header("Content-Type", "application/json")
                    .json(&serde_json::json!({
                        "model": self.model,
                        "max_tokens": 4096,
                        "system": system,
                        "messages": [{"role": "user", "content": user}]
                    }))
                    .send()
                    .await
                    .map_err(|e| AiError::ProviderError(e.to_string()))?;

                let body: serde_json::Value = resp.json().await
                    .map_err(|e| AiError::ProviderError(e.to_string()))?;

                body["content"][0]["text"]
                    .as_str()
                    .map(String::from)
                    .ok_or_else(|| AiError::ProviderError("Empty response".into()))
            }
            _ => Err(AiError::ProviderError("Not implemented".into())),
        }
    }
}

#[derive(Debug, Clone, Serialize)]
pub struct StepResult {
    pub step_number: u32,
    pub status: StepStatus,
    pub output: Option<serde_json::Value>,
}

#[derive(Debug, Clone, Serialize)]
pub enum StepStatus { Success, Failed, Skipped }

#[derive(Debug, thiserror::Error)]

    #[error("AI provider error: {0}")]
    ProviderError(String),
    #[error("Failed to parse AI plan: {0}")]
    PlanParseFailed(String),
    #[error("Tool execution failed: {0}")]
    ToolFailed(String),
}
```

```rust
    pub severity: SuggestionSeverity,
    pub title: String,
    pub description: String,
    pub packages: Vec<String>,
    pub estimated_savings: Option<SavingsEstimate>,

#[derive(Debug, Clone, Serialize)]
pub enum SuggestionCategory {
    Consolidate,     // Multiple packages for same purpose
    Downsize,        // Lighter alternative available
    Remove,          // Unused or unnecessary
    Upgrade,         // Outdated with benefits
    Security,        // Security concern
    License,         // License incompatibility
}

#[derive(Debug, Clone, Serialize)]
pub enum SuggestionSeverity { High, Medium, Low }

#[derive(Debug, Clone, Serialize)]

    pub fewer_packages: u32,
}

#[derive(Debug, Clone, Serialize)]
pub struct OverallHealth {
    pub score: u8,           // 0-100
    pub dep_count_rating: String,  // "lean", "moderate", "heavy"
    pub freshness_rating: String,  // "current", "somewhat outdated", "stale"
    pub security_rating: String,   // "clean", "minor issues", "critical issues"
}

/// AI-powered dependency review
pub async fn review_dependencies(
    project_root: &std::path::Path,
    ai_engine: &AiEngine,
) -> Result<DependencyReview, ReviewError> {
    // 1. Collect all dependency data
    let deps = collect_all_deps(project_root)?;
    let usage_data = scan_all_imports(project_root)?;
    let reputation_data = collect_reputation_scores(&deps).await?;
    let size_data = collect_size_data(&deps)?;

    // 2. Rule-based analysis (no AI needed for obvious cases)
    let mut suggestions = vec![];

    // Detect unused deps
    for dep in &deps {
        if !usage_data.is_used(&dep.name) && !dep.is_dev_dep {
            suggestions.push(ReviewSuggestion {
                category: SuggestionCategory::Remove,
                severity: SuggestionSeverity::Medium,
                title: format!("Unused: {}", dep.name),
                description: format!("{} is not imported anywhere in the source code", dep.name),
                packages: vec![dep.name.clone()],
                estimated_savings: Some(SavingsEstimate {
                    size_reduction_kb: dep.size_bytes / 1024,
                    fewer_packages: 1,
                }),
                action: format!("better remove {}", dep.name),
            });
        }
    }

    // Detect duplicate purpose packages (multiple HTTP clients, date libs, etc.)
    let duplicates = detect_duplicate_purpose(&deps);
    for dup in duplicates {
        suggestions.push(ReviewSuggestion {
            category: SuggestionCategory::Consolidate,
            severity: SuggestionSeverity::Medium,
            title: format!("Multiple {} libraries", dup.category),
            description: format!(
                "Using {} packages for {}: {}. Consider consolidating.",
                dup.packages.len(), dup.category, dup.packages.join(", ")
            ),
            packages: dup.packages.clone(),
            estimated_savings: None,
            action: format!("better ai \"consolidate {} libraries\"", dup.category),
        });
    }

    // 3. AI-powered analysis for nuanced suggestions
    let ai_suggestions = ai_engine.plan(&AiRequest {
        intent: format!(
            "Review these dependencies and suggest improvements:\n{}",
            deps.iter()
                .map(|d| format!("  {}@{} ({}KB, score: {})",
                    d.name, d.version, d.size_bytes / 1024,
                    reputation_data.get(&d.name).map(|r| r.score).unwrap_or(0)))
                .collect::<Vec<_>>()
                .join("\n")
        ),
        project_context: build_project_context(project_root)?,
        available_tools: vec![],
    }).await?;

    // 4. Merge rule-based and AI suggestions
    // Parse AI response into additional suggestions...

    let total_size = deps.iter().map(|d| d.size_bytes).sum::<u64>();

    Ok(DependencyReview {
        project: project_root.display().to_string(),
        total_deps: deps.len(),
        total_size_mb: total_size as f64 / 1_048_576.0,
        suggestions,
        overall_health: compute_overall_health(&deps, &reputation_data),
    })
}
```

The migration engine:
1. Fetches changelog + migration guide (via `better context`)
2. Sends to LLM with project source code context
3. LLM generates a migration plan with file-level diffs
4. Applies changes one file at a time, testing after each
5. Commits incrementally or rolls back on failure

```rust
#[derive(Debug, Clone, Serialize)]
pub struct MigrationPlan {
    pub from_version: String,
    pub to_version: String,
    pub breaking_changes: Vec<String>,
    pub file_changes: Vec<FileMigration>,
    pub test_commands: Vec<String>,
}

#[derive(Debug, Clone, Serialize)]
pub struct FileMigration {
    pub file_path: String,
    pub changes: Vec<CodeChange>,
    pub reason: String,
}

#[derive(Debug, Clone, Serialize)]
pub struct CodeChange {
    pub line_start: usize,
    pub line_end: usize,
    pub old_code: String,
    pub new_code: String,
    pub explanation: String,
}

pub async fn plan_migration(
    project_root: &std::path::Path,
    package: &str,
    target_version: &str,
    ai_engine: &AiEngine,
) -> Result<MigrationPlan, MigrateError> {
    // 1. Get current version and changelog
    let current = get_installed_version(project_root, package)?;
    let changelog = fetch_changelog(package).await?;
    let migration_guide = fetch_migration_guide(package, &current, target_version).await.ok();

    // 2. Scan project for usage of this package
    let usage = scan_package_usage(project_root, package)?;

    // 3. Build context for LLM
    let context = format!(
        "Migrate {} from {} to {}.\n\nChangelog:\n{}\n\nMigration guide:\n{}\n\nFiles using this package:\n{}",
        package, current, target_version,
        changelog,
        migration_guide.as_deref().unwrap_or("Not available"),
        usage.files.iter()
            .map(|f| format!("--- {} ---\n{}", f.path, f.relevant_code))
            .collect::<Vec<_>>()
            .join("\n\n")
    );

    // 4. Ask LLM for migration plan
    let plan = ai_engine.plan(&AiRequest {
        intent: context,
        project_context: build_project_context(project_root)?,
        available_tools: vec![],
    }).await?;

    // Parse LLM response into MigrationPlan
    parse_migration_plan(&plan)
}
```

```rust
/// Execute a pipeline with gates and rollback support
pub async fn execute_pipeline(
    pipeline: &Pipeline,
    project_root: &std::path::Path,
    gate_handler: impl Fn(&Gate) -> bool,
) -> Result<PipelineResult, PipelineError> {
    let start = std::time::Instant::now();
    let mut stage_results = vec![];
    let mut checkpoints: Vec<String> = vec![];

    for stage in &pipeline.stages {
        // Check dependencies are completed
        let deps_met = stage.depends_on.iter().all(|dep| {
            stage_results.iter().any(|r: &StageResult| {
                r.name == *dep && matches!(r.status, StageStatus::Completed)
            })
        });
        if !deps_met {
            stage_results.push(StageResult {
                name: stage.name.clone(),
                status: StageStatus::Skipped,
                duration_ms: 0,
                output: serde_json::Value::Null,
            });
            continue;
        }

        // Create checkpoint for rollback
        if pipeline.rollback_on_failure {
            checkpoints.push(create_checkpoint(project_root)?);
        }

        // Execute stage with retry
        let mut last_err = None;
        let mut result = None;
        for attempt in 0..=stage.retries {
            match execute_stage(&stage.action, project_root).await {
                Ok(output) => {
                    result = Some(output);
                    break;
                }
                Err(e) => {
                    last_err = Some(e);
                    if attempt < stage.retries {
                        tokio::time::sleep(std::time::Duration::from_secs(2)).await;
                    }
                }
            }
        }

        let stage_result = match result {
            Some(output) => StageResult {
                name: stage.name.clone(),
                status: StageStatus::Completed,
                duration_ms: 0,
                output,
            },
            None => {
                if pipeline.rollback_on_failure {
                    for checkpoint in checkpoints.iter().rev() {
                        rollback_to(project_root, checkpoint)?;
                    }
                }
                return Ok(PipelineResult {
                    pipeline: pipeline.name.clone(),
                    stages: stage_results,
                    total_duration_ms: start.elapsed().as_millis() as u64,
                    success: false,
                    halted_at: Some(stage.name.clone()),
                });
            }
        };

        stage_results.push(stage_result);

        // Check gates after this stage
        for gate in &pipeline.gates {
            if gate.after_stage == stage.name {
                let approved = match &gate.gate_type {
                    GateType::HumanApproval { .. } => gate_handler(gate),
                    GateType::TestPass => run_tests(project_root).await.unwrap_or(false),
                    GateType::AuditClean => run_audit_check(project_root).await.unwrap_or(false),
                    GateType::PolicyPass => run_policy_check(project_root).await.unwrap_or(false),
                    GateType::CostUnder { max_monthly_usd } => {
                        check_cost_under(project_root, *max_monthly_usd).await.unwrap_or(false)
                    }
                };

                if !approved {
                    return Ok(PipelineResult {
                        pipeline: pipeline.name.clone(),
                        stages: stage_results,
                        total_duration_ms: start.elapsed().as_millis() as u64,
                        success: false,
                        halted_at: Some(format!("gate after {}", stage.name)),
                    });
                }
            }
        }
    }

    Ok(PipelineResult {
        pipeline: pipeline.name.clone(),
        stages: stage_results,
        total_duration_ms: start.elapsed().as_millis() as u64,
        success: true,
        halted_at: None,
    })
}

#[derive(Debug, thiserror::Error)]
pub enum PipelineError {
    #[error("Pipeline failed: {0}")]
    Failed(String),
    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),
}
```

```rust
use serde::Serialize;

#[derive(Debug, Clone, Serialize)]
pub struct HealAction {
    pub vulnerability: String,
    pub package: String,
    pub action_type: HealActionType,
    pub status: HealStatus,
    pub pr_url: Option<String>,
}

#[derive(Debug, Clone, Serialize)]
pub enum HealActionType {
    Upgrade { from: String, to: String },
    MigrateAlternative { from: String, to_package: String },
    Workaround { description: String },
}

#[derive(Debug, Clone, Serialize)]
pub enum HealStatus {
    Fixed,
    PrCreated,
    TestsFailed,
    NoPatchAvailable,
    ManualRequired(String),
}

/// Watch for new CVEs and auto-heal
pub async fn watch_and_heal(
    project_root: &std::path::Path,
    check_interval_secs: u64,
    auto_pr: bool,
) -> Result<(), HealError> {
    loop {
        // 1. Run audit
        let audit = run_audit(project_root).await?;

        if audit.new_since_last_check > 0 {
            println!("New vulnerabilities detected: {}", audit.new_since_last_check);

            for vuln in &audit.new_vulnerabilities {
                let action = heal_vulnerability(project_root, vuln, auto_pr).await?;
                println!("  {} {}: {:?}", vuln.package, vuln.id, action.status);
            }
        }

        // 2. Sleep until next check
        tokio::time::sleep(std::time::Duration::from_secs(check_interval_secs)).await;
    }
}

/// Attempt to heal a single vulnerability
async fn heal_vulnerability(
    project_root: &std::path::Path,
    vuln: &Vulnerability,
    auto_pr: bool,
) -> Result<HealAction, HealError> {
    // Strategy 1: Direct upgrade to patched version
    if let Some(ref fixed_in) = vuln.fixed_in {
        // Create branch
        let branch = format!("better-heal/{}-{}", vuln.package, vuln.id);
        create_branch(project_root, &branch)?;

        // Upgrade
        upgrade_package(project_root, &vuln.package, fixed_in).await?;

        // Test
        let tests_ok = run_tests(project_root).await.unwrap_or(false);

        if tests_ok {
            // Commit
            commit_changes(project_root, &format!(
                "fix(deps): upgrade {} to {} (fixes {})",
                vuln.package, fixed_in, vuln.id
            ))?;

            if auto_pr {
                let pr_url = create_pull_request(project_root, &branch, &format!(
                    "fix(deps): {} {} -> {} ({})",
                    vuln.package, vuln.installed_version, fixed_in, vuln.id
                )).await?;

                return Ok(HealAction {
                    vulnerability: vuln.id.clone(),
                    package: vuln.package.clone(),
                    action_type: HealActionType::Upgrade {
                        from: vuln.installed_version.clone(),
                        to: fixed_in.clone(),
                    },
                    status: HealStatus::PrCreated,
                    pr_url: Some(pr_url),
                });
            }

            return Ok(HealAction {
                vulnerability: vuln.id.clone(),
                package: vuln.package.clone(),
                action_type: HealActionType::Upgrade {
                    from: vuln.installed_version.clone(),
                    to: fixed_in.clone(),
                },
                status: HealStatus::Fixed,
                pr_url: None,
            });
        } else {
            // Rollback
            checkout_main(project_root)?;
        }
    }

    // Strategy 2: Find alternative package
    let alternatives = find_alternatives(&vuln.package).await?;
    if let Some(alt) = alternatives.first() {
        return Ok(HealAction {
            vulnerability: vuln.id.clone(),
            package: vuln.package.clone(),
            action_type: HealActionType::MigrateAlternative {
                from: vuln.package.clone(),
                to_package: alt.package.clone(),
            },
            status: HealStatus::ManualRequired(format!(
                "Consider migrating to {} (reputation: {})",
                alt.package, alt.reputation_score
            )),
            pr_url: None,
        });
    }

    Ok(HealAction {
        vulnerability: vuln.id.clone(),
        package: vuln.package.clone(),
        action_type: HealActionType::Workaround {
            description: "No patch or alternative available".into(),
        },
        status: HealStatus::NoPatchAvailable,
        pr_url: None,
    })
}

#[derive(Debug, thiserror::Error)]
pub enum HealError {
    #[error("Heal failed: {0}")]
    Failed(String),
}
```

```rust
use serde::Serialize;
use std::collections::HashMap;

#[derive(Debug, Clone, Serialize)]
pub struct OrgInsights {
    pub projects_analyzed: usize,
    pub total_unique_deps: usize,
    pub total_dep_instances: usize,
    pub version_inconsistencies: Vec<VersionInconsistency>,
    pub shared_configs: Vec<SharedConfigSuggestion>,
    pub consolidation_opportunities: Vec<ConsolidationOpportunity>,
    pub security_summary: SecuritySummary,
    pub standardization_score: u8,  // 0-100
}

#[derive(Debug, Clone, Serialize)]
pub struct VersionInconsistency {
    pub package: String,
    pub versions: HashMap<String, Vec<String>>,  // version -> [project names]
    pub recommended: String,
    pub effort_to_align: String,
}

#[derive(Debug, Clone, Serialize)]
pub struct SharedConfigSuggestion {
    pub config_type: String,       // "tsconfig", "eslint", "prettier", "better.toml"
    pub projects_using: Vec<String>,
    pub divergences: Vec<String>,
    pub suggestion: String,
}

#[derive(Debug, Clone, Serialize)]
pub struct ConsolidationOpportunity {
    pub category: String,          // "http-client", "date-library", etc.
    pub current: HashMap<String, Vec<String>>,  // package -> [projects]
    pub recommended: String,
    pub reason: String,
}

#[derive(Debug, Clone, Serialize)]
pub struct SecuritySummary {
    pub total_vulns: usize,
    pub critical: usize,
    pub projects_with_vulns: usize,
    pub most_vulnerable_project: String,
    pub shared_vuln_packages: Vec<String>,  // Vulns present in multiple projects
}

/// Analyze dependencies across all projects in a directory
pub async fn analyze_org(
    root_dir: &std::path::Path,
) -> Result<OrgInsights, InsightsError> {
    // 1. Discover all projects
    let projects = discover_projects(root_dir)?;

    // 2. Collect dependency data from each
    let mut all_deps: HashMap<String, HashMap<String, Vec<String>>> = HashMap::new(); // pkg -> version -> projects
    let mut total_instances = 0;

    for project in &projects {
        let deps = collect_project_deps(&project.path)?;
        for dep in &deps {
            total_instances += 1;
            all_deps
                .entry(dep.name.clone())
                .or_default()
                .entry(dep.version.clone())
                .or_default()
                .push(project.name.clone());
        }
    }

    // 3. Find version inconsistencies
    let inconsistencies: Vec<VersionInconsistency> = all_deps.iter()
        .filter(|(_, versions)| versions.len() > 1)
        .map(|(pkg, versions)| {
            let recommended = versions.keys()
                .filter_map(|v| semver::Version::parse(v).ok())
                .max()
                .map(|v| v.to_string())
                .unwrap_or_default();

            VersionInconsistency {
                package: pkg.clone(),
                versions: versions.clone(),
                recommended,
                effort_to_align: if versions.len() <= 2 { "Low" } else { "Medium" }.into(),
            }
        })
        .collect();

    // 4. Detect consolidation opportunities
    let consolidations = detect_cross_project_duplicates(&all_deps);

    // 5. Security summary
    let security = cross_project_security_summary(&projects).await?;

    // 6. Standardization score
    let score = compute_standardization_score(
        &inconsistencies,
        &consolidations,
        projects.len(),
    );

    Ok(OrgInsights {
        projects_analyzed: projects.len(),
        total_unique_deps: all_deps.len(),
        total_dep_instances: total_instances,
        version_inconsistencies: inconsistencies,
        shared_configs: vec![],
        consolidation_opportunities: consolidations,
        security_summary: security,
        standardization_score: score,
    })
}

#[derive(Debug, thiserror::Error)]
pub enum InsightsError {
    #[error("Analysis failed: {0}")]
    Failed(String),
    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),
}
```
