```rust
// crates/better-core/src/registry/cid.rs

use serde::{Deserialize, Serialize};

/// Content Identifier — uniquely identifies a package version by its content hash.
/// Format: better:{hash_algo}:{hex_digest}
/// Example: better:sha256:a1b2c3d4e5f6...
#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub struct ContentId {
    pub algorithm: HashAlgorithm,
    pub digest: String,
}

#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub enum HashAlgorithm {
    Sha256,
    Blake3,
}

impl ContentId {
    pub fn from_bytes(data: &[u8], algo: HashAlgorithm) -> Self {
        let digest = match algo {
            HashAlgorithm::Sha256 => {
                use sha2::{Sha256, Digest};
                hex::encode(Sha256::digest(data))
            }
            HashAlgorithm::Blake3 => {
                hex::encode(blake3::hash(data).as_bytes())
            }
        };
        Self { algorithm: algo, digest }
    }

    pub fn to_string(&self) -> String {
        let algo_str = match self.algorithm {
            HashAlgorithm::Sha256 => "sha256",
            HashAlgorithm::Blake3 => "blake3",
        };
        format!("better:{}:{}", algo_str, self.digest)
    }

    pub fn parse(s: &str) -> Result<Self, CidError> {
        let parts: Vec<&str> = s.split(':').collect();
        if parts.len() != 3 || parts[0] != "better" {
            return Err(CidError::InvalidFormat(s.to_string()));
        }
        let algorithm = match parts[1] {
            "sha256" => HashAlgorithm::Sha256,
            "blake3" => HashAlgorithm::Blake3,
            _ => return Err(CidError::UnsupportedAlgorithm(parts[1].to_string())),
        };
        Ok(Self { algorithm, digest: parts[2].to_string() })
    }

    /// Convert to IPFS-compatible CID v1 (dag-pb, sha2-256)
    pub fn to_ipfs_cid(&self) -> Option<String> {
        match self.algorithm {
            HashAlgorithm::Sha256 => {
                // CID v1: multibase(base32) + version(1) + codec(dag-pb) + multihash(sha2-256)
                // Simplified: use cid crate in actual implementation
                Some(format!("bafybeig{}", &self.digest[..44]))
            }
            _ => None,
        }
    }
}

#[derive(Debug, thiserror::Error)]
pub enum CidError {
    #[error("Invalid CID format: {0}")]
    InvalidFormat(String),
    #[error("Unsupported hash algorithm: {0}")]
    UnsupportedAlgorithm(String),
}
```

```rust
// crates/better-core/src/registry/dht.rs

use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use std::net::SocketAddr;

/// Kademlia-inspired DHT for package discovery.
/// Peers announce which CIDs they can serve.
/// Lookup: hash(package_name) -> closest peers -> resolve CID -> fetch content.
pub struct DhtNode {
    pub node_id: [u8; 32],
    pub listen_addr: SocketAddr,
    routing_table: RoutingTable,
    store: HashMap<ContentId, Vec<PeerInfo>>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PeerInfo {
    pub node_id: [u8; 32],
    pub addr: SocketAddr,
    pub last_seen: u64,
    pub capabilities: PeerCapabilities,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PeerCapabilities {
    pub can_serve: bool,       // Has content, can upload
    pub bandwidth_mbps: u32,
    pub storage_gb: u32,
    pub ecosystems: Vec<String>,
}

struct RoutingTable {
    buckets: Vec<Vec<PeerInfo>>,  // 256 k-buckets
    k: usize,                     // bucket size (default: 20)
}

impl DhtNode {
    pub fn new(listen_addr: SocketAddr) -> Self {
        let node_id = generate_node_id();
        Self {
            node_id,
            listen_addr,
            routing_table: RoutingTable {
                buckets: vec![vec![]; 256],
                k: 20,
            },
            store: HashMap::new(),
        }
    }

    /// Announce that this node has a specific CID available
    pub async fn announce(&mut self, cid: &ContentId) -> Result<(), DhtError> {
        let key = dht_key_from_cid(cid);
        let closest = self.routing_table.find_closest(&key, self.routing_table.k);

        for peer in closest {
            // Send STORE RPC to closest peers
            send_store_rpc(&peer, cid, &self.peer_info()).await?;
        }

        Ok(())
    }

    /// Find peers that have a specific CID
    pub async fn find_providers(
        &self,
        cid: &ContentId,
    ) -> Result<Vec<PeerInfo>, DhtError> {
        let key = dht_key_from_cid(cid);

        // Check local store first
        if let Some(providers) = self.store.get(cid) {
            if !providers.is_empty() {
                return Ok(providers.clone());
            }
        }

        // Iterative lookup
        let mut queried = std::collections::HashSet::new();
        let mut closest = self.routing_table.find_closest(&key, self.routing_table.k);
        let mut providers = vec![];

        for _ in 0..3 {  // max 3 rounds
            let mut new_peers = vec![];
            for peer in &closest {
                if queried.contains(&peer.node_id) { continue; }
                queried.insert(peer.node_id);

                match send_find_providers_rpc(peer, cid).await {
                    Ok(result) => {
                        providers.extend(result.providers);
                        new_peers.extend(result.closer_peers);
                    }
                    Err(_) => continue,
                }
            }
            if providers.len() >= 5 || new_peers.is_empty() { break; }
            closest = new_peers;
        }

        Ok(providers)
    }

    fn peer_info(&self) -> PeerInfo {
        PeerInfo {
            node_id: self.node_id,
            addr: self.listen_addr,
            last_seen: current_timestamp(),
            capabilities: PeerCapabilities {
                can_serve: true,
                bandwidth_mbps: 100,
                storage_gb: 50,
                ecosystems: vec!["npm".into(), "python".into()],
            },
        }
    }
}

fn dht_key_from_cid(cid: &ContentId) -> [u8; 32] {
    use sha2::{Sha256, Digest};
    let hash = Sha256::digest(cid.to_string().as_bytes());
    let mut key = [0u8; 32];
    key.copy_from_slice(&hash);
    key
}

fn generate_node_id() -> [u8; 32] {
    let mut id = [0u8; 32];
    getrandom::fill(&mut id).expect("failed to generate random bytes");
    id
}

#[derive(Debug, thiserror::Error)]
pub enum DhtError {
    #[error("Network error: {0}")]
    Network(String),
    #[error("Timeout")]
    Timeout,
    #[error("No providers found")]
    NoProviders,
}
```

```rust
use serde::{Deserialize, Serialize};
use super::cid::ContentId;

/// Lightweight package index — maps package names to their CIDs.
/// Stored as a Merkle tree for verifiable lookups.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PackageIndex {
    pub root_hash: String,
    pub entries: Vec<PackageIndexEntry>,
    pub updated_at: u64,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PackageIndexEntry {
    pub name: String,
    pub ecosystem: String,
    pub versions: Vec<VersionEntry>,
    pub latest: String,
    pub publisher: PublisherInfo,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct VersionEntry {
    pub version: String,
    pub cid: ContentId,
    pub published_at: u64,
    pub size_bytes: u64,
    pub signature: Option<String>,  // Ed25519 signature from publisher
    pub yanked: bool,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PublisherInfo {
    pub name: String,
    pub public_key: String,       // Ed25519 public key
    pub verified: bool,
}

impl PackageIndex {
    /// Resolve a package name + version constraint to a CID
    pub fn resolve(
        &self,
        name: &str,
        constraint: &str,
    ) -> Option<&VersionEntry> {
        let entry = self.entries.iter().find(|e| e.name == name)?;
        let matching: Vec<&VersionEntry> = entry.versions.iter()
            .filter(|v| !v.yanked)
            .filter(|v| version_matches(&v.version, constraint))
            .collect();
        matching.last()  // Latest matching version
    }

    /// Verify Merkle root of all entries
    pub fn verify_integrity(&self) -> bool {
        let computed_root = compute_merkle_root(&self.entries);
        computed_root == self.root_hash
    }
}
```

```rust
// crates/better-core/src/registry/federation.rs

use serde::{Deserialize, Serialize};
use std::path::PathBuf;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct RegistryConfig {
    pub registries: Vec<RegistryEntry>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct RegistryEntry {
    pub name: String,
    pub url: String,
    pub registry_type: RegistryType,
    pub priority: u32,               // Lower = checked first
    pub scopes: Vec<String>,         // e.g., ["@mycompany/*"] — empty = all
    pub auth_token: Option<String>,
    pub enabled: bool,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum RegistryType {
    Npm,           // npm-compatible registry
    Packagist,     // Composer/Packagist
    PyPI,          // Python Package Index
    NuGet,         // NuGet.org
    CratesIo,      // Rust crates
    BetterNative,  // better decentralized registry
    OspRegistry,   // OSP service registry
}

pub struct FederatedResolver {
    registries: Vec<RegistryEntry>,
    cas_root: PathBuf,
}

impl FederatedResolver {
    pub fn new(config: RegistryConfig, cas_root: PathBuf) -> Self {
        let mut registries = config.registries;
        registries.sort_by_key(|r| r.priority);
        Self { registries, cas_root }
    }

    /// Resolve a package by trying registries in priority order.
    /// Scope-matched registries are tried first.
    pub async fn resolve(
        &self,
        name: &str,
        version_constraint: &str,
        ecosystem: &str,
    ) -> Result<FederatedResult, FederationError> {
        let mut tried = vec![];

        for registry in &self.registries {
            if !registry.enabled { continue; }

            // Check scope match
            if !registry.scopes.is_empty() {
                let matches_scope = registry.scopes.iter().any(|scope| {
                    if scope.ends_with("/*") {
                        name.starts_with(&scope[..scope.len() - 2])
                    } else {
                        name == scope
                    }
                });
                if !matches_scope { continue; }
            }

            tried.push(registry.name.clone());

            match self.try_resolve(registry, name, version_constraint, ecosystem).await {
                Ok(result) => return Ok(result),
                Err(e) => {
                    // Log and try next registry
                    tracing::debug!("Registry {} failed for {}: {}", registry.name, name, e);
                    continue;
                }
            }
        }

        Err(FederationError::NotFound {
            package: name.to_string(),
            tried,
        })
    }

    async fn try_resolve(
        &self,
        registry: &RegistryEntry,
        name: &str,
        version_constraint: &str,
        ecosystem: &str,
    ) -> Result<FederatedResult, FederationError> {
        match registry.registry_type {
            RegistryType::Npm => resolve_npm(registry, name, version_constraint).await,
            RegistryType::BetterNative => resolve_decentralized(registry, name, version_constraint).await,
            RegistryType::OspRegistry => resolve_osp(registry, name, version_constraint).await,
            _ => Err(FederationError::UnsupportedType),
        }
    }
}

#[derive(Debug, Clone, Serialize)]
pub struct FederatedResult {
    pub package: String,
    pub version: String,
    pub registry: String,
    pub cid: Option<ContentId>,
    pub download_url: String,
    pub integrity: String,
}

#[derive(Debug, thiserror::Error)]
pub enum FederationError {
    #[error("Package {package} not found in registries: {tried:?}")]
    NotFound { package: String, tried: Vec<String> },
    #[error("Unsupported registry type")]
    UnsupportedType,
    #[error("Auth failed: {0}")]
    AuthFailed(String),
    #[error("Network error: {0}")]
    Network(String),
}
```

```rust
// crates/better-core/src/signing/keys.rs

use ed25519_dalek::{SigningKey, VerifyingKey, Signer, Verifier, Signature};
use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PublisherKeyPair {
    pub key_id: String,            // fingerprint of public key
    pub public_key: String,         // base64-encoded
    pub created_at: u64,
    pub expires_at: Option<u64>,
    pub publisher_name: String,
}

pub struct KeyStore {
    keys_path: std::path::PathBuf,  // ~/.better/keys/
}

impl KeyStore {
    pub fn new() -> Self {
        let keys_path = dirs::config_dir()
            .unwrap_or_default()
            .join("better")
            .join("keys");
        Self { keys_path }
    }

    /// Generate a new Ed25519 signing key pair
    pub fn generate_keypair(
        &self,
        publisher_name: &str,
    ) -> Result<PublisherKeyPair, SigningError> {
        use rand::rngs::OsRng;
        let signing_key = SigningKey::generate(&mut OsRng);
        let verifying_key = signing_key.verifying_key();

        let public_bytes = verifying_key.to_bytes();
        let public_b64 = base64::engine::general_purpose::STANDARD.encode(&public_bytes);
        let key_id = hex::encode(&sha256_hash(&public_bytes)[..8]);

        // Store private key encrypted
        let private_path = self.keys_path.join(format!("{}.key", key_id));
        std::fs::create_dir_all(&self.keys_path)?;
        let encrypted = encrypt_private_key(&signing_key.to_bytes())?;
        std::fs::write(&private_path, encrypted)?;

        // Store public key
        let pub_path = self.keys_path.join(format!("{}.pub", key_id));
        std::fs::write(&pub_path, &public_b64)?;

        Ok(PublisherKeyPair {
            key_id,
            public_key: public_b64,
            created_at: current_timestamp(),
            expires_at: None,
            publisher_name: publisher_name.to_string(),
        })
    }

    /// Load signing key by key_id
    pub fn load_signing_key(&self, key_id: &str) -> Result<SigningKey, SigningError> {
        let private_path = self.keys_path.join(format!("{}.key", key_id));
        let encrypted = std::fs::read(&private_path)?;
        let key_bytes = decrypt_private_key(&encrypted)?;
        SigningKey::from_bytes(&key_bytes.try_into().map_err(|_| {
            SigningError::InvalidKey("Wrong key length".into())
        })?)
        .map_err(|e| SigningError::InvalidKey(e.to_string()))
    }

    /// Import a trusted public key
    pub fn import_public_key(
        &self,
        key_id: &str,
        public_key_b64: &str,
        publisher_name: &str,
        trust_level: TrustLevel,
    ) -> Result<(), SigningError> {
        let trust_path = self.keys_path.join("trusted");
        std::fs::create_dir_all(&trust_path)?;

        let entry = TrustedKey {
            key_id: key_id.to_string(),
            public_key: public_key_b64.to_string(),
            publisher_name: publisher_name.to_string(),
            trust_level,
            imported_at: current_timestamp(),
        };

        let file = trust_path.join(format!("{}.json", key_id));
        std::fs::write(&file, serde_json::to_string_pretty(&entry)?)?;
        Ok(())
    }
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TrustedKey {
    pub key_id: String,
    pub public_key: String,
    pub publisher_name: String,
    pub trust_level: TrustLevel,
    pub imported_at: u64,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub enum TrustLevel {
    Full,       // Directly verified (met publisher, verified fingerprint)
    Marginal,   // Vouched by another trusted key
    Unknown,    // Not yet verified
}

#[derive(Debug, thiserror::Error)]
pub enum SigningError {
    #[error("Invalid key: {0}")]
    InvalidKey(String),
    #[error("Signature verification failed")]
    VerificationFailed,
    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),
}
```

```rust
// crates/better-core/src/signing/sign.rs

use ed25519_dalek::{Signer, Verifier, Signature};

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PackageSignature {
    pub key_id: String,
    pub signature: String,    // base64-encoded Ed25519 signature
    pub signed_at: u64,
    pub payload_hash: String, // SHA256 of the signed content
}

/// Sign a package tarball before publishing
pub fn sign_package(
    package_data: &[u8],
    signing_key: &SigningKey,
    key_id: &str,
) -> Result<PackageSignature, SigningError> {
    let payload_hash = sha256_hex(package_data);

    // Sign the hash, not the full content
    let signature = signing_key.sign(payload_hash.as_bytes());
    let sig_b64 = base64::engine::general_purpose::STANDARD.encode(signature.to_bytes());

    Ok(PackageSignature {
        key_id: key_id.to_string(),
        signature: sig_b64,
        signed_at: current_timestamp(),
        payload_hash,
    })
}

/// Verify a package signature against a known public key
pub fn verify_signature(
    package_data: &[u8],
    sig: &PackageSignature,
    verifying_key: &VerifyingKey,
) -> Result<bool, SigningError> {
    let payload_hash = sha256_hex(package_data);

    if payload_hash != sig.payload_hash {
        return Ok(false);
    }

    let sig_bytes = base64::engine::general_purpose::STANDARD.decode(&sig.signature)
        .map_err(|_| SigningError::VerificationFailed)?;
    let signature = Signature::from_bytes(&sig_bytes.try_into()
        .map_err(|_| SigningError::VerificationFailed)?);

    Ok(verifying_key.verify(payload_hash.as_bytes(), &signature).is_ok())
}
```

```rust
    // 2. Build from source in clean environment
    let build_output = build_from_source(&source, package_name).await?;
    let local_hash = sha256_hex(&build_output);

    // 3. Hash published artifact
    let published_hash = sha256_hex(registry_artifact);

    // 4. Compare
    Ok(VerifyResult {
        package: package_name.to_string(),
        version: version.to_string(),
        source_match: true,
        build_match: local_hash == published_hash,
        published_hash,
        local_hash,
        env_diff: vec![],
    })
}

#[derive(Debug, thiserror::Error)]
pub enum ReproducibleError {
    #[error("Source not available: {0}")]
    SourceNotAvailable(String),
    #[error("Build failed: {0}")]
    BuildFailed(String),
    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),
}
```

```rust
// crates/better-core/src/registry/unified.rs

use serde::Serialize;

#[derive(Debug, Clone, Serialize)]
pub struct UnifiedSearchResult {
    pub results: Vec<SearchEntry>,
    pub sources: Vec<SearchSource>,
    pub total: usize,
}

#[derive(Debug, Clone, Serialize)]
pub struct SearchEntry {
    pub name: String,
    pub entry_type: EntryType,
    pub description: String,
    pub ecosystem: String,
    pub score: f64,
    pub source: String,
}

#[derive(Debug, Clone, Serialize)]
pub enum EntryType {
    Package,       // npm, PyPI, etc.
    OspService,    // OSP provider
    Plugin,        // better plugin
}

#[derive(Debug, Clone, Serialize)]
pub struct SearchSource {
    pub name: String,
    pub result_count: usize,
    pub latency_ms: u64,
}

/// Search across all registries: packages + OSP services + plugins
pub async fn unified_search(
    query: &str,
    ecosystems: &[String],
) -> Result<UnifiedSearchResult, SearchError> {
    let mut results = vec![];
    let mut sources = vec![];

    // 1. Search package registries in parallel
    let (npm_results, pypi_results, osp_results) = tokio::join!(
        search_npm(query),
        search_pypi(query),
        search_osp_registry(query),
    );

    if let Ok(npm) = npm_results {
        sources.push(SearchSource { name: "npm".into(), result_count: npm.len(), latency_ms: 0 });
        results.extend(npm);
    }
    if let Ok(pypi) = pypi_results {
        sources.push(SearchSource { name: "pypi".into(), result_count: pypi.len(), latency_ms: 0 });
        results.extend(pypi);
    }
    if let Ok(osp) = osp_results {
        sources.push(SearchSource { name: "osp".into(), result_count: osp.len(), latency_ms: 0 });
        results.extend(osp);
    }

    // 2. Rank by relevance
    results.sort_by(|a, b| b.score.partial_cmp(&a.score).unwrap_or(std::cmp::Ordering::Equal));

    let total = results.len();
    Ok(UnifiedSearchResult { results, sources, total })
}
```

## Task 107: Content-Addressed Publishing

**Goal:** CAS hash as canonical package ID. Same content = same hash regardless of registry.

### Subtask 107.1: CAS-Based Publish Flow

**Files to modify:**
- `crates/better-core/src/registry/cid.rs` (extend)
- `src/commands/publish.ts` (modify publish flow)

The publish flow:
1. `better publish` computes Blake3 hash of the package tarball
2. Generates `ContentId`: `better:blake3:{digest}`
3. Signs the CID with publisher's Ed25519 key
4. Uploads to configured registries (npm, decentralized, private)
5. Each registry stores the mapping: `{name}@{version}` -> `ContentId`
6. On install, content is verified by CID regardless of which registry served it

**Code:**
