```rust
/// Raw signals collected for reputation scoring
#[derive(Debug, Clone, Serialize)]
pub struct PackageSignals {
    pub package: String,
    pub ecosystem: String,
    pub version: String,

    // Maintainer signals
    pub maintainer_count: u32,
    pub maintainer_active_days_90: u32,     // How many days at least one maintainer was active in last 90d
    pub maintainer_has_2fa: Option<bool>,
    pub bus_factor: u32,                     // Number of people who can publish

    // Activity signals
    pub days_since_last_publish: u32,
    pub publish_frequency_30d: f64,          // Publishes per 30 days over last year
    pub commit_frequency_30d: f64,
    pub open_issues: u32,
    pub issue_response_time_median_hours: f64,
    pub open_prs: u32,

    // Download signals
    pub weekly_downloads: u64,
    pub download_trend: DownloadTrend,       // Growing, Stable, Declining, Anomalous
    pub download_anomaly_score: f64,         // 0.0 = normal, 1.0 = extremely anomalous

    // Security signals
    pub total_cves: u32,
    pub unpatched_cves: u32,
    pub days_since_last_cve: Option<u32>,
    pub has_provenance: bool,
    pub has_signature: bool,

    // Quality signals
    pub has_types: bool,
    pub has_tests: bool,
    pub has_ci: bool,
    pub license: Option<String>,
    pub readme_length: u32,

    // Typosquat signals
    pub typosquat_distance_min: u32,         // Min Levenshtein distance to popular packages
    pub typosquat_suspect: bool,
    pub age_days: u32,                       // Days since first publish
}

#[derive(Debug, Clone, Serialize)]
pub enum DownloadTrend {
    Growing,
    Stable,
    Declining,
    Anomalous,
}

/// Collect signals from multiple sources
pub async fn collect_signals(
    package: &str,
    ecosystem: &str,
) -> Result<PackageSignals, SignalError> {
    let (registry_data, github_data, osv_data, download_data) = tokio::join!(
        fetch_registry_metadata(package, ecosystem),
        fetch_github_metadata(package, ecosystem),
        fetch_osv_data(package, ecosystem),
        fetch_download_stats(package, ecosystem),
    );

    let registry = registry_data?;
    let github = github_data.ok();
    let osv = osv_data?;
    let downloads = download_data?;

    Ok(PackageSignals {
        package: package.to_string(),
        ecosystem: ecosystem.to_string(),
        version: registry.latest_version,
        maintainer_count: registry.maintainers.len() as u32,
        maintainer_active_days_90: github.as_ref()
            .map(|g| g.active_days_90).unwrap_or(0),
        maintainer_has_2fa: registry.has_2fa,
        bus_factor: registry.maintainers.len().min(3) as u32,
        days_since_last_publish: registry.days_since_last_publish,
        publish_frequency_30d: registry.publish_frequency,
        commit_frequency_30d: github.as_ref()
            .map(|g| g.commit_frequency).unwrap_or(0.0),
        open_issues: github.as_ref().map(|g| g.open_issues).unwrap_or(0),
        issue_response_time_median_hours: github.as_ref()
            .map(|g| g.issue_response_median_hours).unwrap_or(999.0),
        open_prs: github.as_ref().map(|g| g.open_prs).unwrap_or(0),
        weekly_downloads: downloads.weekly,
        download_trend: downloads.trend,
        download_anomaly_score: downloads.anomaly_score,
        total_cves: osv.total,
        unpatched_cves: osv.unpatched,
        days_since_last_cve: osv.days_since_last,
        has_provenance: registry.has_provenance,
        has_signature: registry.has_signature,
        has_types: registry.has_types,
        has_tests: github.as_ref().map(|g| g.has_tests).unwrap_or(false),
        has_ci: github.as_ref().map(|g| g.has_ci).unwrap_or(false),
        license: registry.license,
        readme_length: registry.readme_length,
        typosquat_distance_min: compute_typosquat_distance(package),
        typosquat_suspect: compute_typosquat_distance(package) <= 2 && registry.age_days < 30,
        age_days: registry.age_days,
    })
}
```

```rust
// crates/better-core/src/intelligence/scoring.rs

use super::signals::PackageSignals;
use serde::Serialize;

#[derive(Debug, Clone, Serialize)]
pub struct ReputationScore {
    pub package: String,
    pub score: u8,              // 0-100
    pub grade: Grade,
    pub breakdown: ScoreBreakdown,
    pub flags: Vec<ReputationFlag>,
    pub computed_at: u64,
}

#[derive(Debug, Clone, Serialize)]
pub enum Grade { A, B, C, D, F }

#[derive(Debug, Clone, Serialize)]
pub struct ScoreBreakdown {
    pub maintainer_health: f64,    // 0-25
    pub security_posture: f64,     // 0-25
    pub activity_vitality: f64,    // 0-25
    pub community_trust: f64,      // 0-25
}

#[derive(Debug, Clone, Serialize)]
pub struct ReputationFlag {
    pub flag_type: FlagType,
    pub severity: FlagSeverity,
    pub message: String,
}

#[derive(Debug, Clone, Serialize)]
pub enum FlagType {
    SingleMaintainer,
    DownloadAnomaly,
    Typosquat,
    NoProvenance,
    UnpatchedVulns,
    Unmaintained,
    NewPackage,
}

#[derive(Debug, Clone, Serialize)]
pub enum FlagSeverity { Critical, High, Medium, Low, Info }

/// Compute reputation score from signals
pub fn compute_score(signals: &PackageSignals) -> ReputationScore {
    let mut flags = vec![];

    // === Maintainer Health (0-25) ===
    let maintainer_score = {
        let mut s = 0.0;
        // Bus factor
        s += match signals.bus_factor {
            0 => 0.0,
            1 => { flags.push(flag(FlagType::SingleMaintainer, FlagSeverity::Medium,
                "Single maintainer — bus factor risk")); 5.0 },
            2 => 10.0,
            _ => 12.0,
        };
        // 2FA
        if signals.maintainer_has_2fa == Some(true) { s += 5.0; }
        // Active maintainers
        s += (signals.maintainer_active_days_90 as f64 / 90.0 * 8.0).min(8.0);
        s.min(25.0)
    };

    // === Security Posture (0-25) ===
    let security_score = {
        let mut s = 25.0;
        // Unpatched CVEs
        if signals.unpatched_cves > 0 {
            flags.push(flag(FlagType::UnpatchedVulns, FlagSeverity::High,
                &format!("{} unpatched vulnerabilities", signals.unpatched_cves)));
            s -= (signals.unpatched_cves as f64 * 5.0).min(15.0);
        }
        // Provenance
        if !signals.has_provenance {
            flags.push(flag(FlagType::NoProvenance, FlagSeverity::Low,
                "No build provenance attestation"));
            s -= 3.0;
        }
        if !signals.has_signature { s -= 2.0; }
        // Typosquat
        if signals.typosquat_suspect {
            flags.push(flag(FlagType::Typosquat, FlagSeverity::Critical,
                "Possible typosquat — similar name to popular package, recently published"));
            s -= 15.0;
        }
        s.max(0.0).min(25.0)
    };

    // === Activity Vitality (0-25) ===
    let activity_score = {
        let mut s = 0.0;
        // Recent publish activity
        s += match signals.days_since_last_publish {
            0..=30 => 8.0,
            31..=90 => 6.0,
            91..=180 => 4.0,
            181..=365 => 2.0,
            _ => {
                flags.push(flag(FlagType::Unmaintained, FlagSeverity::Medium,
                    &format!("No publish in {} days", signals.days_since_last_publish)));
                0.0
            }
        };
        // Commit frequency
        s += (signals.commit_frequency_30d * 2.0).min(8.0);
        // Issue response time
        s += if signals.issue_response_time_median_hours < 24.0 { 5.0 }
            else if signals.issue_response_time_median_hours < 72.0 { 3.0 }
            else if signals.issue_response_time_median_hours < 168.0 { 1.0 }
            else { 0.0 };
        // Has CI + tests
        if signals.has_ci { s += 2.0; }
        if signals.has_tests { s += 2.0; }
        s.min(25.0)
    };

    // === Community Trust (0-25) ===
    let community_score = {
        let mut s = 0.0;
        // Download volume (log scale)
        s += match signals.weekly_downloads {
            0..=100 => 2.0,
            101..=1000 => 5.0,
            1001..=10000 => 8.0,
            10001..=100000 => 12.0,
            _ => 15.0,
        };
        // Download anomaly
        if signals.download_anomaly_score > 0.7 {
            flags.push(flag(FlagType::DownloadAnomaly, FlagSeverity::High,
                "Unusual download pattern detected"));
            s -= 5.0;
        }
        // Package age
        if signals.age_days < 7 {
            flags.push(flag(FlagType::NewPackage, FlagSeverity::Info,
                "Published less than 7 days ago"));
        }
        s += match signals.age_days {
            0..=7 => 0.0,
            8..=30 => 2.0,
            31..=365 => 5.0,
            _ => 8.0,
        };
        // Has types + license + readme
        if signals.has_types { s += 1.0; }
        if signals.license.is_some() { s += 0.5; }
        if signals.readme_length > 500 { s += 0.5; }
        s.max(0.0).min(25.0)
    };

    let total = (maintainer_score + security_score + activity_score + community_score) as u8;
    let grade = match total {
        80..=100 => Grade::A,
        60..=79 => Grade::B,
        40..=59 => Grade::C,
        20..=39 => Grade::D,
        _ => Grade::F,
    };

    ReputationScore {
        package: signals.package.clone(),
        score: total,
        grade,
        breakdown: ScoreBreakdown {
            maintainer_health: maintainer_score,
            security_posture: security_score,
            activity_vitality: activity_score,
            community_trust: community_score,
        },
        flags,
        computed_at: current_timestamp(),
    }
}

fn flag(flag_type: FlagType, severity: FlagSeverity, message: &str) -> ReputationFlag {
    ReputationFlag { flag_type, severity, message: message.to_string() }
}
```

```rust
use serde::Serialize;

#[derive(Debug, Clone, Serialize)]
pub struct ChangelogAnalysis {
    pub package: String,
    pub from_version: String,
    pub to_version: String,
    pub breaking_changes: Vec<BreakingChange>,
    pub notable_changes: Vec<String>,
    pub migration_steps: Vec<MigrationStep>,
    pub risk_level: RiskLevel,
}

#[derive(Debug, Clone, Serialize)]
pub struct BreakingChange {
    pub description: String,
    pub affected_api: String,
    pub migration_hint: String,
}

#[derive(Debug, Clone, Serialize)]
pub struct MigrationStep {
    pub step: u32,
    pub description: String,
    pub file_pattern: String,     // Glob pattern for affected files
    pub search_pattern: String,   // Regex to find usage
    pub replacement: String,      // Suggested replacement
    pub automated: bool,          // Can be auto-applied
}

#[derive(Debug, Clone, Serialize)]
pub enum RiskLevel { Low, Medium, High, Critical }

/// Fetch and analyze changelog between two versions
pub async fn analyze_changelog(
    package: &str,
    from: &str,
    to: &str,
    ecosystem: &str,
) -> Result<ChangelogAnalysis, ChangelogError> {
    // 1. Fetch CHANGELOG.md / HISTORY.md / GitHub releases
    let changelog_text = fetch_changelog(package, ecosystem).await?;

    // 2. Extract entries between from and to versions
    let relevant_entries = extract_version_range(&changelog_text, from, to)?;

    // 3. Parse breaking changes (look for "BREAKING", "Breaking Change", semver major bumps)
    let breaking_changes = parse_breaking_changes(&relevant_entries);

    // 4. Generate migration steps from breaking changes
    let migration_steps = generate_migration_steps(&breaking_changes, package);

    // 5. Assess risk
    let risk_level = if breaking_changes.is_empty() { RiskLevel::Low }
        else if breaking_changes.len() <= 2 { RiskLevel::Medium }
        else if breaking_changes.len() <= 5 { RiskLevel::High }
        else { RiskLevel::Critical };

    Ok(ChangelogAnalysis {
        package: package.to_string(),
        from_version: from.to_string(),
        to_version: to.to_string(),
        breaking_changes,
        notable_changes: relevant_entries,
        migration_steps,
        risk_level,
    })
}

#[derive(Debug, thiserror::Error)]
pub enum ChangelogError {
    #[error("Changelog not found for {0}")]
    NotFound(String),
    #[error("Parse error: {0}")]
    ParseFailed(String),
}
```

```rust
use serde::Serialize;

#[derive(Debug, Clone, Serialize)]
pub struct SmartUpgradeResult {
    pub package: String,
    pub from_version: String,
    pub to_version: String,
    pub steps_applied: Vec<UpgradeStep>,
    pub tests_passed: bool,
    pub rollback_applied: bool,
}

#[derive(Debug, Clone, Serialize)]
pub struct UpgradeStep {
    pub description: String,
    pub files_modified: Vec<String>,
    pub status: StepStatus,
}

#[derive(Debug, Clone, Serialize)]
pub enum StepStatus { Applied, Skipped, Failed(String) }

/// Perform a smart upgrade: analyze -> migrate -> test -> commit/rollback
pub async fn smart_upgrade(
    project_root: &std::path::Path,
    package: &str,
    target_version: Option<&str>,
) -> Result<SmartUpgradeResult, UpgradeError> {
    // 1. Determine current and target versions
    let current = get_installed_version(project_root, package)?;
    let target = match target_version {
        Some(v) => v.to_string(),
        None => get_latest_version(package).await?,
    };

    // 2. Analyze changelog for breaking changes
    let analysis = analyze_changelog(package, &current, &target, "npm").await?;

    // 3. If high risk, warn user
    if matches!(analysis.risk_level, RiskLevel::High | RiskLevel::Critical) {
        println!("Warning: {} -> {} has {} breaking changes (risk: {:?})",
            current, target, analysis.breaking_changes.len(), analysis.risk_level);
        println!("Breaking changes:");
        for bc in &analysis.breaking_changes {
            println!("  - {}", bc.description);
        }
    }

    // 4. Create git checkpoint
    create_checkpoint(project_root)?;

    // 5. Update the package
    update_package(project_root, package, &target).await?;

    // 6. Apply automated migration steps
    let mut steps = vec![];
    for migration in &analysis.migration_steps {
        if migration.automated {
            let result = apply_migration(project_root, migration)?;
            steps.push(result);
        }
    }

    // 7. Run tests
    let tests_passed = run_tests(project_root).await?;

    if !tests_passed {
        // 8. Rollback
        rollback_to_checkpoint(project_root)?;
        return Ok(SmartUpgradeResult {
            package: package.to_string(),
            from_version: current,
            to_version: target,
            steps_applied: steps,
            tests_passed: false,
            rollback_applied: true,
        });
    }

    // 9. Commit
    commit_upgrade(project_root, package, &current, &target)?;

    Ok(SmartUpgradeResult {
        package: package.to_string(),
        from_version: current,
        to_version: target,
        steps_applied: steps,
        tests_passed: true,
        rollback_applied: false,
    })
}
```

```rust
    pub transitive_packages_removed: usize,
    pub size_reduction_bytes: u64,

pub enum ImpactRisk { Safe, Moderate, High, Critical }

    pub name: String,
    pub description: String,

    pub size_bytes: u64,
    pub migration_effort: MigrationEffort,

#[derive(Debug, Clone, Serialize)]
pub enum MigrationEffort { Trivial, Easy, Moderate, Hard }

/// Analyze the impact of a dependency in the project
pub async fn analyze_impact(
    project_root: &std::path::Path,
    package: &str,
) -> Result<ImpactAnalysis, ImpactError> {
    // 1. Scan source files for imports/requires of this package
    let import_scan = scan_imports(project_root, package)?;

    // 2. Analyze dependency graph for transitive dependents
    let graph_analysis = analyze_graph_position(project_root, package)?;

    // 3. Determine what would break if removed
    let removal_impact = simulate_removal(project_root, package, &import_scan, &graph_analysis)?;

    // 4. Find alternatives
    let alternatives = find_alternatives(package).await?;

    Ok(ImpactAnalysis {
        package: package.to_string(),
        version: graph_analysis.version,
        usage: UsageAnalysis {
            direct_imports: import_scan.files.len(),
            transitive_dependents: graph_analysis.dependents.len(),
            import_files: import_scan.files,
            imported_exports: import_scan.exports_used,
            depth_in_graph: graph_analysis.depth,
            dependents: graph_analysis.dependents,
        },
        removal_impact,
        alternatives,
    })
}

/// Scan source files for import/require statements matching a package
fn scan_imports(
    project_root: &std::path::Path,
    package: &str,
) -> Result<ImportScan, ImpactError> {
    let mut files = vec![];
    let mut exports_used = vec![];

    // Walk source files (.ts, .js, .tsx, .jsx, .py, .rs, .rb, .go, .php)
    for entry in walkdir::WalkDir::new(project_root)
        .max_depth(10)
        .into_iter()
        .filter_map(|e| e.ok())
        .filter(|e| is_source_file(e.path()))
    {
        let content = std::fs::read_to_string(entry.path())?;

        // JavaScript/TypeScript: import ... from 'package' or require('package')
        // Python: import package or from package import ...
        // Rust: use package::...
        // Go: import "package"
        if content.contains(package) {
            files.push(entry.path().display().to_string());

            // Extract which exports are used
            let used = extract_used_exports(&content, package);
            exports_used.extend(used);
        }
    }

    exports_used.sort();
    exports_used.dedup();

    Ok(ImportScan { files, exports_used })
}

struct ImportScan {
    files: Vec<String>,
    exports_used: Vec<String>,
}

#[derive(Debug, thiserror::Error)]
pub enum ImpactError {
    #[error("Analysis failed: {0}")]
    AnalysisFailed(String),
    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),
}
```

```rust
// crates/better-core/src/intelligence/audit_fix.rs

use serde::Serialize;

#[derive(Debug, Clone, Serialize)]
pub struct AuditFixResult {
    pub fixes_attempted: usize,
    pub fixes_applied: usize,
    pub fixes_rolled_back: usize,
    pub remaining_vulns: usize,
    pub details: Vec<FixAttempt>,
}

#[derive(Debug, Clone, Serialize)]
pub struct FixAttempt {
    pub vulnerability: String,
    pub package: String,
    pub from_version: String,
    pub to_version: String,
    pub status: FixStatus,
}

#[derive(Debug, Clone, Serialize)]
pub enum FixStatus {
    Applied,                // Upgraded, tests passed, committed
    RolledBack(String),     // Upgraded, tests failed, rolled back (reason)
    NoFix,                  // No patched version available
    Skipped(String),        // Skipped (reason)
}

/// For each vulnerability with a fix available:
/// 1. Create checkpoint
/// 2. Upgrade to patched version
/// 3. Run tests
/// 4. If pass: commit
/// 5. If fail: rollback
pub async fn audit_fix(
    project_root: &std::path::Path,
    auto_commit: bool,
) -> Result<AuditFixResult, FixError> {
    let audit = run_audit(project_root).await?;
    let mut details = vec![];
    let mut applied = 0;
    let mut rolled_back = 0;

    for vuln in audit.vulnerabilities {
        let fixed_version = match &vuln.fixed_in {
            Some(v) => v.clone(),
            None => {
                details.push(FixAttempt {
                    vulnerability: vuln.id.clone(),
                    package: vuln.package.clone(),
                    from_version: vuln.installed_version.clone(),
                    to_version: String::new(),
                    status: FixStatus::NoFix,
                });
                continue;
            }
        };

        // 1. Checkpoint
        create_checkpoint(project_root)?;

        // 2. Upgrade
        upgrade_package(project_root, &vuln.package, &fixed_version).await?;

        // 3. Test
        let tests_ok = run_tests(project_root).await.unwrap_or(false);

        if tests_ok {
            if auto_commit {
                commit_fix(project_root, &vuln)?;
            }
            applied += 1;
            details.push(FixAttempt {
                vulnerability: vuln.id.clone(),
                package: vuln.package.clone(),
                from_version: vuln.installed_version.clone(),
                to_version: fixed_version,
                status: FixStatus::Applied,
            });
        } else {
            rollback_to_checkpoint(project_root)?;
            rolled_back += 1;
            details.push(FixAttempt {
                vulnerability: vuln.id.clone(),
                package: vuln.package.clone(),
                from_version: vuln.installed_version.clone(),
                to_version: fixed_version,
                status: FixStatus::RolledBack("Tests failed after upgrade".into()),
            });
        }
    }

    Ok(AuditFixResult {
        fixes_attempted: details.len(),
        fixes_applied: applied,
        fixes_rolled_back: rolled_back,
        remaining_vulns: audit.total - applied,
        details,
    })
}
```

```rust
// crates/better-core/src/intelligence/predict.rs

use serde::Serialize;

#[derive(Debug, Clone, Serialize)]
pub struct MaintenancePrediction {
    pub package: String,
    pub current_status: MaintenanceStatus,
    pub predicted_status_6mo: MaintenanceStatus,
    pub confidence: f64,            // 0.0 - 1.0
    pub signals: Vec<PredictionSignal>,
    pub recommended_action: Action,
    pub alternatives: Vec<AlternativeSuggestion>,
}

#[derive(Debug, Clone, Serialize)]
pub enum MaintenanceStatus {
    Active,
    SlowingDown,
    AtRisk,
    Unmaintained,
    Deprecated,
}

#[derive(Debug, Clone, Serialize)]
pub struct PredictionSignal {
    pub signal: String,
    pub trend: Trend,
    pub weight: f64,
}

#[derive(Debug, Clone, Serialize)]
pub enum Trend { Improving, Stable, Declining, Critical }

#[derive(Debug, Clone, Serialize)]
pub enum Action {
    NoAction,
    Monitor,
    PlanMigration { to: String, effort: String },
    MigrateNow { to: String, reason: String },
}

#[derive(Debug, Clone, Serialize)]
pub struct AlternativeSuggestion {
    pub package: String,
    pub reason: String,
    pub reputation_score: u8,
    pub api_compatibility: f64,   // 0.0 - 1.0, how similar the API is
}

/// Analyze historical data and predict future maintenance status
pub async fn predict_maintenance(
    package: &str,
    ecosystem: &str,
) -> Result<MaintenancePrediction, PredictError> {
    // Collect historical data points (6 months of snapshots)
    let history = fetch_historical_signals(package, ecosystem, 180).await?;

    // Analyze trends
    let commit_trend = analyze_trend(&history.commit_counts);
    let publish_trend = analyze_trend(&history.publish_counts);
    let issue_response_trend = analyze_trend(&history.issue_response_times);
    let download_trend = analyze_trend(&history.download_counts);

    let mut signals = vec![];
    let mut risk_score = 0.0;

    // Declining commits
    if matches!(commit_trend, Trend::Declining) {
        risk_score += 0.3;
        signals.push(PredictionSignal {
            signal: "Commit frequency declining".into(),
            trend: commit_trend, weight: 0.3,
        });
    }

    // No recent publishes
    if matches!(publish_trend, Trend::Critical) {
        risk_score += 0.3;
        signals.push(PredictionSignal {
            signal: "No releases in extended period".into(),
            trend: publish_trend, weight: 0.3,
        });
    }

    // Issue response time increasing
    if matches!(issue_response_trend, Trend::Declining) {
        risk_score += 0.2;
        signals.push(PredictionSignal {
            signal: "Issue response time increasing".into(),
            trend: issue_response_trend, weight: 0.2,
        });
    }

    let predicted = if risk_score > 0.7 { MaintenanceStatus::Unmaintained }
        else if risk_score > 0.5 { MaintenanceStatus::AtRisk }
        else if risk_score > 0.3 { MaintenanceStatus::SlowingDown }
        else { MaintenanceStatus::Active };

    let alternatives = find_alternatives_with_compatibility(package, ecosystem).await?;

    let action = if risk_score > 0.7 {
        Action::MigrateNow {
            to: alternatives.first().map(|a| a.package.clone()).unwrap_or_default(),
            reason: "Package predicted unmaintained within 6 months".into(),
        }
    } else if risk_score > 0.5 {
        Action::PlanMigration {
            to: alternatives.first().map(|a| a.package.clone()).unwrap_or_default(),
            effort: "medium".into(),
        }
    } else if risk_score > 0.3 {
        Action::Monitor
    } else {
        Action::NoAction
    };

    Ok(MaintenancePrediction {
        package: package.to_string(),
        current_status: determine_current_status(&history),
        predicted_status_6mo: predicted,
        confidence: 0.6 + (history.data_points as f64 / 180.0 * 0.3),
        signals,
        recommended_action: action,
        alternatives,
    })
}

#[derive(Debug, thiserror::Error)]
pub enum PredictError {
    #[error("Insufficient data for prediction: {0}")]
    InsufficientData(String),
    #[error("Fetch error: {0}")]
    FetchFailed(String),
}
```
