// crates/better-core/src/deploy/provision.rs

use serde::Serialize;
use std::collections::HashMap;
use std::path::Path;

#[derive(Debug, Clone, Serialize)]
pub struct PreDeployResult {
    pub services_provisioned: Vec<ProvisionedService>,
    pub env_vars_injected: usize,
    pub total_cost_estimate: CostEstimate,
}

#[derive(Debug, Clone, Serialize)]
pub struct ProvisionedService {
    pub service: String,           // "supabase/postgres"
    pub tier: String,              // "free"
    pub status: String,            // "provisioned", "already_exists"
    pub credentials_injected: Vec<String>, // env var names
}

#[derive(Debug, Clone, Serialize)]
pub struct CostEstimate {
    pub monthly_usd: f64,
    pub breakdown: Vec<CostBreakdown>,
}

#[derive(Debug, Clone, Serialize)]
pub struct CostBreakdown {
    pub service: String,
    pub tier: String,
    pub monthly_usd: f64,
}

/// Read .env.osp, provision any unprovisioned services, inject creds
pub async fn pre_deploy_provision(
    project_root: &Path,
    environment: &str,
) -> Result<PreDeployResult, DeployError> {
    let env_osp_path = project_root.join(".env.osp");
    if !env_osp_path.exists() {
        return Ok(PreDeployResult {
            services_provisioned: vec![],
            env_vars_injected: 0,
            total_cost_estimate: CostEstimate { monthly_usd: 0.0, breakdown: vec![] },
        });
    }

    let env_osp = parse_env_osp(&env_osp_path)?;
    let mut provisioned = vec![];
    let mut env_vars: HashMap<String, String> = HashMap::new();

    for (var_name, osp_uri) in &env_osp {
        // Parse osp://provider.com/service/credential_path
        let parsed = parse_osp_uri(osp_uri)?;

        // Check if already provisioned in vault
        if let Some(value) = vault_lookup(&parsed, environment).await? {
            env_vars.insert(var_name.clone(), value);
            continue;
        }

        // Provision the service
        let result = provision_service(&parsed, environment).await?;
        provisioned.push(ProvisionedService {
            service: parsed.service_id.clone(),
            tier: parsed.tier.clone().unwrap_or("free".into()),
            status: "provisioned".into(),
            credentials_injected: result.credentials.keys().cloned().collect(),
        });

        // Store credentials and inject
        for (cred_key, cred_value) in &result.credentials {
            if cred_key == &parsed.credential_path {
                env_vars.insert(var_name.clone(), cred_value.clone());
            }
        }
    }

    // Write .env for the deploy
    let env_file = project_root.join(format!(".env.{}", environment));
    let env_content: String = env_vars.iter()
        .map(|(k, v)| format!("{}={}", k, v))
        .collect::<Vec<_>>()
        .join("\n");
    std::fs::write(&env_file, env_content)?;

    Ok(PreDeployResult {
        services_provisioned: provisioned,
        env_vars_injected: env_vars.len(),
        total_cost_estimate: estimate_costs(&provisioned).await?,
    })
}

/// Parse .env.osp file: KEY=osp://provider.com/service/path
fn parse_env_osp(path: &Path) -> Result<Vec<(String, String)>, DeployError> {
    let content = std::fs::read_to_string(path)?;
    Ok(content.lines()
        .filter(|l| !l.starts_with('#') && l.contains('='))
        .filter_map(|l| {
            let (key, value) = l.split_once('=')?;
            if value.starts_with("osp://") {
                Some((key.trim().to_string(), value.trim().to_string()))
            } else {
                None
            }
        })
        .collect())
}

// crates/better-core/src/services.rs

use serde::{Deserialize, Serialize};
use std::collections::HashMap;

/// Services declared in package.json or better.toml
/// Example in package.json:
/// {
///   "better": {
///     "services": {
///       "db": "osp://supabase.com/postgres@free",
///       "cache": "osp://upstash.com/redis@pro",
///       "auth": "osp://clerk.com/auth@free"
///     }
///   }
/// }
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ServiceDependencies {
    pub services: HashMap<String, ServiceSpec>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ServiceSpec {
    pub provider: String,        // "supabase.com"
    pub service: String,         // "postgres"
    pub tier: String,            // "free", "pro"
    pub config: HashMap<String, serde_json::Value>,
    pub env_mapping: HashMap<String, String>,  // "connection_string" -> "DATABASE_URL"
}

/// Parse osp:// URI: osp://provider.com/service@tier
pub fn parse_service_uri(uri: &str) -> Result<ServiceSpec, ServiceError> {
    if !uri.starts_with("osp://") {
        return Err(ServiceError::InvalidUri(uri.to_string()));
    }
    let rest = &uri[6..];  // after "osp://"

    let (provider_service, tier) = if rest.contains('@') {
        let parts: Vec<&str> = rest.splitn(2, '@').collect();
        (parts[0], parts[1].to_string())
    } else {
        (rest, "free".to_string())
    };

    let parts: Vec<&str> = provider_service.splitn(2, '/').collect();
    if parts.len() != 2 {
        return Err(ServiceError::InvalidUri(uri.to_string()));
    }

    Ok(ServiceSpec {
        provider: parts[0].to_string(),
        service: parts[1].to_string(),
        tier,
        config: HashMap::new(),
        env_mapping: HashMap::new(),
    })
}

/// During `better install`, resolve and provision service dependencies
pub async fn resolve_services(
    services: &ServiceDependencies,
    environment: &str,
) -> Result<Vec<ResolvedService>, ServiceError> {
    let mut resolved = vec![];

    for (alias, spec) in &services.services {
        // Check if already provisioned
        let existing = check_vault(spec, environment).await?;
        if let Some(creds) = existing {
            resolved.push(ResolvedService {
                alias: alias.clone(),
                spec: spec.clone(),
                status: ServiceStatus::Active,
                credentials: creds,
            });
            continue;
        }

        // Provision via OSP
        let result = provision_via_osp(spec, environment).await?;
        resolved.push(ResolvedService {
            alias: alias.clone(),
            spec: spec.clone(),
            status: ServiceStatus::JustProvisioned,
            credentials: result.credentials,
        });
    }

    Ok(resolved)
}

#[derive(Debug, Clone)]
pub struct ResolvedService {
    pub alias: String,
    pub spec: ServiceSpec,
    pub status: ServiceStatus,
    pub credentials: HashMap<String, String>,
}

#[derive(Debug, Clone)]
pub enum ServiceStatus {
    Active,
    JustProvisioned,
    Failed(String),
}

#[derive(Debug, thiserror::Error)]
pub enum ServiceError {
    #[error("Invalid service URI: {0}")]
    InvalidUri(String),
    #[error("Provision failed: {0}")]
    ProvisionFailed(String),
    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),
}

// crates/better-core/src/env_manager.rs

use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use std::path::{Path, PathBuf};

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct EnvironmentConfig {
    pub environments: HashMap<String, Environment>,
    pub current: String,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct Environment {
    pub name: String,
    pub services: HashMap<String, ServiceInstance>,
    pub env_vars: HashMap<String, String>,
    pub created_at: u64,
    pub last_active: u64,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ServiceInstance {
    pub service_id: String,       // OSP service ID
    pub provider: String,
    pub tier: String,
    pub status: String,
    pub credentials_ref: String,  // vault reference
}

pub struct EnvManager {
    config_path: PathBuf,
    vault_path: PathBuf,
}

impl EnvManager {
    pub fn new(project_root: &Path) -> Self {
        Self {
            config_path: project_root.join(".better").join("environments.json"),
            vault_path: dirs::config_dir()
                .unwrap_or_default()
                .join("better")
                .join("vault"),
        }
    }

    /// Switch to a different environment
    pub fn switch(&self, env_name: &str) -> Result<Environment, EnvError> {
        let mut config = self.load_config()?;
        let env = config.environments.get(env_name)
            .ok_or_else(|| EnvError::NotFound(env_name.to_string()))?
            .clone();

        config.current = env_name.to_string();
        self.save_config(&config)?;

        // Regenerate .env from this environment's vault entries
        self.generate_env_file(&env)?;

        Ok(env)
    }

    /// Create a new environment
    pub fn create(&self, name: &str) -> Result<Environment, EnvError> {
        let mut config = self.load_config()?;
        if config.environments.contains_key(name) {
            return Err(EnvError::AlreadyExists(name.to_string()));
        }

        let env = Environment {
            name: name.to_string(),
            services: HashMap::new(),
            env_vars: HashMap::new(),
            created_at: current_timestamp(),
            last_active: current_timestamp(),
        };

        config.environments.insert(name.to_string(), env.clone());
        self.save_config(&config)?;
        Ok(env)
    }

    /// Clone an environment (provision new service instances)
    pub async fn clone_env(
        &self,
        source: &str,
        target: &str,
    ) -> Result<Environment, EnvError> {
        let config = self.load_config()?;
        let source_env = config.environments.get(source)
            .ok_or_else(|| EnvError::NotFound(source.to_string()))?;

        let mut target_env = Environment {
            name: target.to_string(),
            services: HashMap::new(),
            env_vars: HashMap::new(),
            created_at: current_timestamp(),
            last_active: current_timestamp(),
        };

        // Provision new instances of each service
        for (alias, service) in &source_env.services {
            let new_instance = provision_for_env(
                &service.provider,
                &service.tier,
                target,
            ).await?;
            target_env.services.insert(alias.clone(), new_instance);
        }

        // Copy non-secret env vars
        target_env.env_vars = source_env.env_vars.clone();

        Ok(target_env)
    }

    /// List all environments with status
    pub fn list(&self) -> Result<Vec<EnvSummary>, EnvError> {
        let config = self.load_config()?;
        Ok(config.environments.values()
            .map(|env| EnvSummary {
                name: env.name.clone(),
                is_current: env.name == config.current,
                service_count: env.services.len(),
                last_active: env.last_active,
            })
            .collect())
    }
}

#[derive(Debug, Clone, Serialize)]
pub struct EnvSummary {
    pub name: String,
    pub is_current: bool,
    pub service_count: usize,
    pub last_active: u64,
}

#[derive(Debug, thiserror::Error)]
pub enum EnvError {
    #[error("Environment not found: {0}")]
    NotFound(String),
    #[error("Environment already exists: {0}")]
    AlreadyExists(String),
    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),
}

use std::time::Duration;

#[derive(Debug, Clone, Serialize)]
pub struct PreviewDeployment {
    pub preview_id: String,
    pub url: String,
    pub services: Vec<EphemeralService>,
    pub ttl: Duration,
    pub expires_at: u64,
    pub pr_number: Option<u32>,

#[derive(Debug, Clone, Serialize)]
pub struct EphemeralService {
    pub service_id: String,
    pub provider: String,
    pub service_type: String,
    pub status: String,
    pub ttl: Duration,
    pub auto_deprovision: bool,
}

/// Create a preview deployment with ephemeral OSP services
pub async fn create_preview(
    project_root: &std::path::Path,
    branch: &str,
    pr_number: Option<u32>,
    ttl: Duration,
) -> Result<PreviewDeployment, PreviewError> {
    let preview_id = generate_preview_id(branch);

    // 1. Provision ephemeral services with TTL
    let env_osp = parse_env_osp(&project_root.join(".env.osp"))?;
    let mut services = vec![];

    for (_var, uri) in &env_osp {
        let spec = parse_service_uri(uri)?;
        let ephemeral = provision_ephemeral(
            &spec,
            &preview_id,
            ttl,
        ).await?;
        services.push(ephemeral);
    }

    // 2. Generate .env for preview
    let env_vars = collect_preview_env_vars(&services)?;

    // 3. Deploy to preview URL
    let deploy_result = deploy_preview(project_root, &preview_id, &env_vars).await?;

    // 4. Schedule auto-deprovision
    schedule_deprovision(&preview_id, ttl).await?;

    let expires_at = current_timestamp() + ttl.as_secs();

    Ok(PreviewDeployment {
        preview_id,
        url: deploy_result.url,
        services,
        ttl,
        expires_at,
        pr_number,
        branch: branch.to_string(),
    })
}

/// Clean up expired preview deployments
pub async fn cleanup_expired_previews() -> Result<usize, PreviewError> {
    let previews = list_active_previews().await?;
    let mut cleaned = 0;

    for preview in previews {
        if current_timestamp() > preview.expires_at {
            // Deprovision all ephemeral services
            for service in &preview.services {
                deprovision_service(&service.service_id).await?;
            }
            // Remove preview deployment
            remove_preview_deployment(&preview.preview_id).await?;
            cleaned += 1;
        }
    }

    Ok(cleaned)
}

fn generate_preview_id(branch: &str) -> String {
    let sanitized = branch.replace('/', "-").to_lowercase();
    let short_hash = &sha256_hex(branch.as_bytes())[..8];
    format!("preview-{}-{}", sanitized, short_hash)
}

    DeployFailed(String),
    #[error("Service error: {0}")]
    ServiceError(String),
    #[error("IO error: {0}")]
    Io(#[from] std::io::Error),
}

#[derive(Debug, Clone, Serialize)]
pub struct CostReport {

#[derive(Debug, Clone, Serialize)]
pub struct ServiceCost {
    pub provider: String,
    pub service: String,
    pub tier: String,
    pub environment: String,
    pub monthly_usd: f64,
    pub usage_pct: f64,      // % of tier limit used
}

#[derive(Debug, Clone, Serialize)]
pub struct CostTrend {
    pub current_month: f64,
    pub previous_month: f64,
    pub change_pct: f64,
    pub projected_month_end: f64,
}

#[derive(Debug, Clone, Serialize)]
pub struct CostOptimization {
    pub suggestion: String,
    pub savings_usd: f64,
    pub service: String,
    pub action: OptimizationAction,
}

#[derive(Debug, Clone, Serialize)]
pub enum OptimizationAction {
    Downgrade { from_tier: String, to_tier: String },
    Consolidate { services: Vec<String>, into: String },
    Remove { reason: String },
    SwitchProvider { from: String, to: String },
}

/// Aggregate costs across all provisioned OSP services
pub async fn generate_cost_report() -> Result<CostReport, CostError> {
    let environments = list_all_environments().await?;
    let mut by_service = vec![];
    let mut by_environment: HashMap<String, f64> = HashMap::new();
    let mut total = 0.0;

    for env in &environments {
        let mut env_total = 0.0;
        for (_, service) in &env.services {
            let cost = query_service_cost(service).await?;
            env_total += cost.monthly_usd;
            by_service.push(ServiceCost {
                provider: service.provider.clone(),
                service: service.service_id.clone(),
                tier: service.tier.clone(),
                environment: env.name.clone(),
                monthly_usd: cost.monthly_usd,
                usage_pct: cost.usage_pct,
            });
        }
        by_environment.insert(env.name.clone(), env_total);
        total += env_total;
    }

    let optimizations = generate_optimizations(&by_service);

    Ok(CostReport {
        total_monthly_usd: total,
        by_environment,
        by_service,
        by_project: HashMap::new(),
        trend: calculate_trend(total).await?,
        optimizations,
    })
}

fn generate_optimizations(services: &[ServiceCost]) -> Vec<CostOptimization> {
    let mut suggestions = vec![];

    for service in services {
        // Suggest downgrade if usage is very low
        if service.usage_pct < 10.0 && service.tier != "free" {
            suggestions.push(CostOptimization {
                suggestion: format!(
                    "{}/{} on {} tier but only {}% utilization — consider free tier",
                    service.provider, service.service, service.tier, service.usage_pct
                ),
                savings_usd: service.monthly_usd,
                service: format!("{}/{}", service.provider, service.service),
                action: OptimizationAction::Downgrade {
                    from_tier: service.tier.clone(),
                    to_tier: "free".into(),
                },
            });
        }
    }

    suggestions
}

#[derive(Debug, thiserror::Error)]
pub enum CostError {
    #[error("Failed to fetch costs: {0}")]
    FetchFailed(String),
}
